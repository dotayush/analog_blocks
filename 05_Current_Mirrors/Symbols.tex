\section*{Symbols and Labels}
\addcontentsline{toc}{section}{Symbols and Labels}
Every symbol and label that appears in this chapter. The index \(k\) runs over the transistors \(M_0 \ldots M_3\) of the mirror being analysed. Upper case node subscripts are large signal (Q5), lower case are small signal (Q4).
\begin{symtable}
\multicolumn{3}{|l|}{\textbf{Circuit}} \\ \hline
\(M_k\) & transistor \(k\) (\(M_0\) the diode connected input device) & - \\ \hline
\(V_{cc},\ V_{ee}\) & positive and negative supply & V \\ \hline
\(G_2,\ R_2\) & input resistor (basic, Widlar) that sets the input current from \(V_{cc}\) & S, \(\Omega\) \\ \hline
\(G_3,\ R_3\) & output load used only for analysis (basic, Widlar); input resistor (Wilson) & S, \(\Omega\) \\ \hline
\(G_4,\ R_4\) & output load (Wilson); input resistor (modified Wilson, cascode) & S, \(\Omega\) \\ \hline
\(G_5\) & output load used only for analysis (modified Wilson, cascode) & S \\ \hline
\(G_S,\ R_S\) & source degeneration of the Widlar mirror & S, \(\Omega\) \\ \hline
\(R_L,\ R_{L,\max}\) & load resistance and its largest value keeping the output device saturated & \(\Omega\) \\ \hline
\(V_{L,\max}\) & largest voltage the load may drop, \(V_{cc} - V_{OUT,\min}\) & V \\ \hline
\(W_k,\ L_k,\ L_{\min}\) & width and length of \(M_k\); smallest allowed length & m \\ \hline
\(K_P,\ K_{CLM}\) & process transconductance and channel length modulation coefficient & A/V\(^2\), V\(^{-1}\) \\ \hline
\(k_n,\ k_{n_k}\) & device factor \(\frac{K_P}{2}\frac{W}{L}\), of \(M_k\) & A/V\(^2\) \\ \hline
\(V_{th}\) & threshold voltage & V \\ \hline
\multicolumn{3}{|l|}{\textbf{Large Signal (Q5)}} \\ \hline
\(V_X, V_Y, V_Z\) & bias voltages of the internal nodes \(V_x, V_y, V_z\) of the schematic & V \\ \hline
\(V_S\) & bias voltage of the Widlar source node & V \\ \hline
\(V_{OUT},\ V_{OUT,\min},\ V_{OUT,\max}\) & bias output voltage and its saturation limits & V \\ \hline
\(I_{IN},\ I_{OUT}\) & bias input and output currents & A \\ \hline
\(I_{M_k}\) & drain current of \(M_k\) & A \\ \hline
\(I_G,\ I_{G_1},\ I_{G_2}\) & gate current (zero for MOS), into the gates at node 1 and node 2 & A \\ \hline
\(I_D\) & drain current in general & A \\ \hline
\(V_{GS},\ V_{DS},\ V_{GS_k},\ V_{DS_k}\) & gate-source and drain-source voltage, of \(M_k\) & V \\ \hline
\(V_{ov},\ V_{ov_k}\) & overdrive \(V_{GS} - V_{th}\), of \(M_k\) & V \\ \hline
\(u,\ a, b, c\) & \(u = \sqrt{I_{OUT}}\) and the quadratic coefficients of the Widlar equation & A\(^{1/2}\), - \\ \hline
\(dI_{M_1},\ dV_{OUT}\) & small changes used to find \(g_{o_1}\) from the Q5 ratio & A, V \\ \hline
\multicolumn{3}{|l|}{\textbf{Small Signal (Q4)}} \\ \hline
\(V_x, V_y, V_z,\ V_s,\ V_{out}\) & small signal node voltages (Widlar \(V_s\) is the source node) & V \\ \hline
\(V_{gs},\ V_{gs_0},\ V_{ds_0}\) & small signal gate-source and drain-source voltages & V \\ \hline
\(I_{in},\ I_{out}\) & small signal input and output currents & A \\ \hline
\(I_{test}\) & test current used to find an impedance & A \\ \hline
\(V_s,\ I_s\) & (basic mirror) a test source forcing \(V_x\) and the current it supplies & V, A \\ \hline
\(g_m, g_{m_k},\ g_o, g_{o_k},\ g_{mb}\) & transconductance, output conductance (of \(M_k\)), body transconductance & S \\ \hline
\(r_{o_k}\) & output resistance \(\frac{1}{g_{o_k}}\) & \(\Omega\) \\ \hline
\(H(s),\ N(s),\ D(s)\) & voltage transfer function, numerator, denominator & - \\ \hline
\(Z_{in},\ Z_{out}\) & input and output impedance & \(\Omega\) \\ \hline
\(Z_{out,basic},\ Z_{out,mirror}\) & output impedance of a basic mirror; of the mirror alone (load removed) & \(\Omega\) \\ \hline
\(S\) & Wilson, modified Wilson: \(g_{o_1} + g_{m_1} + g_{o_2} + g_{m_2}\); cascode: \(g_{m_0} + g_{o_0} + g_{m_3} + g_{o_3}\) & S \\ \hline
\(H\) & Wilson: \(G_3 + g_{o_0}\) & S \\ \hline
\(P\) & modified Wilson: \(g_{m_3} + g_{o_3}\); cascode: \(g_{m_2} + g_{o_1} + g_{o_2}\) & S \\ \hline
\(M\) & modified Wilson: \(G_4(P + g_{o_0}) + g_{o_0} P\) & S\(^2\) \\ \hline
\(a\) & cascode: \(\frac{g_{m_3} + g_{o_3}}{S}\), so \(V_x = aV_z\) & - \\ \hline
\(s\) & Laplace variable & rad/s \\ \hline
\multicolumn{3}{|l|}{\textbf{Units}} \\ \hline
\(\mu A,\ \mu m,\ \mu S,\ mS,\ k\Omega,\ M\Omega,\ G\Omega\) & micro (\(10^{-6}\)), milli (\(10^{-3}\)), kilo, mega, giga prefixes & - \\ \hline
\(aS,\ aV_z\) & products of \(a\) with \(S\), \(V_z\) & S, V \\ \hline
\(X\) & column of the voltage table naming each terminal & - \\ \hline
\(KCL\) & Kirchhoff's Current Law & - \\ \hline
\end{symtable}
