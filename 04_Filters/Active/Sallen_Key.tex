\subsection{Sallen-Key (Unity Gain) Low Pass and High Pass}\label{sec:filt_sallen_key}

\subsubsection{Overview}
Two RC sections with the op-amp as a unity gain buffer; positive feedback
through \(C_1\) from the output to the middle node creates complex poles.

\subsubsection{Transfer Function}
\textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)} (linear elements only; at DC
the capacitors are open and \(V_{OUT} = V_{IN}\)) with an ideal op-amp wired as a buffer: \(V_- = V_{out}\) and \(V_+ = V_-\), so
\(V_{out} = V_y\); no current enters \(V_+\).\\
KCL at \(V_y\):
\begin{align*}
  (V_y - V_x) G_2 + (V_y - 0) sC_2 &= 0 \\
  V_y G_2 + V_y sC_2 &= V_x G_2 \\
  V_x &= V_y \frac{G_2 + sC_2}{G_2} \tag{1}
\end{align*}
Subtracting \(V_y\) from (1):
\begin{align*}
  V_x - V_y &= V_y \frac{G_2 + sC_2}{G_2} - V_y\frac{G_2}{G_2} = V_y\frac{sC_2}{G_2} \tag{1a}
\end{align*}
KCL at \(V_x\) (\(C_1\) returns to \(V_{out} = V_y\)):
\begin{align*}
  (V_x - V_{in}) G_1 + (V_x - V_y) G_2 + (V_x - V_y) sC_1 &= 0 \tag{2}
\end{align*}
Substituting (1) for \(V_x\) in the first term and (1a) for \(V_x - V_y\) in the
other two:
\begin{align*}
  V_y\frac{(G_2 + sC_2) G_1}{G_2} - V_{in} G_1 + V_y\frac{sC_2}{G_2} G_2 + V_y\frac{sC_2}{G_2} sC_1 &= 0 \\
  V_y\frac{(G_2 + sC_2) G_1}{G_2} - V_{in} G_1 + V_y sC_2 + V_y\frac{s^2 C_1 C_2}{G_2} &= 0 \\
  V_y\left((G_2 + sC_2) G_1 + sC_2 G_2 + s^2 C_1 C_2\right) - V_{in} G_1 G_2 &= 0 \hspace{1cm} [\times G_2] \\
  V_y\left(G_1 G_2 + sC_2 G_1 + sC_2 G_2 + s^2 C_1 C_2\right) &= V_{in} G_1 G_2 \\
  H_{LP}(s) = \frac{V_{out}}{V_{in}} = \frac{V_y}{V_{in}} &= \frac{G_1 G_2}{s^2 C_1 C_2 + s C_2 (G_1 + G_2) + G_1 G_2} \tag{3}
\end{align*}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: (3) with top and bottom divided by \(C_1 C_2\)
    (Section~\ref{sec:theory_cs_second_order}),
    \begin{align*}
      H_{LP}(s) &= \frac{\frac{G_1 G_2}{C_1 C_2}}{s^2 + s\frac{G_1 + G_2}{C_1} + \frac{G_1 G_2}{C_1 C_2}}
      \equiv \frac{\omega_0^2}{s^2 + s\frac{\omega_0}{Q} + \omega_0^2}
    \end{align*}
    Matching coefficients:
    \begin{align*}
      \omega_0^2 &= \frac{G_1 G_2}{C_1 C_2} = \frac{1}{R_1 R_2 C_1 C_2} \hspace{1cm} \frac{\omega_0}{Q} = \frac{G_1 + G_2}{C_1} \\
      Q &= \frac{\omega_0 C_1}{G_1 + G_2} = \frac{C_1 R_1 R_2}{R_1 + R_2}\cdot\frac{1}{\sqrt{R_1 R_2 C_1 C_2}} \hspace{1cm} \left[G_1 + G_2 = \tfrac{R_1 + R_2}{R_1 R_2}\right] \\
      Q &= \frac{\sqrt{R_1 R_2}}{R_1 + R_2}\sqrt{\frac{C_1}{C_2}} \\
      s_p &= -\frac{\omega_0}{2Q} \pm j\omega_0\sqrt{1 - \frac{1}{4Q^2}} \hspace{1cm} [\text{quadratic formula}, Q > \tfrac{1}{2}]
    \end{align*}
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(N(s) = G_1 G_2 \neq 0\): no finite zero; \(-40\,\)dB/decade above \(f_0\).
  \item \textbf{DC Gain (\(H(0)\))}: \(H_{LP}(0) = \frac{G_1 G_2}{0 + 0 + G_1 G_2} = 1\).
  \item \textbf{Gain-Bandwidth Product (\(GBW\))}: \(|H(0)| f_0 = \frac{1}{2\pi\sqrt{R_1 R_2 C_1 C_2}}\) (the \(-3\,\)dB
    point equals \(f_0\) only for Butterworth, \(Q = \frac{1}{\sqrt{2}}\)).
  \item \textbf{Equal Resistors} \(R_1 = R_2 = R\):
    \begin{align*}
      Q &= \frac{\sqrt{R \cdot R}}{R + R}\sqrt{\frac{C_1}{C_2}} = \frac{1}{2}\sqrt{\frac{C_1}{C_2}} \\
      \frac{1}{2}\sqrt{\frac{C_1}{C_2}} &= \frac{1}{\sqrt{2}} \hspace{1cm} \Rightarrow \hspace{1cm} \sqrt{\frac{C_1}{C_2}} = \sqrt{2} \hspace{1cm} \Rightarrow \hspace{1cm} C_1 = 2 C_2 \hspace{1cm} [\text{Butterworth}]
    \end{align*}
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(s \to 0\): \(Z_{in} \to \infty\) (no DC path to ground); \(s \to \infty\): \(C_1\), \(C_2\)
    short \(V_x\) to \(V_{out}\) and \(V_y\) to ground, \(Z_{in} \to R_1\).
    \paragraph{At Output:} the buffer output, \(Z_{out} = \frac{R_o}{1 + A} \approx 0\) (Section~\ref{sec:amp_buffer}).
  \item \textbf{Advantages \& Limitations}: two poles from one unity gain buffer (\(F = 1\), widest
    op-amp bandwidth), \(Q\) set by component ratios, unity DC gain exact; but no gain, \(Q\) is
    sensitive to \(\frac{C_1}{C_2}\), and above \(GB\) the op-amp output impedance lets the signal through
    \(C_1\) (high frequency feedthrough).
\end{enumerate}

\subsubsection{High Pass Transfer Function}
\textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}, buffer
\(V_{out} = V_y\).\\
KCL at \(V_y\):
\begin{align*}
  (V_y - V_x) sC_2 + (V_y - 0) G_2 &= 0 \\
  V_x &= V_y \frac{sC_2 + G_2}{sC_2} \tag{3} \\
  V_x - V_y &= V_y \frac{sC_2 + G_2}{sC_2} - V_y\frac{sC_2}{sC_2} = V_y\frac{G_2}{sC_2} \tag{3a}
\end{align*}
KCL at \(V_x\):
\begin{align*}
  (V_x - V_{in}) sC_1 + (V_x - V_y) sC_2 + (V_x - V_y) G_1 &= 0 \\
  V_y\frac{(sC_2 + G_2) sC_1}{sC_2} - V_{in} sC_1 + V_y\frac{G_2}{sC_2} sC_2 + V_y\frac{G_2}{sC_2} G_1 &= 0 \hspace{1cm} \text{(Substituting (3), (3a))} \\
  V_y\left((sC_2 + G_2) sC_1 + sC_2 G_2 + G_1 G_2\right) - V_{in} s^2 C_1 C_2 &= 0 \hspace{1cm} [\times sC_2] \\
  V_y\left(s^2 C_1 C_2 + sC_1 G_2 + sC_2 G_2 + G_1 G_2\right) &= V_{in} s^2 C_1 C_2 \\
  H_{HP}(s) = \frac{V_{out}}{V_{in}} &= \frac{s^2 C_1 C_2}{s^2 C_1 C_2 + s G_2 (C_1 + C_2) + G_1 G_2}
\end{align*}
\paragraph{High Pass Analysis:}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: top and bottom divided by \(C_1 C_2\),
    \(\omega_0^2 = \frac{G_1 G_2}{C_1 C_2} = \frac{1}{R_1 R_2 C_1 C_2}\) (as the low pass),
    \(\frac{\omega_0}{Q} = \frac{G_2 (C_1 + C_2)}{C_1 C_2}\):
    \begin{align*}
      Q &= \frac{\omega_0 C_1 C_2}{G_2 (C_1 + C_2)} = \frac{C_1 C_2 R_2}{C_1 + C_2}\cdot\frac{1}{\sqrt{R_1 R_2 C_1 C_2}}
         = \frac{\sqrt{C_1 C_2}}{C_1 + C_2}\sqrt{\frac{R_2}{R_1}}
    \end{align*}
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(s_z^2 C_1 C_2 = 0 \Rightarrow s_z = 0\) (double): \(+40\,\)dB/decade below \(f_0\).
  \item \textbf{DC Gain (\(H(0)\))}: \(H_{HP}(0) = 0\); high frequency gain
    \(H_{HP}(s \to \infty) = \frac{s^2 C_1 C_2}{s^2 C_1 C_2} = 1\).
  \item \textbf{Equal Capacitors} \(C\): \(Q = \frac{C}{2C}\sqrt{\frac{R_2}{R_1}} = \frac{1}{2}\sqrt{\frac{R_2}{R_1}}\);
    Butterworth needs \(\sqrt{\frac{R_2}{R_1}} = \sqrt{2}\), \(R_2 = 2R_1\).
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item Low pass: \(R_1 = R_2 = R\), \(C_1 = 4Q^2 C_2\) [from \(Q = \frac{1}{2}\sqrt{\frac{C_1}{C_2}}\)],
    \(C_2 = \frac{1}{2Q\,\omega_0 R}\) [from \(\omega_0 = \frac{1}{R\sqrt{C_1 C_2}} = \frac{1}{2Q R C_2}\)].
  \item High pass: \(C_1 = C_2 = C\), \(R_2 = 4Q^2 R_1\), \(R_1 = \frac{1}{2Q\,\omega_0 C}\).
\end{enumerate}

\subsubsection{Question}
Butterworth low pass, \(f_0 = 1\,\)kHz, \(R_1 = R_2 = 10\,k\Omega\).
\paragraph{Solution:}
Equal resistors so \(C_1 = 2C_2\); then
\begin{align*}
  \omega_0 &= \frac{1}{\sqrt{R \cdot R \cdot 2C_2 \cdot C_2}} = \frac{1}{R\sqrt{2}\,C_2} \\
  C_2 &= \frac{1}{\omega_0 R \sqrt{2}} = \frac{1}{2\pi \cdot 1000\,Hz \cdot 10^4\,\Omega \cdot 1.4142}
       = \frac{1}{8.886 \times 10^7\,\Omega/s} = 11.25\,nF \\
  C_1 &= 2 \cdot 11.25\,nF = 22.5\,nF
\end{align*}
