\subsection{Multiple Feedback (MFB) Band Pass}\label{sec:filt_mfb}

\subsubsection{Overview}
An inverting op-amp with two feedback paths: \(C_2\) from the output to the
middle node \(V_x\) and \(G_2\) from the output to \(V_-\); \(V_x\) reaches \(V_-\)
through \(C_1\).

\subsubsection{Transfer Function}
\textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)} (linear elements only; at DC
\(C_1\), \(C_2\) are open and \(V_{OUT} = 0\)) with an ideal op-amp: \(V_+ = 0\) so \(V_- = 0\), and no
current enters \(V_-\).\\
KCL at \(V_-\):
\begin{align*}
  (V_- - V_x) sC_1 + (V_- - V_{out}) G_2 &= 0 \\
  (0 - V_x) sC_1 + (0 - V_{out}) G_2 &= 0 \\
  -V_x sC_1 &= V_{out} G_2 \\
  V_x &= -\frac{G_2}{sC_1} V_{out} \tag{1}
\end{align*}
KCL at \(V_x\) (\(C_1\) returns to \(V_- = 0\)):
\begin{align*}
  (V_x - V_{in}) G_1 + (V_x - 0) sC_1 + (V_x - V_{out}) sC_2 &= 0 \\
  V_x G_1 + V_x sC_1 + V_x sC_2 - V_{out} sC_2 &= V_{in} G_1 \\
  V_x (G_1 + s(C_1 + C_2)) - V_{out} sC_2 &= V_{in} G_1 \tag{2}
\end{align*}
Substituting (1) into (2):
\begin{align*}
  -\frac{G_2}{sC_1} V_{out}(G_1 + s(C_1 + C_2)) - V_{out} sC_2 &= V_{in} G_1 \\
  -V_{out}\left(\frac{G_2 (G_1 + s(C_1 + C_2))}{sC_1} + \frac{sC_2 \cdot sC_1}{sC_1}\right) &= V_{in} G_1 \\
  -V_{out}\frac{G_1 G_2 + s G_2 (C_1 + C_2) + s^2 C_1 C_2}{sC_1} &= V_{in} G_1 \\
  H(s) = \frac{V_{out}}{V_{in}} &= -\frac{s C_1 G_1}{s^2 C_1 C_2 + s G_2 (C_1 + C_2) + G_1 G_2} \tag{3}
\end{align*}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: (3) with top and bottom divided by \(C_1 C_2\)
    (Section~\ref{sec:theory_cs_second_order}),
    \begin{align*}
      H(s) &= -\frac{s\frac{G_1}{C_2}}{s^2 + s\frac{G_2 (C_1 + C_2)}{C_1 C_2} + \frac{G_1 G_2}{C_1 C_2}}
      \equiv -H_0\frac{s\frac{\omega_0}{Q}}{s^2 + s\frac{\omega_0}{Q} + \omega_0^2} \tag{4}
    \end{align*}
    Matching coefficients:
    \begin{align*}
      \omega_0^2 &= \frac{G_1 G_2}{C_1 C_2} = \frac{1}{R_1 R_2 C_1 C_2} \hspace{1cm}
      \frac{\omega_0}{Q} = \frac{G_2 (C_1 + C_2)}{C_1 C_2} \\
      Q &= \frac{\omega_0 C_1 C_2}{G_2 (C_1 + C_2)} = \frac{C_1 C_2}{G_2 (C_1 + C_2)}\cdot\frac{\sqrt{G_1 G_2}}{\sqrt{C_1 C_2}}
         = \frac{\sqrt{C_1 C_2}}{C_1 + C_2}\sqrt{\frac{R_2}{R_1}} \\
      s_p &= -\frac{\omega_0}{2Q} \pm j\omega_0\sqrt{1 - \frac{1}{4Q^2}} \hspace{1cm} [\text{quadratic formula}, Q > \tfrac{1}{2}]
    \end{align*}
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(s_z C_1 G_1 = 0 \Rightarrow s_z = 0\); first order numerator over
    second order denominator, so the other zero is at \(s \to \infty\): band pass.
  \item \textbf{DC Gain (\(H(0)\))}: \(H(0) = \frac{0}{G_1 G_2} = 0\). Centre gain at \(s = j\omega_0\)
    (\(s^2 C_1 C_2 = -\omega_0^2 C_1 C_2 = -G_1 G_2\) cancels the constant term):
    \begin{align*}
      H(j\omega_0) &= -\frac{j\omega_0 C_1 G_1}{-G_1 G_2 + j\omega_0 G_2 (C_1 + C_2) + G_1 G_2}
       = -\frac{C_1 G_1}{G_2 (C_1 + C_2)} = -\frac{R_2}{R_1}\cdot\frac{C_1}{C_1 + C_2}
    \end{align*}
  \item \textbf{Gain-Bandwidth Product (\(GBW\))}: \(H(0) = 0\), so \(GBW = 0\); the band pass figure is
    the centre gain times the \(-3\,\)dB bandwidth \(BW = \frac{f_0}{Q}\) (as the RLC band filter,
    Section~\ref{sec:filt_rlc_band}):
    \(|H(j\omega_0)| \cdot BW = \frac{R_2 C_1}{R_1(C_1 + C_2)}\cdot\frac{f_0}{Q}\).
  \item \textbf{Equal Capacitors} \(C_1 = C_2 = C\):
    \begin{align*}
      Q &= \frac{\sqrt{C \cdot C}}{C + C}\sqrt{\frac{R_2}{R_1}} = \frac{1}{2}\sqrt{\frac{R_2}{R_1}}
      \hspace{1cm} \Rightarrow \hspace{1cm} \frac{R_2}{R_1} = 4Q^2 \\
      H(j\omega_0) &= -\frac{R_2}{R_1}\cdot\frac{C}{2C} = -\frac{R_2}{2R_1} = -\frac{4Q^2}{2} = -2Q^2
    \end{align*}
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(I_{in} = (V_{in} - V_x) G_1\), with \(V_x\) from (1) and (3):
    \begin{align*}
      Z_{in}(s) &= \frac{s^2 C_1 C_2 + s G_2 (C_1 + C_2) + G_1 G_2}{G_1 s\left(s C_1 C_2 + G_2 (C_1 + C_2)\right)}
    \end{align*}
    \(s \to 0\): \(Z_{in} \to \infty\) (no DC path). \(s = j\omega_0\): numerator \(= j\omega_0 G_2 (C_1 + C_2)\) and
    \(G_2 (C_1 + C_2) = \frac{\omega_0 C_1 C_2}{Q}\) [from \(\frac{\omega_0}{Q}\)]:
    \begin{align*}
      Z_{in}(j\omega_0) &= \frac{G_2 (C_1 + C_2)}{G_1 \left(j\omega_0 C_1 C_2 + G_2 (C_1 + C_2)\right)} = \frac{\frac{\omega_0 C_1 C_2}{Q}}{G_1 \omega_0 C_1 C_2 \left(j + \frac{1}{Q}\right)} = \frac{R_1}{1 + jQ}
    \end{align*}
    \paragraph{At Output:} the op-amp output, \(Z_{out} = \frac{R_o}{1 + AF} \approx 0\) (Section~\ref{sec:amp_inverting}).
  \item \textbf{Advantages \& Limitations}: band pass from one op-amp, low \(Z_{out}\), \(Q\) set by
    \(\sqrt{\frac{R_2}{R_1}}\) (insensitive to op-amp gain at low \(Q\)); but centre gain is tied to \(Q\)
    (\(-2Q^2\) for equal \(C\)), high \(Q\) needs a large resistor spread (\(4Q^2\)), and the op-amp needs
    \(GB \gg 2Q^2 f_0\).
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item Choose \(C_1 = C_2 = C\); \(\frac{R_2}{R_1} = 4Q^2\).
  \item \(R_1 = \frac{1}{2Q\,\omega_0 C}\) [from \(\omega_0 = \frac{1}{\sqrt{R_1 \cdot 4Q^2 R_1}\,C} = \frac{1}{2Q R_1 C}\)], \(R_2 = 4Q^2 R_1\).
\end{enumerate}

\subsubsection{Question}
\(f_0 = 1\,\)kHz, \(Q = 5\), \(C_1 = C_2 = 10\,\)nF.
\paragraph{Solution:}
\begin{align*}
  \frac{R_2}{R_1} &= 4Q^2 = 4 \cdot 5^2 = 100 \hspace{1cm} \Rightarrow \hspace{1cm} R_2 = 100 R_1 \\
  \omega_0 &= \frac{1}{\sqrt{R_1 R_2}\,C} = \frac{1}{\sqrt{R_1 \cdot 100 R_1}\,C} = \frac{1}{10 R_1 C} \\
  R_1 &= \frac{1}{10\,\omega_0 C} = \frac{1}{10 \cdot 2\pi \cdot 1000\,Hz \cdot 10 \times 10^{-9}\,F}
       = \frac{1}{6.283 \times 10^{-4}\,S} = 1.59\,k\Omega \\
  R_2 &= 100 \cdot 1.59\,k\Omega = 159\,k\Omega \\
  H(j\omega_0) &= -2Q^2 = -2 \cdot 25 = -50 \\
  BW &= \frac{f_0}{Q} = \frac{1000\,Hz}{5} = 200\,Hz
\end{align*}
