\subsection{Multiple Feedback (MFB) Band Pass}\label{sec:filt_mfb}

\subsubsection{Overview}
An inverting op-amp with two feedback paths: \(C_2\) from the output to the
middle node \(V_x\) and \(G_2\) from the output to \(V_-\); \(V_x\) reaches \(V_-\)
through \(C_1\).

\subsubsection{Transfer Function}
\textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}, \(V_- = 0\).\\
KCL at \(V_-\):
\begin{align*}
  (0 - V_x) sC_1 + (0 - V_{out}) G_2 &= 0 \hspace{0.5cm} \Rightarrow \hspace{0.5cm} V_x = -\frac{G_2}{sC_1} V_{out} \tag{1}
\end{align*}
KCL at \(V_x\):
\begin{align*}
  (V_x - V_{in}) G_1 + V_x sC_1 + (V_x - V_{out}) sC_2 &= 0 \\
  V_x (G_1 + s(C_1 + C_2)) - V_{out} sC_2 &= V_{in} G_1 \tag{2}
\end{align*}
Substituting (1) into (2):
\begin{align*}
  -V_{out}\left(\frac{G_2 (G_1 + s(C_1 + C_2))}{sC_1} + sC_2\right) &= V_{in} G_1 \\
  -V_{out}\frac{G_1 G_2 + s G_2 (C_1 + C_2) + s^2 C_1 C_2}{sC_1} &= V_{in} G_1 \\
  H(s) &= -\frac{s C_1 G_1}{s^2 C_1 C_2 + s G_2 (C_1 + C_2) + G_1 G_2}
\end{align*}

\subsubsection{Analysis}
\begin{enumerate}
  \item \(\omega_0^2 = \frac{G_1 G_2}{C_1 C_2}\), \(\frac{\omega_0}{Q} = \frac{G_2 (C_1 + C_2)}{C_1 C_2}\):
    \[
      Q = \frac{\sqrt{C_1 C_2}}{C_1 + C_2}\sqrt{\frac{G_1}{G_2}} = \frac{\sqrt{C_1 C_2}}{C_1 + C_2}\sqrt{\frac{R_2}{R_1}}
    \]
  \item Centre gain (\(s = j\omega_0\), the \(s^2\) and constant terms cancel):
    \(H(j\omega_0) = -\frac{C_1 G_1}{G_2 (C_1 + C_2)} = -\frac{R_2}{R_1}\cdot\frac{C_1}{C_1 + C_2}\).
  \item Equal capacitors: \(Q = \frac{1}{2}\sqrt{\frac{R_2}{R_1}}\),
    \(H(j\omega_0) = -\frac{R_2}{2R_1} = -2Q^2\); bandwidth \(\frac{f_0}{Q}\)
    (as derived for the RLC band filter).
  \item Zero at \(s = 0\) (and one at infinity): band pass.
\end{enumerate}

\subsubsection{Question}
\(f_0 = 1\,\)kHz, \(Q = 5\), \(C_1 = C_2 = 10\,\)nF.
\paragraph{Solution:}
\(\frac{R_2}{R_1} = 4Q^2 = 100\); \(\omega_0 = \frac{1}{C\sqrt{R_1 R_2}} = \frac{1}{10 C R_1}\):
\(R_1 = \frac{1}{10 \cdot 10\,n \cdot 2\pi \cdot 1000} = 1.59\,k\Omega\),
\(R_2 = 159\,k\Omega\), centre gain \(-50\).
