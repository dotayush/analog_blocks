\subsection{Second Order Systems}\label{sec:theory_cs_second_order}

A second order system is one that can be described by a
differential equation of order 2 in time domain which
transforms into polynomial with degree 2 in Frequency
domain. The general form of a second order transfer
function is given by:
\begin{align*}
  H(s) = \frac{V_{out}}{V_{in}} = \frac{\text{Zeros}}{\text{Poles}} = \frac{N(s)}{D(s)} = \frac{b_2 s^2 + b_1 s + b_0}{a_2 s^2 + a_1 s + a_0} = \underbrace{\frac{b_2 s^2}{a_2 s^2+a_1 s+a_0}}_{\text{Highpass}} + \underbrace{\frac{b_1 s}{a_2 s^2+a_1 s+a_0}}_{\text{Bandpass}} + \underbrace{\frac{b_0}{a_2 s^2+a_1 s+a_0}}_{\text{Lowpass}}
\end{align*}

\paragraph{Mathematical Overview of Poles:}
There are multiple ways to represent a second order
polynomial. I have listed four common forms below:
\begin{enumerate}
  \item The Mathematical form: This is the general representation
    of a second order polynomial.
    \begin{equation*}
      a x^2 + b x + c
    \end{equation*}
  \item The Normalized form: We divide the entire polynomial
    by the coefficient of \( s^2 \) (= \( x^2 \)) to make
    the leading coefficient 1.
    \begin{equation*}
      s^2 + \frac{a_1}{a_2}s + \frac{a_0}{a_2}
    \end{equation*}
  \item The Factored form: We express the polynomial
    in terms of its roots (poles).
    \begin{align*}
      (s + p_1)(s + p_2) \\
      s^2 + (p_1 + p_2)s + p_1 p_2
    \end{align*}
    where the roots (poles) of the polynomial are \( -p_1 \) and \( -p_2 \)
    (\(p_1, p_2 > 0\) for a stable system).
  \item The Natural form: We express the polynomial
    in terms of its natural frequency and damping ratio.
    \begin{align*}
      s^2 + 2 D_R \omega_n s + \omega_n^2 \\
      s^2 + \frac{\omega_n}{Q} s + \omega_n^2
    \end{align*}
    where \( \omega_n \) is the natural frequency and \( D_R \) is the
    damping ratio and \( Q\) is the quality factor.
\end{enumerate}

Next, we explore the relationships between coefficients, poles,
natural frequency, and damping ratio through three key comparisons.
\begin{enumerate}
  \item \textbf{Normalized Form vs. Factored Form}
    \noindent From the table, we have:
    \[ \frac{a_1}{a_2} = p_1 + p_2 \quad \text{and} \quad \frac{a_0}{a_2} = p_1 p_2 \]
    The discriminant
    \(\displaystyle B^2 - 4AC = \left( \frac{a_1}{a_2} \right)^2 - 4 \left( \frac{a_0}{a_2} \right) \)
    (positive, zero or negative) determines three pole configurations:
    \begin{enumerate}
      \item \textbf{Two distinct real poles (overdamped):} Assume \( p_2 = k p_1 \) with \( k \in \mathbb{R} \), \( k \neq 1 \).
        \begin{align*}
          \frac{a_1}{a_2} &= p_1(1 + k) \quad \Rightarrow \quad p_1 = \frac{a_1}{a_2(1 + k)} \tag{i} \\
          \frac{a_0}{a_2} &= k p_1^2 \quad \Rightarrow \quad k p_1^2 = \frac{a_0}{a_2} \tag{ii}
        \end{align*}
        Substituting (i) into (ii):
        \begin{align*}
        k \left( \frac{a_1}{a_2(1 + k)} \right)^2 &= \frac{a_0}{a_2} \\
        \frac{k\, a_1^2}{a_2^2 (1 + k)^2} &= \frac{a_0}{a_2} \\
        \frac{k}{(1+k)^2} &= \frac{a_0}{a_2}\cdot\frac{a_2^2}{a_1^2} = \frac{a_0 a_2}{a_1^2} \hspace{1cm} [\times \tfrac{a_2^2}{a_1^2}]
        \end{align*}
        For large \(k\), \((1 + k)^2 \approx k^2\), so \(\frac{k}{(1+k)^2} \approx \frac{k}{k^2} = \frac{1}{k}\):
        \[ \frac{1}{k} \approx \frac{a_0 a_2}{a_1^2} \hspace{1cm} \Rightarrow \hspace{1cm} k \approx \frac{a_1^2}{a_0 a_2} \]
      \item \textbf{One repeated real pole (critically damped):} Let \( p_1 = p_2 = p \).
        \begin{align*}
          \frac{a_1}{a_2} &= 2p \quad \Rightarrow \quad p = \frac{a_1}{2a_2} \tag{i} \\
          \frac{a_0}{a_2} &= p^2 \quad \Rightarrow \quad p^2 = \frac{a_0}{a_2} \tag{ii}
        \end{align*}
        Substituting (i) into (ii):
        \begin{align*}
          \left( \frac{a_1}{2a_2} \right)^2 &= \frac{a_0}{a_2} \\
          \frac{a_1^2}{4 a_2^2} &= \frac{a_0}{a_2} \\
          a_1^2 &= 4 a_0 a_2 \hspace{1cm} [\times 4 a_2^2]
        \end{align*}
      \item \textbf{Two complex conjugate poles (underdamped):} Let \( p_{1,2} = p_r \pm j p_i \), with \( p_r, p_i \in \mathbb{R} \).
        \begin{align*}
          \frac{a_1}{a_2} &= 2p_r \quad \Rightarrow \quad p_r = \frac{a_1}{2a_2} \tag{i} \\
          \frac{a_0}{a_2} &= p_r^2 + p_i^2 \quad \Rightarrow \quad p_r^2 + p_i^2 = \frac{a_0}{a_2} \tag{ii}
        \end{align*}
        Substituting (i) into (ii):
        \begin{align*}
          \left( \frac{a_1}{2a_2} \right)^2 + p_i^2 &= \frac{a_0}{a_2} \\
          p_i^2 &= \frac{a_0}{a_2} - \frac{a_1^2}{4a_2^2} \\
          p_i &= \sqrt{\frac{a_0}{a_2} - \frac{a_1^2}{4a_2^2}} \hspace{1cm} [\text{real for } a_1^2 < 4 a_0 a_2]
        \end{align*}
    \end{enumerate}
    These relations help determine pole locations from
    polynomial coefficients and help in pole placement
    design (choose k to set pole separation).
  \item \textbf{Normalized Form vs. Natural Form}
    \noindent From the table:
    \[ \frac{a_1}{a_2} = 2D_R\omega_n \quad \text{and} \quad \frac{a_0}{a_2} = \omega_n^2 \]
    From the second equation:
    \[ \omega_n = \sqrt{\frac{a_0}{a_2}} \tag{ii} \]
    Substituting (ii) into the first:
    \begin{align*}
    \frac{a_1}{a_2} &= 2D_R \sqrt{\frac{a_0}{a_2}} \\
    D_R &= \frac{a_1}{2 a_2}\sqrt{\frac{a_2}{a_0}} = \frac{a_1}{2\sqrt{a_2^2}}\cdot\frac{\sqrt{a_2}}{\sqrt{a_0}} = \frac{a_1}{2\sqrt{a_0 a_2}}
    \end{align*}
    Since quality factor \( Q = \frac{1}{2D_R} \) (matching \(2D_R\omega_n = \frac{\omega_n}{Q}\)):
    \[ Q = \frac{1}{2\cdot\frac{a_1}{2\sqrt{a_0 a_2}}} = \frac{\sqrt{a_0 a_2}}{a_1} \]
    Unlike the factored form, comparison with the natural form does
    not require separate algebraic cases; the three damping regimes
    emerge solely from the numerical value of \(D_R\). These
    relations help determine natural frequency and damping ratio from
    polynomial coefficients.
  \item \textbf{Factored Form vs. Natural Form}
    \noindent From the table:
    \[ p_1 + p_2 = 2D_R\omega_n \quad \text{and} \quad p_1 p_2 = \omega_n^2 \]
    From the second equation:
    \[ \omega_n = \sqrt{p_1 p_2} \tag{ii} \]
    Substituting (ii) into the first:
    \begin{align*}
      p_1 + p_2 &= 2D_R \sqrt{p_1 p_2} \\
      D_R &= \frac{p_1 + p_2}{2\sqrt{p_1 p_2}}
    \end{align*}
    Expressed in terms of quality factor:
    \[ Q = \frac{1}{2D_R} = \frac{2\sqrt{p_1 p_2}}{2(p_1 + p_2)} = \frac{\sqrt{p_1 p_2}}{p_1 + p_2}\]
    We now examine the three pole configurations:
    \begin{enumerate}
      \item \textbf{Two distinct real poles (overdamped):} With \( p_2 = k p_1 \),
        \[ Q = \frac{\sqrt{k p_1^2}}{p_1 + k p_1} = \frac{p_1\sqrt{k}}{p_1(1+k)} = \frac{\sqrt{k}}{1+k} \]
        For large k, \(1 + k \approx k\):
        \[ Q \approx \frac{\sqrt{k}}{k} = \frac{1}{\sqrt{k}} \]
      \item \textbf{One repeated real pole (critically damped):} With \( p_1 = p_2 = p \),
        \[ Q = \frac{\sqrt{p^2}}{p + p} = \frac{p}{2p} = \frac{1}{2} \hspace{1cm} [D_R = 1] \]
      \item \textbf{Two complex conjugate poles (underdamped):} With \( p_{1,2} = p_r \pm j p_i \),
        \begin{align*}
           Q &= \frac{\sqrt{(p_r + j p_i)(p_r - j p_i)}}{(p_r + j p_i) + (p_r - j p_i)}
           = \frac{\sqrt{p_r^2 - j^2 p_i^2}}{2p_r} = \frac{\sqrt{p_r^2 + p_i^2}}{2p_r} \hspace{1cm} [j^2 = -1] \\
           Q &> \frac{\sqrt{p_r^2}}{2p_r} = \frac{1}{2} \hspace{1cm} [p_i \neq 0 \Rightarrow p_r^2 + p_i^2 > p_r^2]
         \end{align*}
    \end{enumerate}
    These relations help determine natural frequency and
    damping ratio from pole locations.
\end{enumerate}

    \textbf{Comparison} & \textbf{Key Equations} & \textbf{Pole Configurations} & \textbf{Design Insights} \\ \hline

  \end{tabularx}
\end{center}

\paragraph{Example:}
\begin{enumerate}
  \item Given the second order polynomial \( 3s^2 + 12s + 27 \),
  determine the pole configuration, natural frequency, and
  damping ratio.
  \paragraph{Solution:}
  \begin{enumerate}
    \item Identify coefficients:
      \[ a_2 = 3, \quad a_1 = 12, \quad a_0 = 27 \]
    \item Calculate natural frequency:
      \begin{align*}
        \omega_n = \sqrt{\frac{a_0}{a_2}} = \sqrt{\frac{27}{3}} = \sqrt{9} = 3 \text{ rad/s}
      \end{align*}
    \item Calculate damping ratio:
      \begin{align*}
        D_R = \frac{a_1}{2\sqrt{a_0 a_2}} = \frac{12}{2\sqrt{27 \cdot 3}} = \frac{12}{2\sqrt{81}} = \frac{12}{18} = \frac{2}{3} \approx 0.6667
      \end{align*}
    \item Determine pole configuration:
      Since \( D_R \approx 0.6667 < 1 \), the system is underdamped with
      complex conjugate poles. Check with the normalised form
      \(s^2 + \frac{12}{3}s + \frac{27}{3} = s^2 + 4s + 9\):
      \begin{align*}
        p_r &= \frac{a_1}{2a_2} = \frac{12}{6} = 2 \hspace{1cm}
        p_i = \sqrt{\frac{27}{3} - \frac{144}{36}} = \sqrt{9 - 4} = \sqrt{5} = 2.236 \\
        s &= -p_r \mp j p_i = -2 \mp j\,2.236 \hspace{1cm}
        Q = \frac{1}{2D_R} = \frac{1}{2 \cdot \frac{2}{3}} = \frac{3}{4} = 0.75
      \end{align*}
  \end{enumerate}
  \item Given the second order polynomial \( 2s^2 + (8 + x)s + 16 \),
  determine the value of \( x \) for critical damping.
  \paragraph{Solution:}
  \begin{enumerate}
    \item Identify coefficients:
      \[ a_2 = 2, \quad a_1 = 8 + x, \quad a_0 = 16 \]
    \item Apply critical damping condition:
      \begin{align*}
        a_1^2 &= 4 a_0 a_2 \\
        (8 + x)^2 &= 4 \cdot 16 \cdot 2 \\
        (8 + x)^2 &= 128 \\
        8 + x &= \sqrt{128} = 8\sqrt{2} \hspace{1cm} [a_1 > 0 \text{ for stable poles; } -\sqrt{128} \text{ rejected}] \\
        x &= 8\sqrt{2} - 8 = 11.3137 - 8 \approx 3.3137
      \end{align*}
  \end{enumerate}
  Thus, \( x \approx 3.3137 \) for critical damping.
\end{enumerate}
