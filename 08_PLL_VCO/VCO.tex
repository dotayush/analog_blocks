\section{Voltage Controlled Oscillators}\label{sec:vco}

An oscillator is a loop with gain \(A(s)\) and feedback \(F(s)\) whose
closed loop \(\frac{A}{1 - AF}\) (positive feedback) has poles on the
\(j\omega\) axis. The Barkhausen conditions at \(\omega_0\):
\[
  |A(j\omega_0) F(j\omega_0)| = 1 \hspace{1cm} \angle A(j\omega_0) F(j\omega_0) = 0 + 2\pi k
\]
(in practice \(|AF| > 1\) at start-up, and amplitude limiting brings it to 1:
a Q6 effect). A VCO adds a control \(V_c\) with gain
\(K_{vco} = \frac{df_0}{dV_c}\) [Hz/V], \(K_v = 2\pi K_{vco}\) [rad/s/V].

\subsection{Ring Oscillator VCO}\label{sec:vco_ring}
\subsubsection{Overview}
\(N\) (odd) inverting stages in a loop. Each stage: DC gain \(-A_0\), pole
\(\omega_p = \frac{g_o}{C}\).

\subsubsection{Schematic (\(N = 3\))}
\begin{circuitfig}
  \draw (0,0) node[not port](I1){} (3,0) node[not port](I2){} (6,0) node[not port](I3){};
  \draw (I1.out) -- (I2.in 1); \draw (I2.out) -- (I3.in 1);
  \draw (I3.out) -- ++(0.6,0) coordinate(o) node[circ]{} node[above]{$V_{out}$} -- ++(0,-1.4) -| ($(I1.in 1) + (-0.6,0)$) -- (I1.in 1);
  \foreach \n in {I1, I2, I3} \draw (\n.out) ++(0.3,0) node[circ]{} to[capacitor, l_={$C$}] ++(0,-1) node[ground]{};
\end{circuitfig}

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: loop gain of \(N\) stages,
\begin{align*}
  A(s) &= \left(\frac{-A_0}{1 + \frac{s}{\omega_p}}\right)^N, \hspace{1cm} F = 1 \tag{1}
\end{align*}
\(N\) odd gives \(180^\circ\) from the inversions (\((-A_0)^N = -A_0^N\)); the poles must give the other
\(180^\circ\), \(\frac{180^\circ}{N}\) each. At \(s = j\omega\):
\begin{align*}
  A(j\omega) &= \frac{-A_0^N}{\left(1 + j\frac{\omega}{\omega_p}\right)^N} \hspace{0.5cm}
  |A(j\omega)| = \frac{A_0^N}{\left(1 + \frac{\omega^2}{\omega_p^2}\right)^{N/2}} \hspace{0.5cm}
  \angle A(j\omega) = \pi - N\tan^{-1}\frac{\omega}{\omega_p} \\
  \angle A(j\omega_0) = 0 &\hspace{0.3cm} \Rightarrow \hspace{0.3cm} N\tan^{-1}\frac{\omega_0}{\omega_p} = \pi \hspace{0.3cm} \Rightarrow \hspace{0.3cm} \tan^{-1}\frac{\omega_0}{\omega_p} = \frac{\pi}{N} \hspace{0.3cm} \Rightarrow \hspace{0.3cm} \omega_0 = \omega_p \tan\frac{\pi}{N} \\
  |A(j\omega_0)| \geq 1 &\hspace{0.3cm} \Rightarrow \hspace{0.3cm} \frac{A_0^N}{\left(1 + \tan^2\frac{\pi}{N}\right)^{N/2}} \geq 1 \hspace{0.3cm} \Rightarrow \hspace{0.3cm} A_0 \geq \left(1 + \tan^2\frac{\pi}{N}\right)^{1/2} = \sqrt{\frac{1}{\cos^2\frac{\pi}{N}}} = \frac{1}{\cos\frac{\pi}{N}}
\end{align*}
  \item \textbf{Large Signal, Time Domain, AC Input Analysis (Q6)}: each stage switches fully; a
    transition takes \(t_d\), and the signal must travel around the loop twice per period:
\[
  f_0 = \frac{1}{2 N t_d} \tag{2}
\]
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (Oscillation Frequency)}: from (1), \(\omega_0 = \omega_p \tan\frac{\pi}{N}\) (small signal);
    \(N = 3\): \(\omega_0 = \omega_p\tan 60^\circ = \sqrt{3}\,\omega_p\).
  \item \textbf{Start-up (DC Gain)}: \(A_0 \geq \frac{1}{\cos\frac{\pi}{N}}\); \(N = 3\): \(A_0 \geq \frac{1}{\cos 60^\circ} = \frac{1}{1/2} = 2\).
  \item \textbf{Tuning Gain (\(K_{vco}\)), current starved}: each stage charges its load \(C\) with a current
\(I(V_c)\) through half the supply (\(I\,t_d = C\frac{V_{dd}}{2}\)), \(t_d = \frac{C V_{dd}}{2 I}\):
\begin{align*}
  f_0 &= \frac{1}{2N t_d} = \frac{1}{2N\frac{C V_{dd}}{2I}} = \frac{I}{N C V_{dd}} \hspace{1cm}
  K_{vco} = \frac{df_0}{dV_c} = \frac{1}{N C V_{dd}} \frac{dI}{dV_c} = \frac{g_m}{N C V_{dd}}
\end{align*}
with \(g_m\) the transconductance of the starving device.
  \item \textbf{Supply Sensitivity}: with \(I\) fixed, \(f_0 = \frac{I}{NCV_{dd}}\) gives
    \(\frac{\partial f_0}{\partial V_{dd}} = -\frac{I}{NCV_{dd}^2} = -\frac{f_0}{V_{dd}}\): a \(1\%\) supply change moves \(f_0\) by \(1\%\) (a supply ripple is
    a frequency modulation), so ring VCOs need a regulated supply.
  \item \textbf{Advantages \& Limitations}: small area, wide tuning, \(N\) phases for free (multiphase
    clocks); but poor phase noise (every stage adds noise, no resonator), and \(f_0\) depends on
    \(V_{dd}\) (supply sensitivity).
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(N\) (odd) and \(t_d = \frac{1}{2 N f_0}\); \(I = \frac{C V_{dd}}{2 t_d}\) at the centre of the tuning range.
  \item \(g_m\) of the starving device from \(K_{vco} = \frac{g_m}{N C V_{dd}}\).
\end{enumerate}

\subsubsection{Code}
\begin{lstlisting}[language=Octave]
  clear; clc;
  pkg load symbolic;

  syms w wp A0 N positive
  % Barkhausen: phase pi - N atan(w/wp) = 0, |A| = 1
  w0 = solve(N*atan(w/wp) == sym(pi), w);
  printf('=== oscillation frequency ===\n'); pretty(simplify(w0))
  printf('=== minimum stage gain ===\n'); pretty(simplify(sqrt(1 + tan(sym(pi)/N)^2)))
  for n = [3 5 7]
    printf('N = %d: w0/wp = %.3f, A0 >= %.3f\n', n, tan(pi/n), 1/cos(pi/n));
  end
  % current starved tuning
  syms I C Vdd gm
  f0 = I/(N*C*Vdd);
  printf('=== df0/dVdd ===\n'); pretty(simplify(diff(f0, Vdd)))
\end{lstlisting}

\subsection{LC VCO}\label{sec:vco_lc}
\subsubsection{Overview}
A tank \(L \parallel C\) whose losses are lumped in \(R_p = Q_L \omega_0 L\)
(\(Q_L\) the inductor quality factor), kept oscillating by a cross-coupled
pair \(M_1\), \(M_2\) (tail \(I_T\)) that presents a negative resistance across the
tank. Tuning by a varactor (a reverse biased diode,
\(C_j(V)\), Chapter~\ref{ch:devices}).

\subsubsection{Schematic}
\begin{circuitfig}
  \draw (0,0) node[nmos, xscale=-1](M1){}; \node[left] at (-0.3,0) {$M_1$};
  \draw (4,0) node[nmos](M2){}; \node[right] at (4.3,0) {$M_2$};
  \draw (M1.D) -- (0,1.6) node[circ]{} node[left]{$V_{o1}$};
  \draw (M2.D) -- (4,1.6) node[circ]{} node[right]{$V_{o2}$};
  \draw (0,1.6) to[american inductor, l_={$L$}] (0,3.4);
  \draw (4,1.6) to[american inductor, l={$L$}] (4,3.4);
  \draw (0,1.6) -- (-1.3,1.6) to[capacitor, l={$C$}] (-1.3,3.4) -- (0,3.4);
  \draw (4,1.6) -- (5.3,1.6) to[capacitor, l_={$C$}] (5.3,3.4) -- (4,3.4);
  \draw (0,3.4) -- (4,3.4); \draw (2,3.4) node[vcc]{$V_{dd}$};
  \draw (M1.G) -- (4,1.6); \draw (M2.G) -- (0,1.6);
  \draw (M1.S) |- (2,-1.1) node[circ]{} -| (M2.S);
  \draw (2,-1.1) to[american current source, l={$I_T$}] (2,-2.6) node[ground]{};
\end{circuitfig}
Tank losses are lumped in \(R_p\) across each \(L\) (not drawn); \(C\)
includes the varactor.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: negative resistance: apply \(V_x\) between the two drains (\(+\frac{V_x}{2}\),
\(-\frac{V_x}{2}\)). Each gate sees the other drain, so
\(I_{d1} = g_m\left(-\frac{V_x}{2}\right)\) flows into drain 1:
\[
  I_x = -\frac{g_m V_x}{2} \hspace{0.5cm} \Rightarrow \hspace{0.5cm} R_{neg} = \frac{V_x}{I_x} = -\frac{2}{g_m}
\]
The differential tank (\(2L\), \(\frac{C}{2}\), \(2R_p\)) in parallel with
\(-\frac{2}{g_m}\):
\begin{align*}
  Y(s) &= \frac{1}{2R_p} - \frac{g_m}{2} + s\frac{C}{2} + \frac{1}{2sL} \\
  Y(s)\cdot 2sL &= \frac{sL}{R_p} - g_m sL + s^2 LC + 1 \\
  Y(s) = 0 \hspace{0.3cm} &\Rightarrow \hspace{0.3cm} s^2 LC + s L(1 - g_m R_p)/R_p + 1 = 0 \tag{3}
\end{align*}
  \item \textbf{Large Signal, Time Domain, AC Input Analysis (Q6)}: the pair switches \(I_T\) fully from
    side to side: the tank is driven by a square wave current of amplitude \(\pm\frac{I_T}{2}\) (drain currents
    \(I_T\), \(0\) and then \(0\), \(I_T\): differential current \(\frac{I_{D1} - I_{D2}}{2} = \pm\frac{I_T}{2}\)). A square wave of
    amplitude \(I_a\) has the Fourier series \(\frac{4I_a}{\pi}\left(\sin\omega_0 t + \frac{1}{3}\sin 3\omega_0 t + \cdots\right)\), so its
    fundamental is \(\frac{4}{\pi}\cdot\frac{I_T}{2}\); the tank (\(2R_p\) at \(\omega_0\)) passes only the fundamental:
\[
  V_{amp,diff} = \frac{4}{\pi}\cdot\frac{I_T}{2}\cdot 2R_p = \frac{4}{\pi} I_T R_p \tag{4}
\]
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\)), Start-up}: (3) has roots on (or right of) the \(j\omega\) axis when its
    damping term is \(\leq 0\):
\begin{align*}
  \frac{L(1 - g_m R_p)}{R_p} \leq 0 \hspace{0.3cm} &\Rightarrow \hspace{0.3cm} g_m R_p \geq 1 \hspace{0.5cm} \text{(start-up, usually designed for } g_m R_p \approx 2\text{--}3) \\
  g_m R_p = 1: \hspace{0.3cm} s^2 LC + 1 = 0 \hspace{0.3cm} &\Rightarrow \hspace{0.3cm} s = \pm\frac{j}{\sqrt{LC}} \hspace{0.3cm} \Rightarrow \hspace{0.3cm} \omega_0 = \frac{1}{\sqrt{LC}}
\end{align*}
  \item \textbf{Amplitude}: from (4), \(V_{amp,diff} = \frac{4}{\pi} I_T R_p\) (current limited, while below \(V_{dd}\)).
  \item \textbf{Tuning Gain (\(K_{vco}\))}: \(C = C_{fix} + C_{var}(V_c)\) [chain rule]:
\[
  K_{vco} = \frac{df_0}{dV_c} = \frac{d}{dC}\left(\frac{1}{2\pi\sqrt{L}}C^{-1/2}\right)\frac{dC}{dV_c} = -\frac{1}{2}\cdot\frac{1}{2\pi\sqrt{L}}C^{-3/2}\frac{dC_{var}}{dV_c} = -\frac{1}{2C}\cdot\frac{1}{2\pi\sqrt{LC}}\frac{dC_{var}}{dV_c} = -\frac{f_0}{2C}\frac{dC_{var}}{dV_c}
\]
  \item \textbf{Phase Noise} (Leeson), output power \(P_s\), noise factor \(F_n\), flicker corner \(f_c\),
    offset \(f_m\):
\[
  L(f_m) = 10\log_{10}\left[\frac{2 F_n k T}{P_s}\left(1 + \left(\frac{f_0}{2Q_L f_m}\right)^2\right)\left(1 + \frac{f_c}{f_m}\right)\right] \ \text{[dBc/Hz]}
\]
  \item \textbf{Tuning Range}: \(f_0 = \frac{1}{2\pi\sqrt{LC}}\) with \(C\) from \(C_{\min}\) to \(C_{\max}\):
    \(\frac{f_{\max}}{f_{\min}} = \sqrt{\frac{C_{\max}}{C_{\min}}}\); a \(2:1\) capacitance range gives only \(\sqrt{2} = 1.41\) (\(\pm 17\%\) about the centre),
    and fixed parasitics in \(C_{fix}\) reduce it further.
  \item \textbf{Advantages \& Limitations}: best phase noise (resonator \(Q_L\)), \(f_0 = \frac{1}{2\pi\sqrt{LC}}\) set by
    passives (supply insensitive), high frequency; but inductors cost area, tuning range is narrow
    (\(\propto \frac{C_{\max} - C_{\min}}{2C}\)), and \(K_{vco}\) varies across the range.
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(L\) (and \(Q_L\)) from the technology; \(C = \frac{1}{(2\pi f_0)^2 L}\), \(R_p = Q_L\omega_0 L\).
  \item \(g_m = (2\text{--}3)\frac{1}{R_p}\) (start-up); \(I_T = \frac{\pi V_{amp,diff}}{4 R_p}\).
  \item Varactor \(\frac{dC_{var}}{dV_c} = \frac{2C K_{vco}}{f_0}\).
\end{enumerate}

\subsubsection{Question}
\(L = 2\,\)nH, \(Q_L = 10\), \(f_0 = 5\,\)GHz, \(I_T = 2\,\)mA. Find \(C\), \(R_p\), the
minimum \(g_m\) and the amplitude.
\paragraph{Solution:}
\begin{align*}
  \omega_0 &= 2\pi f_0 = 2\pi \cdot 5 \times 10^9 = 3.14 \times 10^{10}\,\text{rad/s} \\
  C &= \frac{1}{(2\pi f_0)^2 L} = \frac{1}{(3.14 \times 10^{10})^2 \cdot 2\,n} = \frac{1}{9.87 \times 10^{20} \cdot 2 \times 10^{-9}} = \frac{1}{1.97 \times 10^{12}} = 507\,fF \\
  R_p &= Q_L\omega_0 L = 10 \cdot 3.14 \times 10^{10} \cdot 2\,n = 628\,\Omega \\
  g_m &\geq \frac{1}{R_p} = \frac{1}{628} = 1.6\,mS \\
  V_{amp,diff} &= \frac{4}{\pi} I_T R_p = \frac{4}{\pi}\cdot 2\,mA \cdot 628 = 1.273 \cdot 1.257 = 1.6\,V
\end{align*}

\subsubsection{Code}
\begin{lstlisting}[language=Octave]
  clear; clc;
  pkg load symbolic;

  syms s
  syms L C Rp gm positive
  % differential tank (2L, C/2, 2Rp) in parallel with -2/gm
  Y = 1/(2*Rp) - gm/2 + s*C/2 + 1/(2*s*L);
  n = expand(Y*2*s*L);
  printf('=== Y(s) 2sL ===\n'); pretty(n)
  printf('=== roots at gm Rp = 1 ===\n'); pretty(solve(subs(n, gm, 1/Rp) == 0, s))
  syms Cv Vc dCdV
  f0 = 1/(2*sym(pi)*sqrt(L*C));
  printf('=== Kvco = df0/dC * dC/dVc ===\n'); pretty(simplify(diff(f0, C)*dCdV))

  % Question: L = 2 nH, QL = 10, f0 = 5 GHz, IT = 2 mA
  w = 2*pi*5e9; Cn = 1/(w^2*2e-9); Rpn = 10*w*2e-9;
  printf('C = %.0f fF, Rp = %.0f Ohm, gm >= %.2f mS, Vamp = %.2f V\n', Cn*1e15, Rpn, 1/Rpn*1e3, 4/pi*2e-3*Rpn);
  printf('tuning range for Cmax/Cmin = 2: %.3f\n', sqrt(2));
\end{lstlisting}

\subsection{Relaxation VCO}\label{sec:vco_relax}
\subsubsection{Overview}
A current \(I = g_c V_c\) charges and then discharges \(C\) between the two thresholds \(V_H\), \(V_L\)
of a Schmitt trigger (\(dV = V_H - V_L\)).

\subsubsection{Schematic}
\begin{circuitfig}
  \draw (0,4.6) node[vcc]{$V_{dd}$} to[american current source, l={$I = g_c V_c$}] (0,3.2) to[nos, l={$S_1$}] (0,1.5) node[circ]{} coordinate(n);
  \draw (n) to[nos, l={$S_2$}] (0,-0.2) to[american current source, l={$I$}] (0,-1.6) node[ground]{};
  \draw (n) -- (2,1.5) node[circ]{} to[capacitor, l={$C$}] (2,-0.5) node[ground]{};
  \draw (4.5,1.5) node[op amp, scale=0.8](S){};
  \draw (2,1.5) -- (3,1.5) |- (S.+); \draw (S.-) -- ++(-0.3,0) node[left]{\footnotesize ref};
  \draw (S.out) -- ++(0.6,0) node[circ]{} node[right]{$V_{out}$};
  \draw[dashed] ($(S.out) + (0.3,0)$) -- ++(0,1.9) -| (0.5,2.35);
  \draw[dashed] ($(S.out) + (0.3,0)$) -- ++(0,-2.1) -| (0.5,0.65);
  \node at (4.5,2.6) {\footnotesize Schmitt trigger ($V_H$, $V_L$)};
\end{circuitfig}

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, AC Input Analysis (Q6)}: \(C\frac{dV_C}{dt} = I\) on each ramp;
    integrating from \(V_L\) to \(V_H\):
    \begin{align*}
      \int_{V_L}^{V_H} C\, dV_C &= \int_0^{t} I\, dt' \hspace{1cm} [\text{Integrating both sides}] \\
      C\, dV &= I t \hspace{1cm} \Rightarrow \hspace{1cm} t = \frac{C\, dV}{I} \\
      T &= 2t = \frac{2 C\, dV}{I} \hspace{1cm} f_0 = \frac{1}{T} = \frac{I}{2 C\, dV} = \frac{g_c V_c}{2 C\, dV} \tag{5}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Tuning Gain (\(K_{vco}\))}: from (5), \(K_{vco} = \frac{df_0}{dV_c} = \frac{g_c}{2 C\, dV}\), constant (linear in \(V_c\)).
  \item \textbf{Comparator Delay}: each ramp overshoots by the delay \(t_d\) of the Schmitt trigger and switches:
    \(T = 2\left(\frac{C\,dV}{I} + t_d\right)\), \(f_0 = \frac{1}{2\left(\frac{C\,dV}{I} + t_d\right)}\); the tuning law bends below \(\frac{I}{2C\,dV}\) once
    \(t_d\) is comparable with \(\frac{C\,dV}{I}\).
  \item \textbf{Advantages \& Limitations}: very linear and wide tuning (\(f_0 \propto V_c\)), no inductor;
    but threshold noise of the comparator makes its phase noise the worst of the three, and
    comparator delay limits \(f_0\).
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(g_c = 2 C\, dV\, K_{vco}\); \(C\) and \(dV\) from \(f_0 = \frac{g_c V_c}{2 C\, dV}\) at mid range.
\end{enumerate}

\subsubsection{Code}
\begin{lstlisting}[language=Octave]
  clear; clc;
  pkg load symbolic;

  syms C dV I t td gc Vc positive
  tr = solve(C*dV == I*t, t);                          % one ramp
  f0 = 1/(2*tr);
  printf('=== f0, Kvco ===\n'); pretty(subs(f0, I, gc*Vc)); pretty(diff(subs(f0, I, gc*Vc), Vc))
  printf('=== f0 with comparator delay ===\n'); pretty(simplify(1/(2*(tr + td))))

  % simulate the ramp: C = 1 pF, dV = 1 V, I = 10 uA
  c = 1e-12; dv = 1; i = 10e-6; dt = 1e-10; v = 0; dir = 1; tsw = [];
  for k = 1:20000
    v = v + dir*i/c*dt;
    if v >= dv, dir = -1; tsw(end + 1) = k*dt; elseif v <= 0, dir = 1; tsw(end + 1) = k*dt; end
  end
  printf('simulated f0 = %.2f MHz, formula = %.2f MHz\n', 1/(2*mean(diff(tsw)))/1e6, i/(2*c*dv)/1e6);
\end{lstlisting}
