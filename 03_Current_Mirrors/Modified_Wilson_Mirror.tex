\subsection{Modified Wilson Mirror}

\subsubsection{Overview}
Constructed from four transistors, It improves
upon the Wilson Current Mirror by providing a
higher input impedance which in turn improves
the accuracy of current mirroring.

\subsubsection{Transfer Function}
Here's a table of the voltages at each terminal of the
transistors:\\
\begin{tabularx}{\linewidth}{|X|X|X|X|X|X|} \hline
  Transistor & Gate (G) & Source (S) & Drain (D) & $V_{gs}$ & $V_{ds}$ \\ \hline
  $M_0$ & $V_{x}$ & $V_{ee}$ & $V_{y}$ & $V_{x} - V_{ee}$ & $V_{y} - V_{ee}$ \\ \hline
  $M_1$ & $V_{x}$ & $V_{ee}$ & $V_{x}$ & $V_{x} - V_{ee}$ & $V_{x} - V_{ee}$ \\ \hline
  $M_2$ & $V_{z}$ & $V_{x}$ & $V_{out}$ & $V_{z} - V_{x}$ & $V_{out} - V_{x}$ \\ \hline
  $M_3$ & $V_{z}$ & $V_{y}$ & $V_{z}$ & $V_{z} - V_{y}$ & $V_{z} - V_{y}$ \\ \hline
\end{tabularx}\\

\begin{enumerate}
  \item \textbf{Large Signal Analysis}:\\
    Applying KCL at $V_{out}$:
    \begin{align*}
      (V_{out} - V_{cc})G_5 + I_{M_2} &= 0 \\
      V_{out}G_5 - V_{cc}G_5 + I_{M_2} &= 0 \\
      V_{out} &= \frac{V_{cc}G_5 - I_{M_2}}{G_5} \tag{1}
    \end{align*}
    Applying KCL at $V_{x}$:
    \begin{align*}
      I_{M_1} + I_{G_1} - I_{M_2} &= 0 \\
      I_{M_1} + I_{G_1} &= I_{M_2} \\
      I_{M_1} &\approx I_{M_2} \quad \text{[assuming } I_{G_1} \approx 0 \text{ by MOS physics]} \tag{2}
    \end{align*}
    Applying KCL at $V_{y}$:
    \begin{align*}
      I_{M_0} - I_{M_3} &= 0 \\
      I_{M_0} &= I_{M_3} \tag{3}
    \end{align*}
    Applying KCL at $V_{z}$:
    \begin{align*}
      (V_{z} - V_{cc})G_4 + I_{M_3} + I_{G_2} &= 0 \\
      V_{z}G_4 - V_{cc}G_4 + (I_{M_3} + I_{G_2}) &= 0 \\
      V_{z}G_4 &= V_{cc}G_4 - (I_{M_3} + I_{G_2}) \\
      V_{z} &= V_{cc} - \frac{(I_{M_3} + I_{G_2})}{G_4} \tag{4}
    \end{align*}
    Using transistor equations for $I_{M_0}$, $I_{M_1}$, $I_{M_2}$ and $I_{M_3}$
    and applying the voltages:
    \begin{align*}
      I_{M_0} &= k_{n_0}(V_{gs_0} - V_{th})^2 \cdot (1 + \lambda V_{ds_0}) \\
              &= k_{n_0}((V_{x} - V_{ee}) - V_{th})^2 \cdot (1 + \lambda (V_{y} - V_{ee})) \\
              &= k_{n_0}(V_{x} - V_{ee} - V_{th})^2 \quad \text{[assuming } \lambda = 0 \text{]} \tag{5} \\ \\
      I_{M_1} &= k_{n_1}(V_{gs_1} - V_{th})^2 \cdot (1 + \lambda V_{ds_1}) \\
              &= k_{n_1}((V_{x} - V_{ee}) - V_{th})^2 \cdot (1 + \lambda (V_{x} - V_{ee})) \\
              &= k_{n_1}(V_{x} - V_{ee} - V_{th})^2 \quad \text{[assuming } \lambda = 0 \text{]} \tag{6} \\ \\
      I_{M_2} &= k_{n_2}(V_{gs_2} - V_{th})^2 \cdot (1 + \lambda V_{ds_2}) \\
              &= k_{n_2}((V_{z} - V_{x}) - V_{th})^2 \cdot (1 + \lambda (V_{out} - V_{x})) \\
              &= k_{n_2}\left(\left(V_{cc} - \frac{(I_{M_3} + I_{G_2})}{G_4}\right) - V_{x} - V_{th}\right)^2 \cdot \left(1 + \lambda\left(\frac{V_{cc}G_5 - I_{M_2}}{G_5}\right) - V_{x}\right) \\
              &= k_{n_2}\left(\left(V_{cc} - \frac{(I_{M_3} + I_{G_2})}{G_4}\right) - V_{x} - V_{th}\right)^2 \quad \text{[assuming } \lambda = 0 \text{]} \tag{7} \\ \\
      I_{M_3} &= k_{n_3}(V_{gs_3} - V_{th})^2 \cdot (1 + \lambda V_{ds_3}) \\
              &= k_{n_3}((V_{z} - V_{y}) - V_{th})^2 \cdot (1 + \lambda (V_{z} - V_{y})) \\
              &= k_{n_3}\left(\left(V_{cc} - \frac{(I_{M_3} + I_{G_2})}{G_4}\right) - V_{y} - V_{th}\right)^2 \cdot \left(1 + \lambda\left(\left(V_{cc} - \frac{(I_{M_3} + I_{G_2})}{G_4}\right) - V_{y}\right)\right) \\
              &= k_{n_3}\left(\left(V_{cc} - \frac{(I_{M_3} + I_{G_2})}{G_4}\right) - V_{y} - V_{th}\right)^2 \quad \text{[assuming } \lambda = 0 \text{]} \tag{8}
    \end{align*}
    It can be seen that,
    \begin{align*}
      I_{in} &= I_{M_3} = I_{M_0} \quad \text{[from (3)]} \\
      I_{out} &= I_{M_2} = I_{M_1} \quad \text{[from (2)]} \\
      I_{M_0} &= I_{M_1} \quad \text{[from (5) and (6) and assuming } k_{n_0} = k_{n_1} \text{]} \\
      \therefore  I_{in} &= I_{out}
    \end{align*}
    Thus,
    \begin{align*}
      \boxed{\frac{I_{out}}{I_{in}} \approx \frac{I_{M_2}}{I_{M_3}} \approx \frac{I_{M_1}}{I_{M_0}} \approx \frac{k_{n_1}}{k_{n_0}} = \frac{\frac{\mu_n C_{ox} W_1}{L_1}}{\frac{\mu_n C_{ox} W_0}{L_0}} = \frac{\frac{W_1}{L_1}}{\frac{W_0}{L_0}} = \frac{W_1 L_0}{L_1 W_0} = \frac{W_2 L_3}{L_2 W_3}}
    \end{align*}
  \item \textbf{Small Signal Analysis}:\\
    Applying KCL at $V_{out}$:
    \begin{align*}
      V_{out}G_5 + (V_{out} - V_{x})g_{o_2} + g_{m_2}(V_{z} - V_{x}) &= 0 \\
      V_{out}G_5 + V_{out}g_{o_2} - V_{x}g_{o_2} + g_{m_2}V_{z} - g_{m_2}V_{x} &= 0 \\
      V_{out}(G_5 + g_{o_2}) - V_{x}(g_{o_2} + g_{m_2}) + g_{m_2}V_{z} &= 0 \tag{9}
    \end{align*}
    Applying KCL at $V_{x}$:
    \begin{align*}
      (V_{x} - V_{out})g_{o_2} + V_{x}g_{o_1} + g_{m_1}V_{x} - g_{m_2}(V_{z} - V_{x}) &= 0 \\
      V_{x}g_{o_2} - V_{out}g_{o_2} + V_{x}g_{o_1} + g_{m_1}V_{x} - g_{m_2}V_{z} + g_{m_2}V_{x} &= 0 \\
      V_{x}(g_{o_2} + g_{o_1} + g_{m_1} + g_{m_2}) - V_{out}g_{o_2} - g_{m_2}V_{z} &= 0 \tag{10}
    \end{align*}
    Applying KCL at $V_{y}$:
    \begin{align*}
      (V_{y} - V_{z})g_{o_3} - g_{m_3}(V_{z} - V_{y}) + V_{y}g_{o_0} + g_{m_0}V_{x} &= 0 \\
      V_{y}g_{o_3} - V_{z}g_{o_3} - g_{m_3}V_{z} + g_{m_3}V_{y} + V_{y}g_{o_0} + g_{m_0}V_{x} &= 0 \\
      V_{y}(g_{o_3} + g_{m_3} + g_{o_0}) - V_{z}(g_{o_3} + g_{m_3}) + g_{m_0}V_{x} &= 0 \tag{11}
    \end{align*}
    Applying KCL at $V_{z}$:
    \begin{align*}
      V_{z}G_4 + (V_{z} - V_{y})g_{o_3} + g_{m_3}(V_{z} - V_{y}) &= 0 \\
      V_{z}G_4 + V_{z}g_{o_3} - V_{y}g_{o_3} + g_{m_3}V_{z} - g_{m_3}V_{y} &= 0 \\
      V_{z}(G_4 + g_{o_3} + g_{m_3}) - V_{y}(g_{o_3} + g_{m_3}) &= 0 \tag{12}
    \end{align*}
    Adding voltage source $V_s$ at $V_z$ that supplies
    a current $I_s$ into $V_z$, we get:
    \begin{align*}
      V_{z}(G_4 + g_{o_3} + g_{m_3}) - V_{y}(g_{o_3} + g_{m_3}) - I_s &= 0 \tag{13}
    \end{align*}
    Fifth equation arises from Voltage source:
    \begin{align*}
      V_{z} &= V_{s} \tag{14}
    \end{align*}
    Equations (9), (10), (1), (13) and (14) can now be
    solved analytically, but it's cumbersome, so I'll give the
    solution directly from Octave:
    \begin{align*}
      \frac{V_{out}}{V_{s}} = \frac{-g_{m_2}(g_{m_1} + g_{o_1})}{G_5(g_{m_1} + g_{m_2} + g_{o_1} + g_{o_2}) + g_{o_2}(g_{m_1} + g_{o_1})} \tag{15}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Output Load (Max)}: Maximum possible
    load that can be connected at the output before
    the transistor $M_2$ goes out of saturation can
    be calculated as follows:
    \begin{align*}
      R_{out, max} &= \frac{V_{L, max}}{I_{M_2}} \quad \text{[where, } V_{ee} \leq V_{L, max} \leq V_{cc} \text{]}
    \end{align*}
  \item \textbf{Impedances}:
    \paragraph{At Input:}Applying a test current $I_{test}$
      at $V_{z}$, from equation (12),
      \begin{align*}
        V_{z}(G_4 + g_{o_3} + g_{m_3}) - V_{y}(g_{o_3} + g_{m_3}) - I_{test} &= 0 \tag{16}
      \end{align*}
      From equation (11),
      \begin{align*}
        V_{y}(g_{o_3} + g_{m_3} + g_{o_0}) - V_{z}(g_{o_3} + g_{m_3}) + g_{m_0}V_{x} &= 0 \\
        V_{y} &= \frac{V_{z}(g_{o_3} + g_{m_3}) - g_{m_0}V_{x}}{(g_{o_3} + g_{m_3} + g_{o_0})} \tag{17}
      \end{align*}
      From equation (10),
      \begin{align*}
        V_{x}(g_{o_2} + g_{o_1} + g_{m_1} + g_{m_2}) - V_{out}g_{o_2} - g_{m_2}V_{z} &= 0 \\
        V_{x} &= \frac{V_{out}g_{o_2} + g_{m_2}V_{z}}{(g_{o_2} + g_{o_1} + g_{m_1} + g_{m_2})} \tag{18}
      \end{align*}
      From equation (15),
      \begin{align*}
        V_{out} &= \frac{-g_{m_2}(g_{m_1} + g_{o_1})}{G_5(g_{m_1} + g_{m_2} + g_{o_1} + g_{o_2}) + g_{o_2}(g_{m_1} + g_{o_1})} V_{z} \tag{19}
      \end{align*}
      Substitute (19) into (18), result into (17), result into (16)
      to eliminate $V_{y}$ and finally rearranging the result
      for $\frac{V_{z}}{I_{test}}$ would give the input impedance.
      However, the result is incredibly large, so run Input Impedance
      Code from Code section to get the result.
    \paragraph{At Output:} Applying a test current $I_{test}$
      at $V_{out}$, from equation (9),
      \begin{align*}
        V_{out}(G_5 + g_{o_2}) - V_{x}(g_{o_2} + g_{m_2}) + g_{m_2}V_{z} - I_{test} &= 0 \tag{20}
      \end{align*}
      From equation (10),
      \begin{align*}
        V_{x}(g_{o_2} + g_{o_1} + g_{m_1} + g_{m_2}) - V_{out}g_{o_2} - g_{m_2}V_{z} &= 0 \\
        V_{x} &= \frac{V_{out}g_{o_2} + g_{m_2}V_{z}}{(g_{o_2} + g_{o_1} + g_{m_1} + g_{m_2})} \tag{21}
      \end{align*}
      From equation (15),
      \begin{align*}
        V_{z} &= \frac{-G_5(g_{m_1} + g_{m_2} + g_{o_1} + g_{o_2}) + g_{o_2}(g_{m_1} + g_{o_1})}{g_{m_2}(g_{m_1} + g_{o_1})} V_{out} \tag{22}
      \end{align*}
      Substitute (22) into (21), result into (20) to eliminate
      $V_{x}$, Substituting (22) into (20) to eliminate $V_{z}$
      and finally rearranging the result for $\frac{V_{out}}{I_{test}}$
      would give the output impedance. However, the result
      is incredibly large, so run Output Impedance Code from

\end{enumerate}

\subsubsection{Design}
Large Signal Analysis can be used to set the bias
currents and voltages in this circuit.
\begin{enumerate}
  \item Choose appropriate transistor dimensions
    ($W$ and $L$) for transistors $M_0$ and $M_1$
    to achieve the desired current gain.
    \begin{align*}
      \frac{I_{out}}{I_{in}} &= \frac{W_1 L_0}{L_1 W_0} = \frac{W_2 L_3}{L_2 W_3} \\
    \end{align*}
  \item Choose a suitable overdrive voltage value
    ($V_{ov}$) for computing the bias voltages at
    nodes $V_{x}$, $V_{y}$, and $V_{z}$.
    \begin{align*}
      V_{ov_0} &= V_{gs_0} - V_{th} \\
      V_{ov_0} &= V_{x} - V_{ee} - V_{th} \\
      V_{x} &= V_{ov_0} + V_{ee} + V_{th} \\
      \\
      V_{ov_2} &= V_{gs_2} - V_{th} \\
      V_{ov_2} &= V_{z} - V_{x} - V_{th} \\
      V_{z} &= V_{ov_2} + V_{x} + V_{th} \\
      \\
      V_{ov_3} &= V_{gs_3} - V_{th} \\
      V_{ov_3} &= V_{z} - V_{y} - V_{th} \\
      V_{y} &= V_{z} - V_{ov_3} - V_{th}
    \end{align*}
  \item Using the chosen $V_{ov}$, compute $W_0$
    to set the required bias current $I_{in}$.
    \begin{align*}
      I_{in} &\approx I_{M_0} = \frac{1}{2} \mu_n C_{ox} \frac{W_0}{L_0} V_{ov_0}^2 \\
             &\approx I_{M_3} = \frac{1}{2} \mu_n C_{ox} \frac{W_3}{L_3} V_{ov_3}^2 \\
      \therefore W_0 &= \frac{2 I_{in} L_0}{\mu_n C_{ox} V_{ov_0}^2} \\
                 W_3 &= \frac{2 I_{in} L_3}{\mu_n C_{ox} V_{ov_3}^2}
    \end{align*}
    $W_1$ can then be computed using the current
    gain relation from step (a). Similarly, $W_2$
    can be computed using the current gain relation
    from step (a).
  \item Compute the required value of $G_4$ (or
    $R_4$) to ensure that the voltage at node
    $V_{z}$ is as calculated in step (b).
    \begin{align*}
      V_{z} &= V_{cc} - \frac{(I_{M_3} + I_{G_2})}{G_4} \\
      V_{z} &\approx V_{cc} - \frac{I_{M_3}}{G_4} \quad \text{[assuming } I_{G_2} \approx 0 \text{ by MOS physics]} \\
      V_{z} - V_{cc} &\approx - \frac{I_{M_3}}{G_4} \\
      \frac{(I_{M_3})}{(V_{cc} - V_{z})} &\approx G_4 \\
      \frac{(I_{in})}{(V_{cc} - V_{z})} &\approx G_4 \quad \text{[} I_{in} = I_{M_3} = I_{M_0} \text{]} \\
      \therefore R_4 &\approx \frac{V_{cc} - V_{z}}{I_{in}}
    \end{align*}
\end{enumerate}

\subsubsection{Question}
Design a Modified Wilson Current Mirror in
Sink configuration to achieve a current gain of
2, with an input bias current of $I_{in} =
10\mu A$. Assume the following parameters for
the NMOS transistors used:
\begin{align*}
  \mu_n C_{ox} = 100 \mu A/V^2 \spc V_{th} = 0.7 V \spc \lambda = 0.02 V^{-1} \\
  V_{cc} = 5 V \spc V_{ee} = 0 V \spc L_0 = L_1 = L_2 = L_3 = 1 \mu m
\end{align*}
Find the output impedance and the maximum
possible load that can be connected at the output
assuming $V_{L,max} = V_{cc}$
\paragraph{Solution:}
\begin{enumerate}
  \item \textbf{Choose appropriate transistor dimensions:}
    \begin{align*}
      \frac{I_{out}}{I_{in}} &= \frac{W_1 L_0}{L_1 W_0} \\
      2 &= \frac{W_1 \cdot 1 \mu m}{1 \mu m \cdot W_0} \\
      W_1 &= 2 \cdot W_0
    \end{align*}
  \item \textbf{Choose a suitable overdrive voltage and determine
    the resulting bias voltages:}
    \begin{align*}
      V_{ov_0} &= 0.2 V \\
      V_{x} &= V_{ov_0} + V_{ee} + V_{th} = 0.2 + 0 + 0.7 = 0.9 V \\
      \\
      V_{ov_2} &= 0.2 V \\
      V_{z} &= V_{ov_2} + V_{x} + V_{th} = 0.2 + 0.9 + 0.7 = 1.8 V \\
      \\
      V_{ov_3} &= 0.2 V \\
      V_{y} &= V_{z} - V_{ov_3} - V_{th} = 1.8 - 0.2 - 0.7 = 0.9 V
    \end{align*}
  \item \textbf{Compute $W_0$ to set the required bias current:} For $I_{in} = 10 \mu A$
    \begin{align*}
      W_0 &= \frac{2 I_{in} L_0}{\mu_n C_{ox} V_{ov_0}^2} \\
          &= \frac{2 \cdot 10 \mu A \cdot 1 \mu m}{100 \mu A \cdot (0.2)^2} \\
          &= 5 \mu m
    \end{align*}
    Using (i) and (ii):
    \begin{align*}
      W_1 &= 2 \cdot W_0 = 2 \cdot 5 \mu m = 10 \mu m
    \end{align*}
  \item \textbf{Compute the required value of $G_4$ (or $R_4$):}
    \begin{align*}
      R_4 &\approx \frac{V_{cc} - V_{z}}{I_{in}} \\
          &= \frac{5 V - 1.8 V}{10 \mu A} \\
          &= 320 k\Omega
    \end{align*}
  \item \textbf{Output Load Limit:}
    \begin{align*}
      R_{out, max} &= \frac{V_{L, max}}{I_{M_2}} = \frac{5 V}{10 \mu A} = 500 k\Omega
    \end{align*}
  \item \textbf{Impedances:}
    \paragraph{At Input:}
    \begin{align*}
      Z_{in} &\approx \frac{2}{\lambda_{M_0} \lambda_{M_3} I_{M_0} (V_{gs_3} - V_{th})} \\
             &= \frac{2}{0.02 \times 0.02 \times 10 \mu A \times (1.8 - 0.9)} \\
             &= 5.56 G\Omega
    \end{align*}
    \paragraph{At Output:}
    \begin{align*}
      Z_{out} &\approx \frac{2}{\lambda_{M_1} \lambda_{M_2} I_{M_1} (V_{gs_2} - V_{th})} \\
              &= \frac{2}{0.02 \times 0.02 \times 20 \mu A \times (1.8 - 0.9)} \\
              &= 2.78 G\Omega
    \end{align*}
\end{enumerate}
