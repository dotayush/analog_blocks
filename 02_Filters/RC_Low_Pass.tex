\section{RC Low Pass Filter}

\paragraph{Overview:}
Constructed from a resistor and capacitor in series, a RC
Low Pass Filter allows low frequency signals to pass while
attenuating high frequency signals. It is a single pole
filter but can be used in multiple stages to create higher
order filters.

\subsection{Transfer Function}
\raggedright
Applying KCL at $V_{out}$, we get:
\begin{equation}
  \begin{aligned}
    (V_{out} - V_{cc}){G_1} + (V_{out} - V_{ee})sC_1 &= 0 \\
    V_{out}G_1 - V_{cc}G_1 + V_{out}sC_1 - V_{ee}sC_1 &= 0
  \end{aligned}
\end{equation}

Now! A few important equations can be derived:
\begin{enumerate}
  \item Transfer Function: We rearrange KCL for Transfer Function
    in terms of \(\frac{V_{out}}{V_{cc}}\):
    \begin{align*}
      V_{out}(G_1 + sC_1) - V_{cc}G_1 - V_{ee}sC_1 &= 0 \\
      V_{out}(G_1 + sC_1) &= V_{cc}G_1 + V_{ee}sC_1 \\
      V_{out} &= \frac{V_{cc}G_1 + V_{ee}sC_1}{G_1 + sC_1} \\
      \boxed{\frac{V_{out}}{V_{cc}} = \frac{V_{cc}G_1 + V_{ee}sC_1}{V_{cc}(G_1 + sC_1)}} \\
    \end{align*}
  \item Poles: We can find the poles by setting the denominator
    of the transfer function to zero:
    \begin{align*}
      V_{cc}(G_1 + sC_1) &= 0 \\
      G_1 + sC_1 &= 0 \\
      s &= -\frac{G_1}{C_1} \\
      \boxed{\text{Complex Pole at } s = -\frac{1}{R_1 C_1}}
    \end{align*}
    We can calculate the frequency corresponding to this pole:
    \begin{align*}
      s &= -2 \pi f_p \\
      -2 \pi f_p &= -\frac{1}{R_1 C_1} \\
      f_p &= \frac{1}{2 \pi R_1 C_1} \\
      \boxed{f_p = \frac{1}{2 \pi R_1 C_1}}
    \end{align*}
  \item Input Impedance \(Z_{in}\):
    \begin{align*}
      Z_{in} &= \frac{1}{G_1} + \frac{1}{sC_1} [\text{where } G_1 = \frac{1}{R_1}] \\
      \boxed{Z_{in} = R_1 + \frac{1}{sC_1}}
    \end{align*}
  \item Output Impedance \(Z_{out}\):
    \begin{align*}
      Z_{out} &= \frac{1}{G_1 + sC_1} \\
      Z_{out} &= \frac{1}{\frac{1}{R_1} + sC_1} \\
      Z_{out} &= \frac{1}{\frac{1 + sR_1C_1}{R_1}} \\
      Z_{out} &= \frac{R_1}{1 + sR_1C_1} \frac{1 - sR_1C_1}{1 - sR_1C_1} \\
      Z_{out} &= \frac{R_1(1 - sR_1C_1)}{1^2 + (sR_1C_1)^2} \\
      \boxed{Z_{out} = \frac{R_1 - sR_1^2C_1}{1 - (sR_1C_1)^2}}
    \end{align*}
\end{enumerate}

\subsection{Example Design}

\paragraph{Question:}
Design a RC Low Pass Filter with a cutoff frequency of
\(f_p = 45kHz\) using a capacitor value of \(C_1 = 1\mu F\).
Also compute the input and output impedances at
\(f = 10kHz\).

\paragraph{Solution:}
\begin{enumerate}
  \item Calculating \(R_1\):
    \begin{align*}
      f_p &= \frac{1}{2 \pi R_1 C_1} \\
      R_1 &= \frac{1}{2 \pi f_p C_1} \\
      &= \frac{1}{2 \pi (45kHz)(1\mu F)} \\
      R_1 &\approx 3.54 \Omega \\
    \end{align*}
  \item Calculating \(Z_{in}\) and \(Z_{out}\) at \(f = 10kHz\):
    \begin{align*}
      s &= j 2 \pi f = j 2 \pi (10kHz) \\
      Z_{in} &= R_1 + \frac{1}{sC_1} \\
      Z_{in} &= 3.54 \Omega + \frac{1}{j 2 \pi (10kHz)(1\mu F)} \\
      Z_{in} &\approx 3.54 \Omega - j 15.92 \Omega \\
      Z_{out} &= \frac{R_1 - sR_1^2C_1}{1 - (sR_1C_1)^2} \\
              &= \frac{3.54 \Omega - j 2 \pi (10kHz)(3.54 \Omega)^2(1\mu F)}{1 - (j 2 \pi (10kHz)(3.54 \Omega)(1\mu F))^2} \\
      Z_{out} &= \frac{3.54 - j0.787}{1 + 0.0494} \\
      Z_{out} &\approx 3.37 \Omega - j 0.75 \Omega \\
    \end{align*}
\end{enumerate}

\subsection{Applications}
\begin{itemize}
  \item \textbf{Anti-aliasing before ADCs:}
    Limits signal bandwidth so that frequency components above
    the sampling limit do not fold back into the baseband.
    \textit{Very common in microcontroller analog inputs.}

  \item \textbf{PWM to Analog Conversion:}
    Converts a pulse-width-modulated digital signal into a smooth
    DC or slowly varying voltage.
    \textit{Used in DACs made from GPIO pins.}

  \item \textbf{Power Supply Decoupling:}
    Reduces high-frequency noise on supply rails caused by
    switching regulators or digital logic.

  \item \textbf{Sensor Signal Smoothing:}
    Removes rapid noise from temperature, pressure, or light
    sensors where the physical quantity changes slowly.

  \item \textbf{Envelope Detection:}
    Extracts the slowly varying amplitude of a modulated signal
    after rectification.
    \textit{Used in AM demodulation and audio level detection.}
\end{itemize}
