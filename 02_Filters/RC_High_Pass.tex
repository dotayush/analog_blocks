\subsubsection{RC High Pass Filter}

\paragraph{Overview:}
Constructed from a resistor and capacitor in series, a RC
High Pass Filter allows high frequency signals to pass while
attenuating low frequency signals. It is a single pole
filter but can be used in multiple stages to create higher
order filters.

\subsubsubsection{Transfer Function}
Applying KCL at $V_{out}$ on Full Schematic, we get:
\[
  (V_{out} - V_{in}){sC_1} + (V_{out} - V_{ee}){G_1} = 0
\]
\paragraph{Superposition:}
\begin{enumerate}
  \item Due to \(V_{in}\) alone (\(V_{ee} = 0V\)):
    \begin{align*}
      (V_{out} - V_{in}){sC_1} + (V_{out} - 0){G_1} &= 0 \\
      V_{out}sC_1 - V_{in}sC_1 + V_{out}G_1 &= 0 \\
      V_{out}(sC_1 + G_1) - V_{in}sC_1 &= 0 \\
      V_{out}(sC_1 + G_1) &= V_{in}sC_1 \\
      \boxed{H_1(s) = \frac{N_1(s)}{D_1(s)} = \frac{V_{out_1}}{V_{in}} = \frac{sC_1}{sC_1 + G_1}}
    \end{align*}
  \item Due to \(V_{ee}\) alone (\(V_{in} = 0V\)):
    \begin{align*}
      (V_{out} - 0){sC_1} + (V_{out} - V_{ee}){G_1} &= 0 \\
      V_{out}sC_1 + V_{out}G_1 - V_{ee}G_1 &= 0 \\
      V_{out}(sC_1 + G_1) - V_{ee}G_1 &= 0 \\
      V_{out}(sC_1 + G_1) &= V_{ee}G_1 \\
      \boxed{H_2(s) = \frac{N_2(s)}{D_2(s)} = \frac{V_{out_2}}{V_{ee}} = \frac{G_1}{sC_1 + G_1}}
    \end{align*}
\end{enumerate}
\paragraph{Total Output:}
\begin{align*}
  V_{out} &= V_{out_1} + V_{out_2} \\
  &= H_1(s) V_{in} + H_2(s) V_{ee} \\
  &= \frac{sC_1}{sC_1 + G_1} V_{in} + \frac{G_1}{sC_1 + G_1} V_{ee}
\end{align*}

\subsubsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\)):} \\
    For both \(H_1(s)\) and \(H_2(s)\):
    \begin{align*}
      sC_1 + G_1 &= 0 \\
      sC_1 &= -G_1 \\
      s &= -\frac{G_1}{C_1} \\
    \end{align*}
    Negative real pole indicates a stable first-order system.
    Frequency corresponding to this pole:
    \begin{align*}
      s &= -2 \pi f_p \\
      -2 \pi f_p &= -\frac{G_1}{C_1} \\
      \boxed{f_p = \frac{G_1}{2 \pi C_1} = \frac{1}{2 \pi R_1 C_1}}
    \end{align*}
  \item \textbf{Zeroes (\(N(s) = 0\)):} \\
    For \(H_1(s)\):
    \begin{align*}
        N_1(s) = sC_1 = 0 \\
        s_z &= 0 \quad \text{[let \(s = s_z\)]} \\
        f_z &= \frac{|s_z|}{2\pi} = 0 Hz
    \end{align*}
    Zero at the origin for \(H_1(s)\). \\
    For \(H_2(s)\):
    \begin{align*}
      N_2(s) = G_1 = 0
    \end{align*}
    No finite zeroes for \(H_2(s)\).
  \item \textbf{DC Gain \(A_{DC}\):} \\
    For \(H_1(s)\):
    \begin{align*}
      H_1(0) &= \frac{0 \cdot C_1}{0 \cdot C_1 + G_1} = 0 \\
      \boxed{A_{DC_1} = 0 = -\infty dB}
    \end{align*}
    Zero DC gain for \(H_1(s)\). \\
    For \(H_2(s)\):
    \begin{align*}
      H_2(0) &= \frac{G_1}{0 \cdot C_1 + G_1} = 1 \\
      \boxed{A_{DC_2} = 1 = 0 dB}
    \end{align*}
    Unity DC gain for \(H_2(s)\).
  \item \textbf{Gain Bandwidth Product (GBW):}
    We use the DC gain from \(H_1(s)\) since it is
    the high pass characteristic:
    \begin{align*}
      GBW &= A_{DC_1} \cdot f_p \\
          &= 0 \cdot \frac{1}{2 \pi R_1 C_1} \\
      \boxed{GBW = 0}
    \end{align*}
  \item \textbf{Impedances:}
    Same as RC Low Pass Filter.
\end{enumerate}

\subsubsubsection{Question}
Design a high pass filter with a cutoff frequency of
\(f_p = 1kHz\) using a capacitor value of \(C_1 = 1\mu F\).
Also compute the input and output impedances at
\(f = 75Hz\).
\paragraph{Solution:}
\begin{enumerate}
  \item Calculating \(R_1\):
  \begin{align*}
    f_p &= \frac{1}{2 \pi R_1 C_1} \\
    R_1 &= \frac{1}{2 \pi f_p C_1} \\
    &= \frac{1}{2 \pi (1kHz)(1\mu F)} \\
    &= 159.15 \Omega \\
    \boxed{R_1 \approx 159 \Omega}
  \end{align*}
  \item Calculating \(Z_{in}\) and \(Z_{out}\) at \(f = 75Hz\):
  \begin{align*}
    \omega &= 2 \pi f = 2 \pi (75Hz) = 471.24 \text{ rad/s} \\
    Z_{in} &= R_1 - j \frac{1}{\omega C_1} \\
           &= 159 \Omega - j \frac{1}{(471.24)(1\mu F)} \\
           &\approx 159 \Omega - j 2123.72 \Omega \\
    Z_{out} &= \frac{R_1}{1 + (\omega R_1 C_1)^2} - j \frac{\omega R_1^2 C_1}{1 + (\omega R_1 C_1)^2} \\
            &= \frac{159 \Omega}{1 + (471.24 \cdot 159 \Omega \cdot 1\mu F)^2} - j \frac{(471.24)(159 \Omega)^2 (1\mu F)}{1 + (471.24 \cdot 159 \Omega \cdot 1\mu F)^2} \\
            &\approx 153.96 \Omega - j 11.52 \Omega
  \end{align*}
\end{enumerate}
