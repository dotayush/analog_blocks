\subsection{Pipeline ADC}\label{sec:adc_pipeline}

\subsubsection{Overview}
A chain of \(m\) stages, each a sample and hold, a coarse sub-ADC of
\(n_s\) bits, and an MDAC (Section~\ref{sec:dac_mdac}) forming the
amplified residue for the next stage. All stages work concurrently on
successive samples: throughput of one sample per clock, latency \(m\)
clocks.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}, 1.5 bit stage
(\(b \in \{-1, 0, 1\}\), input range \(\pm V_{REF}\)), the MDAC with \(C_1 = C_2\)
(Section~\ref{sec:dac_mdac}):
\begin{align*}
  b &= \begin{cases} -1 & V_{IN} < -\frac{V_{REF}}{4} \\ 0 & -\frac{V_{REF}}{4} \leq V_{IN} \leq \frac{V_{REF}}{4} \\ +1 & V_{IN} > \frac{V_{REF}}{4} \end{cases} \hspace{1cm}
  V_{RES} = 2 V_{IN} - b V_{REF}
\end{align*}
Range of \(V_{RES}\) in each region:
\begin{align*}
  b = -1:&\ V_{IN} \in [-V_{REF}, -\tfrac{V_{REF}}{4}) \Rightarrow V_{RES} = 2V_{IN} + V_{REF} \in [-V_{REF}, \tfrac{V_{REF}}{2}) \\
  b = 0:&\ V_{IN} \in [-\tfrac{V_{REF}}{4}, \tfrac{V_{REF}}{4}] \Rightarrow V_{RES} = 2V_{IN} \in [-\tfrac{V_{REF}}{2}, \tfrac{V_{REF}}{2}] \\
  b = +1:&\ V_{IN} \in (\tfrac{V_{REF}}{4}, V_{REF}] \Rightarrow V_{RES} = 2V_{IN} - V_{REF} \in (-\tfrac{V_{REF}}{2}, V_{REF}]
\end{align*}
so \(V_{RES} \in [-V_{REF}, V_{REF}]\), the next stage's input range. Unrolling
\(m\) stages (\(V_{RES,0} = V_{IN}\), \(V_{RES,k} = 2V_{RES,k-1} - b_k V_{REF}\)):
\begin{align*}
  V_{RES,1} &= 2V_{IN} - b_1 V_{REF} \\
  V_{RES,2} &= 2(2V_{IN} - b_1 V_{REF}) - b_2 V_{REF} = 2^2 V_{IN} - V_{REF}(b_1 2^1 + b_2 2^0) \\
  V_{RES,m} &= 2^m V_{IN} - V_{REF}\sum_{k=1}^{m} b_k 2^{m-k} \hspace{1cm} \text{(Each step doubles the previous sum and appends } b_k 2^0\text{)} \\
  V_{IN} &= V_{REF}\sum_{k=1}^{m} b_k 2^{-k} + \frac{V_{RES,m}}{2^m}
\end{align*}
so \(D = \sum_k b_k 2^{m-k}\) (an overlapping add, digits \(b_k\) may be \(-1\))
resolves \(V_{IN}\) to \(\frac{V_{REF}}{2^m}\), plus a final flash on \(V_{RES,m}\).
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: in the amplify phase each
    MDAC is a closed loop op-amp with \(F = \frac{C_2}{C_1 + C_2} = \frac{1}{2}\) (Section~\ref{sec:dac_mdac}, (3)):
    \begin{align*}
      H(s) &= 2\cdot\frac{A_0 F p_1}{s + p_1(1 + A_0 F)} \tag{1}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: from (1), \(s_p = -p_1(1 + A_0 F) \approx -2\pi F \cdot GB = -\pi GB\).
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(N(s) \neq 0\): no finite zero.
  \item \textbf{DC Gain (\(H(0)\))}: \(H(0) = 2\frac{A_0 F}{1 + A_0 F}\); finite op-amp gain scales the residue by
    \(\frac{AF}{1 + AF}\) (Section~\ref{sec:dac_mdac}):
    \begin{align*}
      V_{RES} &= (2 V_{IN} - b V_{REF})\frac{AF}{1 + AF} = (2 V_{IN} - b V_{REF})\left(1 - \frac{1}{1 + AF}\right) \\
      |V_{RES} - V_{RES,ideal}| &= \frac{|2 V_{IN} - b V_{REF}|}{1 + AF} \leq \frac{V_{REF}}{1 + AF} \approx \frac{V_{REF}}{AF} \hspace{1cm} [|V_{RES,ideal}| \leq V_{REF}]
    \end{align*}
    The back end resolves the remaining \(n - 1\) bits over \(2V_{REF}\), an LSB of
    \(\frac{2V_{REF}}{2^{n-1}} = \frac{V_{REF}}{2^{n-2}}\). Keeping the error below half of it:
    \begin{align*}
      \frac{V_{REF}}{AF} &< \frac{1}{2}\cdot\frac{V_{REF}}{2^{n-2}} = \frac{V_{REF}}{2^{n-1}} \hspace{1cm} \Rightarrow \hspace{1cm}
      AF > 2^{n-1} \hspace{1cm} \Rightarrow \hspace{1cm} A > 2^{n} \ \ [F = \tfrac{1}{2}]
    \end{align*}
    (\(F = \frac{C_2}{C_1 + C_2} = \frac{1}{2}\) for \(C_1 = C_2\), \(C_{in}\) neglected).
  \item \textbf{Settling} (Small Signal, Time Domain, AC Input Analysis (Q2)): residue error
    \(V_{REF} e^{-\pi GB\, t}\) below half a back end LSB \(\frac{V_{REF}}{2^{n-1}}\):
    \begin{align*}
      e^{-\pi GB\, t} &< 2^{-(n-1)} \hspace{1cm} \Rightarrow \hspace{1cm} t > \frac{(n - 1)\ln 2}{\pi GB}
    \end{align*}
    one sample per clock: \(f_s \leq \frac{1}{2t}\) (half a period to amplify).
  \item \textbf{Redundancy}: move the upper threshold to \(\frac{V_{REF}}{4} + d\). An input
    just below it still gets \(b = 0\), \(V_{RES} = 2\left(\frac{V_{REF}}{4} + d\right) = \frac{V_{REF}}{2} + 2d\), which stays
    \(\leq V_{REF}\) for \(d \leq \frac{V_{REF}}{4}\) (and symmetrically for \(d < 0\) and the
    lower threshold). The residue stays in range and later stages absorb the
    decision error, so cheap comparators with offsets up to
    \(\pm\frac{V_{REF}}{4}\) suffice.
  \item \textbf{Noise}: stage \(k\) sits behind a gain of \(2^{k-1}\) (stages
    \(1 \ldots k-1\)), so its noise referred to the input is divided by
    \(2^{k-1}\) (power by \(4^{k-1}\)). The first stage dominates and later stages
    are scaled down in capacitance and power.
  \item \textbf{Impedances}:
    \paragraph{At Input:} the first stage sampling capacitance \(C_1 + C_2\) (switched, \(\frac{kT}{C}\) sized).
  \item \textbf{Capacitor Mismatch}: \(C_1 = C(1 + e)\), \(C_2 = C\) in \(V_{RES} = \left(1 + \frac{C_1}{C_2}\right)V_{IN} - \frac{C_1}{C_2}bV_{REF}\):
    \begin{align*}
      V_{RES} &= (2 + e)V_{IN} - (1 + e)bV_{REF} = (2V_{IN} - bV_{REF}) + e(V_{IN} - bV_{REF})
    \end{align*}
    the error \(e(V_{IN} - bV_{REF})\), with \(|V_{IN} - bV_{REF}| \leq V_{REF}\) [from the regions of Q5], must stay below
    \(\frac{V_{REF}}{2^{n-1}}\) (as the Gain Error): \(|e| < 2^{-(n-1)}\); \(12\) bits need \(|e| < 0.05\%\) (or calibration).
  \item \textbf{Aperture Jitter}: full scale sine at \(f_{in}\), rms jitter \(t_j\) (Section~\ref{sec:dc_char}), kept
    below the quantisation noise:
    \begin{align*}
      SNR_{jitter,dB} = -20\log_{10}(2\pi f_{in} t_j) &> SQNR_{dB} = 6.02n + 1.76 \\
      t_j &< \frac{10^{-\frac{6.02n + 1.76}{20}}}{2\pi f_{in}}
    \end{align*}
    e.g.\ \(n = 12\), \(f_{in} = 50\,\)MHz: \(10^{-74.0/20} = 2.0 \times 10^{-4}\), \(t_j < \frac{2.0 \times 10^{-4}}{3.14 \times 10^{8}} = 0.63\,\)ps.
  \item \textbf{Advantages \& Limitations}: one sample per clock at high resolution (10--14 bits),
    redundancy relaxes comparators to \(\pm\frac{V_{REF}}{4}\), later stages scaled down; but latency \(m\)
    clocks, op-amps with \(A > 2^n\) and fast settling in every stage (power), and the first
    stage sets noise and linearity.
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item First stage: \(C_1 = C_2\) from \(\frac{kT}{C}\), \(A_0 > 2^n\), \(GB > \frac{2 f_s (n - 1)\ln 2}{\pi}\).
  \item Stage \(k\): capacitance and power scaled down by \(\approx 2\) to \(4\) per stage (noise referred
    by \(4^{k-1}\)).
\end{enumerate}

\subsubsection{Question}
12 bit pipeline of 1.5 bit stages, \(C_1 = C_2\). Find the minimum op-amp DC
gain of the first stage.
\paragraph{Solution:}
\begin{align*}
  A &> 2^{n} = 2^{12} = 4096 \hspace{1cm} (20\log_{10} 4096 = 72.2\,dB)
\end{align*}

\subsubsection{Code}
\begin{lstlisting}[language=Octave]
  clear; clc;

  % 1.5 bit stages, input range +/- Vref, comparator offsets up to Vref/4 tolerated
  Vref = 1; m = 10; rand("seed", 2);
  Vin = 0.3141; V = Vin; D = 0;
  for k = 1:m
    os = 0.2*Vref*(2*rand - 1);                    % comparator offset < Vref/4
    if V > Vref/4 + os, b = 1; elseif V < -Vref/4 - os, b = -1; else b = 0; end
    V = 2*V - b*Vref;
    D = 2*D + b;
  end
  printf('Vin = %.4f, D = %d, D/2^m*Vref = %.4f, residue/2^m = %.2e\n', Vin, D, D/2^m*Vref, V/2^m);
  printf('A > 2^12 = %d (%.1f dB)\n', 2^12, 20*log10(2^12));
  printf('cap mismatch limit (12 bit): %.3f %%, jitter (12 bit, 50 MHz): %.2f ps\n', 100*2^-11, 10^(-(6.02*12 + 1.76)/20)/(2*pi*50e6)*1e12);
\end{lstlisting}
