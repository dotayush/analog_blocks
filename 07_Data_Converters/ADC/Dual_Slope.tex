\subsection{Dual Slope ADC}\label{sec:adc_dual_slope}

\subsubsection{Overview}
An integrator (Section~\ref{sec:amp_integrator}) integrates \(V_{IN}\) for a
fixed \(N_1\) clock periods, then integrates \(-V_{REF}\) until its output returns
to zero, counting \(N_2\) periods. The ratio \(\frac{N_2}{N_1}\) is the result.
Slow, but extremely accurate and immune to \(R\), \(C\) and clock values.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, AC Input Analysis (Q6)}: the ideal
integrator (Section~\ref{sec:amp_integrator}) gives
\(V_x(t) = V_x(t_0) - \frac{1}{RC}\int_{t_0}^{t} V\,dt'\) for the applied \(V\); it starts at \(V_x(0) = 0\).
Define the average over phase 1, \(\overline{V_{IN}} = \frac{1}{T_1}\int_0^{T_1} V_{IN}\, dt\).
\begin{align*}
  \text{Phase 1 } (T_1 = N_1 T_{clk}):& \hspace{0.5cm} V_x(T_1) = 0 - \frac{1}{RC}\int_0^{T_1} V_{IN}\, dt = -\frac{\overline{V_{IN}} T_1}{RC} \\
  \text{Phase 2 } (T_2 = N_2 T_{clk}):& \hspace{0.5cm} V_x(T_1 + T_2) = V_x(T_1) - \frac{1}{RC}\int_{T_1}^{T_1 + T_2} (-V_{REF})\, dt = V_x(T_1) + \frac{V_{REF} T_2}{RC}
\end{align*}
Phase 2 ends when the comparator sees \(V_x = 0\):
\begin{align*}
  0 &= -\frac{\overline{V_{IN}} T_1}{RC} + \frac{V_{REF} T_2}{RC} \\
  V_{REF} T_2 &= \overline{V_{IN}} T_1 \\
  T_2 &= \frac{\overline{V_{IN}}}{V_{REF}} T_1 \\
  N_2 T_{clk} &= \frac{\overline{V_{IN}}}{V_{REF}} N_1 T_{clk} \hspace{0.5cm} \Rightarrow \hspace{0.5cm}
  N_2 = N_1 \frac{\overline{V_{IN}}}{V_{REF}} \tag{1}
\end{align*}
\(R\), \(C\) and \(T_{clk}\) cancel (they only need to be constant over one conversion).
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: phase 1 is a linear average
    of \(V_{IN}\) over \(T_1\). For
    \(V_{IN} = e^{j2\pi f t}\) (a sine is the imaginary part of
    \(A e^{j\theta} e^{j2\pi f t}\)):
    \begin{align*}
      H(f) = \frac{1}{T_1}\int_0^{T_1} e^{j2\pi f t}\, dt &= \frac{1}{T_1}\cdot\frac{e^{j2\pi f T_1} - 1}{j2\pi f} \\
      &= \frac{e^{j\pi f T_1}}{T_1}\cdot\frac{e^{j\pi f T_1} - e^{-j\pi f T_1}}{j2\pi f} \hspace{1cm} [\text{factoring } e^{j\pi f T_1}] \\
      &= \frac{e^{j\pi f T_1}}{T_1}\cdot\frac{2j\sin(\pi f T_1)}{j2\pi f} = e^{j\pi f T_1}\frac{\sin(\pi f T_1)}{\pi f T_1} \hspace{1cm} [e^{jx} - e^{-jx} = 2j\sin x] \\
      |H(f)| &= \left|\frac{\sin(\pi f T_1)}{\pi f T_1}\right| \tag{2}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles}: (2) is entire (no poles): a finite time average.
  \item \textbf{Zeroes (\(N(f) = 0\))}: \(\sin(\pi f T_1) = 0\), \(f \neq 0\): \(f_z = \frac{k}{T_1}\) (\(k = 1, 2, \ldots\)); \(T_1 = 100\,\)ms puts
    nulls at every multiple of \(10\,\)Hz, so \(50\,\)Hz, \(60\,\)Hz and all their harmonics are rejected.
  \item \textbf{DC Gain (\(H(0)\))}: \(\lim_{f \to 0}\frac{\sin(\pi f T_1)}{\pi f T_1} = 1\); code gain \(\frac{N_2}{V_{IN}} = \frac{N_1}{V_{REF}}\) [from (1)].
  \item \textbf{Conversion Time}: \(N_1 = 2^n\) and \(N_2 \leq N_1\) (as
    \(\overline{V_{IN}} \leq V_{REF}\)), so at most \((N_1 + N_2) T_{clk} = 2 \cdot 2^n T_{clk} = 2^{n+1} T_{clk}\).
  \item \textbf{Output Swing (Max)}: the integrator peak at full scale,
    \(|V_x(T_1)| = \frac{V_{REF} T_1}{RC} \leq |V_{OL}|\), sets \(RC \geq \frac{V_{REF} T_1}{|V_{OL}|}\).
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(Z_{in} = R\) (virtual ground, Section~\ref{sec:amp_integrator}).
  \item \textbf{Offset}: an integrator offset \(V_{OS}\) (at \(V_+\)) makes the capacitor current \(\frac{V - V_{OS}}{R}\); with the
    comparator threshold at the reset level, phase 1 integrates \(V_{IN} - V_{OS}\) and phase 2 integrates \(-V_{REF} - V_{OS}\):
    \begin{align*}
      (\overline{V_{IN}} - V_{OS})T_1 &= (V_{REF} + V_{OS})T_2 \hspace{1cm} \Rightarrow \hspace{1cm} N_2 = N_1\frac{\overline{V_{IN}} - V_{OS}}{V_{REF} + V_{OS}}
    \end{align*}
    an offset and a gain error; an auto-zero phase stores \(V_{OS}\) on a capacitor and subtracts it.
  \item \textbf{Advantages \& Limitations}: \(R\), \(C\), \(T_{clk}\) cancel (high accuracy), mains rejection by
    choice of \(T_1\), inherently monotonic (a counter); but slow (\(2^{n+1} T_{clk}\)), integrator and
    comparator offsets need an auto-zero phase, and dielectric absorption of \(C\) limits linearity.
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(T_1 = \frac{k}{f_{mains}}\) for both mains (e.g. \(100\,\)ms); \(f_{clk} = \frac{2^n}{T_1}\).
  \item \(RC \geq \frac{V_{REF} T_1}{|V_{OL}|}\).
\end{enumerate}

\subsubsection{Question}
16 bits, \(f_{clk} = 1\,\)MHz, \(V_{REF} = 2\,V\), \(V_{OL} = -10\,V\). Find the worst conversion time and
the smallest \(RC\).
\paragraph{Solution:}
\begin{align*}
  t_{conv} &= 2^{n+1} T_{clk} = 2^{17} \cdot 1\,\mu s = 131072\,\mu s = 131\,ms \\
  T_1 &= 2^{16} \cdot 1\,\mu s = 65.5\,ms \hspace{1cm} RC \geq \frac{2\,V \cdot 65.5\,ms}{10\,V} = 13.1\,ms
\end{align*}

\subsubsection{Code}
\begin{lstlisting}[language=Octave]
  clear; clc;
  pkg load symbolic;

  syms f T1 positive
  H = int(exp(1i*2*sym(pi)*f*sym('t')), sym('t'), 0, T1)/T1;
  printf('=== |H(f)| ===\n'); pretty(simplify(abs(rewrite(simplify(H), 'sin'))))

  % conversion with T1 = 100 ms (N1 = 1e5 counts at 1 MHz): 50 Hz ripple rejected
  Vref = 2; N1 = 1e5; t = (0:N1 - 1)*1e-6;
  Vin = 0.75 + 0.1*sin(2*pi*50*t + 0.3);   % 0.75 V plus 50 Hz ripple
  x = -sum(Vin)*1e-6;                       % phase 1 (RC = 1 normalised)
  N2 = round(-x/(Vref*1e-6));               % phase 2 counts to zero
  printf('N2 = %d, N1*0.75/Vref = %.0f (ripple averaged out)\n', N2, N1*0.75/Vref);
  printf('16 bit: worst conversion time = %.0f ms, RC >= %.1f ms\n', 2^17*1e-3, 2*65.536/10);
\end{lstlisting}
