\subsection{String (R-String) DAC}\label{sec:dac_string}

\subsubsection{Overview}
\(N = 2^n\) equal resistors \(R\) between \(V_{REF}\) and ground give every
level; a decoder closes the one switch \(S_D\) selecting tap \(D\). The
simplest DAC, inherently monotonic.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}: the string
    current is \(\frac{V_{REF}}{N R}\); tap \(D\) sits \(D\) resistors above ground:
    \begin{align*}
      I_{STR} &= \frac{V_{REF}}{N R} \\
      V_{OUT} = V_D &= D R \cdot I_{STR} = D R \cdot \frac{V_{REF}}{N R} = \frac{D}{N} V_{REF} = \frac{D}{2^n} V_{REF}
    \end{align*}
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}:
    looking into tap \(D\) with \(V_{REF}\) shorted, \(DR\) is in parallel with
    \((N - D)R\); the closed switch adds its on resistance \(R_{sw}\) in series
    and the load is \(C_L\):
    \begin{align*}
      R_{out}(D) &= \frac{D R \cdot (N - D) R}{D R + (N - D) R} = \frac{D (N - D) R^2}{N R} = \frac{D (N - D)}{N} R \\
      H(s) &= \frac{V_{out}}{V_D} = \frac{\frac{1}{s C_L}}{R_{out}(D) + R_{sw} + \frac{1}{s C_L}}
      = \frac{1}{1 + s (R_{out}(D) + R_{sw}) C_L} \tag{1}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: from (1),
    \(s_p = -\frac{1}{(R_{out}(D) + R_{sw}) C_L}\), code dependent (slowest at mid scale, below).
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(N(s) = 1 \neq 0\): no finite zero.
  \item \textbf{DC Gain (\(H(0)\))}: \(H(0) = 1\); the static gain from code to output is
    \(\frac{V_{OUT}}{D} = \frac{V_{REF}}{2^n} = 1\,\text{LSB}\).
  \item \textbf{Settling (Small Signal, Time Domain, AC Input Analysis (Q2))}:
    \(R_{out}\) peaks at \(D = \frac{N}{2}\):
    \begin{align*}
      R_{out}(D) &= \frac{(N D - D^2) R}{N} \hspace{1cm}
      \frac{dR_{out}}{dD} = \frac{(N - 2D)R}{N} = 0 \hspace{0.5cm} \Rightarrow \hspace{0.5cm} D = \frac{N}{2} \\
      R_{out,\max} &= \frac{\frac{N}{2}\left(N - \frac{N}{2}\right)}{N} R = \frac{N^2/4}{N} R = \frac{N}{4} R
    \end{align*}
    A full scale step \(V_{REF}\) through \(H(s)\) (\(R_{sw} \ll R_{out}\),
    \(T_c = R_{out} C_L\)) leaves the error \(V_{REF} e^{-t/T_c}\); settling to
    \(\frac{1}{2}\) LSB \(= \frac{V_{REF}}{2^{n+1}}\):
    \begin{align*}
      V_{REF} e^{-t/T_c} &= \frac{V_{REF}}{2^{n+1}} \hspace{0.5cm} \Rightarrow \hspace{0.5cm}
      e^{t/T_c} = 2^{n+1} \hspace{0.5cm} \Rightarrow \hspace{0.5cm}
      t = T_c \ln(2^{n+1}) = (n + 1) \ln(2)\, T_c
    \end{align*}
  \item \textbf{Impedances}:
    \paragraph{At Output:} \(Z_{out}(D) = R_{out}(D) + R_{sw} = \frac{D(N - D)}{N} R + R_{sw}\), up to \(\frac{N}{4} R + R_{sw}\);
    a resistive load draws current from the string (buffer it).
    \paragraph{At Reference:} \(Z_{REF} = N R\) (constant, code independent).
  \item \textbf{Integral Non-Linearity (INL)} from mismatch: let \(X = \sum_{j < D} R_j\),
    \(Y = \sum_{j \geq D} R_j\), \(V_D = V_{REF}\frac{X}{X + Y}\).
    \begin{align*}
      dV_D &= V_{REF}\frac{Y\, dX - X\, dY}{(X + Y)^2} \\
      dV_D &= V_{REF}\frac{dX - dY}{2 N R} \hspace{1cm} \left[D = \tfrac{N}{2},\ X = Y = \tfrac{NR}{2}\right]
    \end{align*}
    The second line substitutes \(X = Y = \frac{NR}{2}\) in the first:
    \[
      V_{REF}\frac{\frac{NR}{2}\, dX - \frac{NR}{2}\, dY}{(NR)^2} = V_{REF}\frac{\frac{NR}{2}(dX - dY)}{N^2 R^2} = V_{REF}\frac{dX - dY}{2 N R}
    \]
    With \(\frac{N}{2}\) independent resistors in each sum (Section~\ref{sec:dc_char}, Mismatch):
    \begin{align*}
      \text{var}(dX - dY) &= \text{var}(dX) + \text{var}(dY) = \frac{N}{2}(S_R R)^2 + \frac{N}{2}(S_R R)^2 = N (S_R R)^2 \\
      SD(dV_D) &= \frac{V_{REF}\sqrt{N} S_R R}{2 N R} = \frac{V_{REF} S_R}{2\sqrt{N}} \\
      SD(INL_{N/2}) &= \frac{SD(dV_D)}{V_{REF}/N} = \frac{V_{REF} S_R}{2\sqrt{N}} \cdot \frac{N}{V_{REF}} = \frac{\sqrt{N}}{2} S_R = \frac{\sqrt{2^n}}{2} S_R \ \text{LSB}
    \end{align*}
  \item \textbf{Monotonicity (DNL)}: tap \(D + 1\) adds \(R_D\) to \(X\):
    \(V_{D+1} - V_D = V_{REF}\frac{X + R_D}{\sum R} - V_{REF}\frac{X}{\sum R} = R_D \cdot \frac{V_{REF}}{\sum R} > 0\) for any
    positive resistors.
  \item \textbf{DNL}: step \(k\) is the voltage across one resistor \(R_k = R(1 + e_k)\):
    \(DNL_k = \frac{R_k I_{STR}}{X_{LSB}} - 1 \approx e_k\) [\(I_{STR} R = X_{LSB}\), total string error neglected], so \(SD(DNL) = S_R\), independent of \(n\).
  \item \textbf{Power}: \(P = \frac{V_{REF}^2}{NR}\) static, every code; a larger \(R\) saves power but slows settling (\(\frac{N}{4}R\,C_L\)).
  \item \textbf{Advantages \& Limitations}: inherently monotonic, simple, code independent reference
    load; but \(2^n\) resistors and \(2^n\) switches (practical to 8--10 bits), code dependent output
    impedance (\(\frac{N}{4}R\) at mid scale) and \(INL\) growing as \(\frac{\sqrt{2^n}}{2} S_R\).
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(n\) from resolution, \(S_R \leq \frac{2 \cdot INL_{\max}}{\sqrt{2^n}}\) (one \(SD\)).
  \item \(R \leq \frac{4 T_c}{N C_L}\) for the settling time \(t = (n + 1)\ln(2)\, T_c\).
\end{enumerate}

\subsubsection{Question}
\(n = 10\), \(R = 100\,\Omega\), \(S_R = 0.2\%\), \(C_L = 2\,\)pF. Find the worst
\(R_{out}\), the time to settle to \(\frac{1}{2}\) LSB and \(SD(INL)\) at mid scale.
\paragraph{Solution:}
\(N = 2^{10} = 1024\), \(R_{sw}\) neglected:
\begin{align*}
  R_{out,\max} &= \frac{N}{4} R = \frac{1024}{4} \cdot 100 = 25.6\,k\Omega \\
  T_c &= R_{out,\max} C_L = 25.6\,k\Omega \cdot 2\,pF = 51.2\,ns \\
  t &= T_c \ln(2^{11}) = 51.2\,ns \cdot \ln(2048) = 51.2\,ns \cdot 7.62 = 390\,ns \\
  SD(INL) &= \frac{\sqrt{2^{10}}}{2} S_R = \frac{32}{2} \cdot 0.002 = 0.032\,\text{LSB}
\end{align*}

\subsubsection{Code}
\begin{lstlisting}[language=Octave]
  clear; clc;
  pkg load symbolic;

  syms D N R Rsw CL s
  Rout = simplify(D*R*(N - D)*R/(D*R + (N - D)*R));
  printf('=== Rout(D) ===\n'); pretty(Rout)
  Dm = solve(diff(Rout, D) == 0, D);
  printf('=== worst code, Rout max ===\n'); pretty(Dm); pretty(simplify(subs(Rout, D, Dm)))

  % ideal transfer of a 3 bit string, numerically
  n = 3; Nn = 2^n; Vref = 1;
  printf('V_D for D = 0..%d: %s\n', Nn - 1, mat2str((0:Nn - 1)/Nn*Vref, 4));

  % Question: n = 10, R = 100, SR = 0.2 %, CL = 2 pF
  Nn = 1024; Tc = Nn/4*100*2e-12;
  printf('Rout max = %.1f k, Tc = %.1f ns, t = %.0f ns, SD(INL) = %.3f LSB\n', ...
         Nn/4*100/1e3, Tc*1e9, Tc*log(2^11)*1e9, sqrt(Nn)/2*0.002);
  % Monte Carlo check of SD(INL) at mid scale
  randn("seed", 1); r = randn(2000, Nn)*0.002 + 1;
  X = sum(r(:, 1:Nn/2), 2); Y = sum(r(:, Nn/2 + 1:end), 2);
  inl = (X./(X + Y) - 0.5)*Nn;
  printf('Monte Carlo SD(INL) = %.3f LSB\n', std(inl));
\end{lstlisting}
