\section*{Symbols and Labels}
\addcontentsline{toc}{section}{Symbols and Labels}
Every symbol and label that appears in this chapter. \(X\) stands for whichever quantity a converter handles (a voltage or a current).
\begin{symtable}
\multicolumn{3}{|l|}{\textbf{Characterisation}} \\ \hline
\(n,\ N\) & resolution in bits and number of levels \(N = 2^n\) & - \\ \hline
\(b_k,\ b_{n-1} \ldots b_0\) & bit \(k\) of the code (1 or 0; \(-1, 0, +1\) in a 1.5 bit pipeline stage); \(b_{n-1}\) is the MSB & - \\ \hline
\(MSB,\ LSB\) & most and least significant bit & - \\ \hline
\(D\) & integer value of the code \(\sum_k b_k 2^k\); in the string DAC also the selected tap & - \\ \hline
\(X_{REF},\ X_{LSB}\) & full scale reference and one LSB step \(\frac{X_{REF}}{2^n}\) & V or A \\ \hline
\(X_{IN},\ X_{OUT},\ X_{OUT}(D)\) & converter input, output, output for code \(D\) & V or A \\ \hline
\(X_k,\ X_0,\ X_j\) & actual output (DAC) or transition (ADC) for code \(k\), the first one, code \(j\) & V or A \\ \hline
\(e,\ \overline{e^2}\) & quantisation error and its mean square value & V, V\(^2\) \\ \hline
\(x(t),\ \overline{x^2}\) & input signal and its mean square (power) & V, V\(^2\) \\ \hline
\(SQNR,\ SQNR_{dB}\) & signal to quantisation noise ratio, in dB & - \\ \hline
\(f_s,\ f_B,\ OSR\) & sampling frequency, signal bandwidth, oversampling ratio \(\frac{f_s}{2f_B}\) & Hz, Hz, - \\ \hline
\(f_{in},\ f_{in,\max}\) & input frequency and its largest value & Hz \\ \hline
\(DNL, INL,\ DNL_k,\ INL_k\) & differential and integral non-linearity of code \(k\) & LSB \\ \hline
\(SINAD,\ ENOB\) & signal to noise and distortion ratio, effective number of bits & dB, bits \\ \hline
\(t_j,\ A,\ SNR_{jitter}\) & rms sampling time error, sine amplitude, jitter limited SNR & s, V, - \\ \hline
\(SFDR\) & spurious free dynamic range & dB \\ \hline
\(S_X,\ S_R,\ S_I,\ S_C\) & relative standard deviation of a unit element: generic, resistor, current, capacitor & - \\ \hline
\(SD(\cdot),\ Var(\cdot)\) & standard deviation, variance & - \\ \hline
\(dX,\ dX_i,\ m\) & error of an element, of element \(i\); number of elements summed & V or A, - \\ \hline
\multicolumn{3}{|l|}{\textbf{DACs}} \\ \hline
\(V_{REF},\ V_{OUT},\ V_{DAC}\) & reference, output, DAC output voltage & V \\ \hline
\(R,\ I_{STR},\ I_{string}\) & unit resistor and string current & \(\Omega\), A \\ \hline
\(S_D,\ S_1 \ldots S_{N-1}\) & switch connecting tap \(D\); the tap switches & - \\ \hline
\(R_{sw},\ R_{out},\ R_{out,\max},\ R_{tot}\) & switch resistance, output resistance and its largest value, total string resistance & \(\Omega\) \\ \hline
\(DR,\ NR\) & resistances \(D \cdot R\) and \(N \cdot R\) & \(\Omega\) \\ \hline
\(X,\ Y\) & (string DAC) string resistance below and above the selected tap & \(\Omega\) \\ \hline
\(R_j,\ R_D\) & resistor \(j\) of the string; the one added at tap \(D + 1\) & \(\Omega\) \\ \hline
\(V_D,\ dV_D\) & string voltage at tap \(D\) and its error & V \\ \hline
\(C_L,\ T_c,\ t\) & load capacitance, settling time constant, settling time & F, s, s \\ \hline
\(V_k,\ V_0 \ldots V_{n-1},\ V_{th,k},\ R_{th,k}\) & R-2R node voltages; Thevenin voltage and resistance at node \(k\) & V, \(\Omega\) \\ \hline
\(I_k,\ R_F\) & R-2R leg current; feedback resistor of the current mode output & A, \(\Omega\) \\ \hline
\(G,\ G_M,\ G_r,\ e_M,\ e_r,\ w\) & leg conductance; MSB leg and rest of the ladder; their relative errors; MSB weight & S, -, - \\ \hline
\(I_u,\ dI_k,\ dI_i,\ g_u,\ r_u\) & unit current, error of source \(k\) (\(i\)), unit output conductance and resistance & A, A, S, \(\Omega\) \\ \hline
\(I_{OUT},\ I_{OUT+},\ I_{OUT-},\ R_L,\ G_L\) & output current, its two differential branches, load resistance and conductance & A, \(\Omega\), S \\ \hline
\(a,\ aD\) & \(g_u R_L\) and its product with the code (output resistance error) & - \\ \hline
\(INL_{\max},\ dINL/dD\) & largest INL and its slope over the code & LSB \\ \hline
\(C_u,\ C_k,\ C_p,\ C_{tot}\) & unit capacitor, capacitor of bit \(k\) (\(2^k C_u\)), top plate parasitic, total & F \\ \hline
\(C_1,\ C_2,\ C_{in}\) & MDAC sampling and feedback capacitors, op-amp input capacitance & F \\ \hline
\(C_b,\ C_{tot,split}\) & bridge capacitor and total area of the split array & F \\ \hline
\(V_x\) & DAC or SAR top plate voltage; integrator output (dual slope) & V \\ \hline
\(A,\ F\) & op-amp gain and feedback factor & - \\ \hline
\(k,\ T,\ kT\) & Boltzmann constant, temperature, thermal energy & J/K, K, J \\ \hline
\(V^{(k)},\ V^{(-1)}\) & serial / pipeline DAC output after step \(k\); its initial value & V \\ \hline
\(L,\ n_q,\ STF,\ NTF\) & modulator order, quantiser bits, signal and noise transfer functions & - \\ \hline
\multicolumn{3}{|l|}{\textbf{ADCs}} \\ \hline
\(V_{IN},\ V_{REF}\) & input and reference voltage & V \\ \hline
\(V_{T,k},\ T_k,\ T_1 \ldots T_{N-1}\) & flash threshold \(k\) and comparator output \(k\) (thermometer code) & V, - \\ \hline
\(V_{os},\ V_{os,k}\) & comparator offset, of comparator \(k\) & V \\ \hline
\(C_{in,comp}\) & comparator input capacitance & F \\ \hline
\(n_1,\ n_2,\ D_1,\ D_2\) & coarse and fine bits and codes of a two-step flash & - \\ \hline
\(V_{RES},\ V_{RES,k},\ V_{RES,ideal}\) & residue, residue after stage \(k\), its ideal value & V \\ \hline
\(b,\ m\) & pipeline stage decision and number of stages & - \\ \hline
\(e,\ ne\) & inter-stage gain error and its \(n\)-fold accumulation & - \\ \hline
\(f_{clk},\ T_{clk}\) & clock frequency and period & Hz, s \\ \hline
\(V^{(k)}\) & cyclic ADC residue after cycle \(k\) & V \\ \hline
\(V_{DAC}^{(k)}\) & SAR trial level for bit \(k\) & V \\ \hline
\(A_p,\ M,\ V_a, V_b, V_{T,a}, V_{T,b},\ V_j\) & preamp gain, interpolation factor, two preamp outputs and thresholds, interpolated tap \(j\) & -, -, V \\ \hline
\(F,\ V_{fold},\ N(F),\ N_{\min}\) & number of folds, folded signal, comparator count, its minimum & -, V, -, - \\ \hline
\(N_1,\ N_2,\ T_1,\ T_2,\ \overline{V_{IN}}\) & dual slope: phase 1 and phase 2 clock counts and durations, average input & -, s, V \\ \hline
\(R,\ C,\ RC\) & integrator resistor, capacitor and time constant & \(\Omega\), F, s \\ \hline
\(\phi_1,\ \phi_2\) & clock phases (switch positions) & - \\ \hline
\(H(f)\) & averaging (sinc) response & - \\ \hline
\(U,\ V,\ Y,\ E\) & modulator input, output, loop filter output, quantisation error & V \\ \hline
\(z,\ z^{-1},\ T\) & z transform variable \(e^{sT}\), one sample delay, sample period \(\frac{1}{f_s}\) & -, -, s \\ \hline
\(P_n,\ P_s\) & in band noise power and signal power & V\(^2\) \\ \hline
\(H(z),\ H(s),\ \mathcal{Z}^{-1}\) & discrete and continuous loop filter, inverse z transform & - \\ \hline
\(\Sigma\) & summing node (block diagrams) & - \\ \hline
\multicolumn{3}{|l|}{\textbf{Other}} \\ \hline
\(n_s\) & bits of a pipeline sub-ADC & - \\ \hline
\(G_m\) & transconductance of a \(G_m\)-C integrator & S \\ \hline
\(Q\) & charge stored on the capacitor array top plate & C \\ \hline
\(V_{out},\ V_1\) & op-amp output in the current mode R-2R; R-2R node 1 & V \\ \hline
\(t_0,\ t\) & start of an integration, time & s \\ \hline
\(d, dF, dN, dY, dG_M, dG_r, dR_{out}\) & derivative operator and small changes of the named quantities & - \\ \hline
\(aN,\ aND,\ nS\) & products of the named quantities; nanosiemens & - \\ \hline
\(J,\ K,\ \mu, \omega, \pi\) & joule, kelvin (units), micro prefix, angular frequency, pi & - \\ \hline
\end{symtable}
