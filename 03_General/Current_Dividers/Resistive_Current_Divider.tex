\subsection{Resistive Current Divider}

This block is constructed from two resistors in parallel and is used
to attenuate a current signal. It is one of the simplest and most
commonly used circuit in electronics.

\subsubsection{Transfer Function}
Only linear devices are present, so the \textbf{Large Signal, Time Domain,
DC Input Analysis (Q5)} and the \textbf{Small Signal, Frequency Domain, AC
Input Analysis (Q4)} give the same result. Let \(V_{in}\) be the voltage of
the node between \(G_1\) and \(G_2\) (both other ends are grounded).
Applying KCL at this node we get:
\begin{align*}
  I_{in} &= I_{G_1} + I_{G_2} \\
  I_{in} &= (V_{in} - 0)G_1 + (V_{in} - 0)G_2 \\
  I_{in} &= V_{in}G_1 + V_{in}G_2 \\
  I_{in} &= V_{in}(G_1 + G_2) \\
  V_{in} &= \frac{I_{in}}{G_1 + G_2}
\end{align*}
For \(I_{G_1}\):
\begin{align*}
  I_{G_1} &= (V_{in} - 0)G_1 \\
  I_{G_1} &= V_{in}G_1 \\
  I_{G_1} &= \frac{I_{in}}{G_1 + G_2}G_1 \hspace{1cm} \text{(Substituting \(V_{in}\))} \\
  H_1(s) = \frac{N_1(s)}{D_1(s)} = \frac{I_{G_1}}{I_{in}} &= \frac{G_1}{G_1 + G_2} = \frac{R_2}{R_1 + R_2}
\end{align*}
For \(I_{G_2}\):
\begin{align*}
  I_{G_2} &= (V_{in} - 0)G_2 \\
  I_{G_2} &= V_{in}G_2 \\
  I_{G_2} &= \frac{I_{in}}{G_1 + G_2}G_2 \hspace{1cm} \text{(Substituting \(V_{in}\))} \\
  H_2(s) = \frac{N_2(s)}{D_2(s)} = \frac{I_{G_2}}{I_{in}} &= \frac{G_2}{G_1 + G_2} = \frac{R_1}{R_1 + R_2}
\end{align*}
(The \(R\) forms follow by multiplying numerator and denominator by
\(R_1 R_2\): \(\frac{G_1 R_1 R_2}{(G_1 + G_2) R_1 R_2} = \frac{R_2}{R_2 + R_1}\).)
\paragraph{Total Output:}
\begin{align*}
  I_{in} &= I_{G_1} + I_{G_2} \\
  I_{in} &= I_{in} \frac{G_1}{G_1 + G_2} + I_{in} \frac{G_2}{G_1 + G_2} \\
  I_{in} &= I_{in} \frac{G_1 + G_2}{G_1 + G_2} = I_{in} \hspace{1cm} \text{(KCL is satisfied)}
\end{align*}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Resistor Ratio (\(\frac{G_1}{G_2}\)):}
    We know,
    \begin{align*}
      I_{G_1} &= (V_{in} - 0)G_1 \\
      I_{G_2} &= (V_{in} - 0)G_2
    \end{align*}
    Dividing both equations we get:
    \begin{align*}
      \frac{I_{G_1}}{I_{G_2}} &= \frac{V_{in}G_1}{V_{in}G_2} = \frac{G_1}{G_2} \\
      \frac{G_1}{G_2} &= \frac{I_{G_1}}{I_{G_2}}
    \end{align*}
    Assume \(I_{G_1} = I_{out}\), then our KCL equation
    becomes:
    \begin{align*}
      I_{in} &= I_{out} + I_{G_2} \\
      I_{G_2} &= I_{in} - I_{out}
    \end{align*}
    Substituting \(I_{G_2}\) in the previous equation
    we get:
    \[
      \frac{G_1}{G_2} = \frac{I_{out}}{I_{in} - I_{out}}
    \]
  \item \textbf{Impedances:}
    \paragraph{At Input:} Looking in from the source node, \(G_1\) and
    \(G_2\) both go to ground, so they are in parallel:
      \begin{align*}
        Z_{in} &= \frac{1}{G_1 + G_2} \\
        Z_{in} &= \frac{1}{\frac{1}{R_1} + \frac{1}{R_2}} \hspace{1cm} \text{(Substituting \(G_1 = \frac{1}{R_1}\) and \(G_2 = \frac{1}{R_2}\))} \\
        Z_{in} &= \frac{1}{\frac{R_1 + R_2}{R_1 R_2}} \\
        Z_{in} &= \frac{R_1 R_2}{R_1 + R_2}
      \end{align*}
    \paragraph{At Output:} Looking into the \(G_1\) branch with \(I_{in}\)
    opened, \(G_1\) is in series with \(G_2\):
      \begin{align*}
        Z_{out} &= \frac{1}{G_1} + \frac{1}{G_2} \\
        Z_{out} &= R_1 + R_2 \hspace{1cm} \text{(Substituting \(G_1 = \frac{1}{R_1}\) and \(G_2 = \frac{1}{R_2}\))}
      \end{align*}
  \item \textbf{Loading Effect:}
    The output of the current divider is a current signal, so a load
    \(R_L\) is inserted in series with the branch whose current we use.
    The branch conductance becomes \(G_1' = \frac{1}{R_1 + R_L}\):
    \begin{align*}
      \frac{I_{out}}{I_{in}} &= \frac{G_1'}{G_1' + G_2} \\
      &= \frac{\frac{1}{R_1 + R_L}}{\frac{1}{R_1 + R_L} + \frac{1}{R_2}} \\
      &= \frac{\frac{1}{R_1 + R_L}}{\frac{R_2 + R_1 + R_L}{(R_1 + R_L) R_2}} \\
      &= \frac{R_2}{R_1 + R_2 + R_L}
    \end{align*}
    To minimize loading effects, the load impedance should be at least
    10 times \textbf{smaller} than the output impedance
    \(Z_{out} = R_1 + R_2\) of the current divider.
\end{enumerate}

\subsubsection{Question}
Design a resistive current divider to attenuate a 5A current signal to
10mA. Establish the ratio of the resistors and calculate the input
and output impedance.
\paragraph{Solution:}
\begin{enumerate}
  \item Calculate the resistor ratio (\(\frac{G_1}{G_2}\)):
    \begin{align*}
      \frac{G_1}{G_2} &= \frac{I_{out}}{I_{in} - I_{out}} \\
      \frac{G_1}{G_2} &= \frac{0.01A}{5A - 0.01A} \\
      \frac{G_1}{G_2} &= \frac{0.01A}{4.99A} \\
      \frac{G_1}{G_2} &\approx 0.002004 \\
      \frac{R_2}{R_1} &\approx 0.002004 \hspace{1cm} \text{(Substituting \(G_1 = \frac{1}{R_1}\) and \(G_2 = \frac{1}{R_2}\))}
    \end{align*}
  \item Calculate the resistor values:
    Assume \(R_1 = 5k\Omega\), then:
    \begin{align*}
      R_2 &= 0.002004 \cdot R_1 \\
      R_2 &= 0.002004 \cdot 5k\Omega \\
      R_2 &\approx 10.02 \Omega
    \end{align*}
  \item Calculate the input impedance:
    \begin{align*}
      Z_{in} &= \frac{R_1 R_2}{R_1 + R_2} \\
      Z_{in} &= \frac{5k\Omega \cdot 10.02 \Omega}{5k\Omega + 10.02 \Omega} \\
      Z_{in} &= \frac{50100 \Omega^2}{5010.02 \Omega} \\
      Z_{in} &\approx 10.0 \Omega
    \end{align*}
    Check: \(V_{in} = I_{out} R_1 = 0.01A \cdot 5k\Omega = 50V\) and
    \(Z_{in} = \frac{V_{in}}{I_{in}} = \frac{50V}{5A} = 10\Omega\).
  \item Calculate the output impedance:
    \begin{align*}
      Z_{out} &= R_1 + R_2 \\
      Z_{out} &= 5k\Omega + 10.02 \Omega \\
      Z_{out} &\approx 5.01 k\Omega
    \end{align*}
\end{enumerate}
