\subsection{Inductive Current Divider}

This block is constructed from two inductors in parallel and is used
to attenuate a voltage signal.

\subsubsection{Transfer Function}
Applying KCL at the node between \(L_1\) and \(L_2\) we get:
\begin{align*}
  I_{in} &= I_{L_1} + I_{L_2} \\
  I_{in} &= \frac{V_{in} - 0}{sL_1} + \frac{V_{in} - 0}{sL_2} \\
  I_{in} &= \frac{V_{in}}{sL_1} + \frac{V_{in}}{sL_2} \\
  I_{in} &= V_{in}\left(\frac{1}{sL_1} + \frac{1}{sL_2}\right) \\
  I_{in} &= V_{in}\left(\frac{L_2 + L_1}{sL_1 L_2}\right) \\
  V_{in} &= \frac{I_{in}}{\frac{L_2 + L_1}{sL_1 L_2}}
\end{align*}
For \(I_{L_1}\):
\begin{align*}
  I_{L_1} &= \frac{V_{in} - 0}{sL_1} \\
  I_{L_1} &= \frac{V_{in}}{sL_1} \\
  I_{L_1} &= \frac{I_{in}}{\frac{L_2 + L_1}{sL_1 L_2}} \cdot \frac{1}{sL_1} \\
  I_{L_1} &= \frac{I_{in} L_2}{L_1 + L_2} \\
  H_1(s) = \frac{N_1(s)}{D_1(s)} = \frac{I_{L_1}}{I_{in}} &= \frac{L_2}{L_1 + L_2}
\end
{align*}
For \(I_{L_2}\):
\begin{align*}
  I_{L_2} &= \frac{V_{in} - 0}{sL_2} \\
  I_{L_2} &= \frac{V_{in}}{sL_2} \\
  I_{L_2} &= \frac{I_{in}}{\frac{L_2 + L_1}{sL_1 L_2}} \cdot \frac{1}{sL_2} \\
  I_{L_2} &= \frac{I_{in} L_1}{L_1 + L_2} \\
  H_2(s) = \frac{N_2(s)}{D_2(s)} = \frac{I_{L_2}}{I_{in}} &= \frac{L_1}{L_1 + L_2}
\end{align*}
\paragraph{Total Output:}
\begin{align*}
  I_{in} &= I_{L_1} + I_{L_2} \\
  I_{in} &= I_{in} \frac{L_2}{L_1 + L_2} + I_{in} \frac{L_1}{L_1 + L_2}
\end{align*}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Inductor Ratio (\(\frac{L_1}{L_2}\)):}
    We know,
    \begin{align*}
      I_{L_1} &= \frac{V_{in} - 0}{sL_1} \\
      I_{L_2} &= \frac{V_{in} - 0}{sL_2}
    \end{align*}
    Dividing the two equations we get:
    \begin{align*}
      \frac{I_{L_1}}{I_{L_2}} = \frac{L_2}{L_1} \\
      \frac{L_1}{L_2} = \frac{I_{L_2}}{I_{L_1}}
    \end{align*}
    Assume \(I_{L_1} = I_{out}\), then our KCL equation
    becomes:
    \begin{align*}
      I_{in} &= I_{out} + I_{L_2} \\
      I_{L_2} &= I_{in} - I_{out}
    \end{align*}
    Substituting \(I_{L_2}\) in the previous equation
    we get:
    \[
      \frac{L_1}{L_2} = \frac{I_{in} - I_{out}}{I_{out}}
    \]
  \item \textbf{Impedance's:}
    \paragraph{At Input:}
      \begin{align*}
        Z_{in} &= \frac{1}{\frac{1}{sL_1} + \frac{1}{sL_2}} \\
        Z_{in} &= \frac{1}{\frac{L_2 + L_1}{sL_1 L_2}} \\
        Z_{in} = \frac{sL_1 L_2}{L_1 + L_2}
      \end{align*}
    \paragraph{At Output:}
      \begin{align*}
        Z_{out} &= sL_1 + sL_2 \\
        Z_{out} = s(L_1 + L_2)
      \end{align*}
\end{enumerate}

\subsubsection{Question}
Determine the output current \(I_{L_1}\) and \(I_{L_2}\) for the following values:
\begin{itemize}
  \item \(L_1 = 10mH\)
  \item \(L_2 = 20mH\)
  \item \(I_{in} = 10mA\)
\end{itemize}
\paragraph{Solution:}
\begin{enumerate}
  \item For \(I_{L_1}\):
    \begin{align*}
      I_{L_1} &= I_{in} \frac{L_2}{L_1 + L_2} \\
      I_{L_1} &= 10mA \frac{20mH}{10mH + 20mH} \\
      I_{L_1} &= 10mA \frac{20}{30} \\
      I_{L_1} &= 6.67mA
    \end{align*}
  \item For \(I_{L_2}\):
    \begin{align*}
      I_{L_2} &= I_{in} \frac{L_1}{L_1 + L_2} \\
      I_{L_2} &= 10mA \frac{10mH}{10mH + 20mH} \\
      I_{L_2} &= 10mA \frac{10}{30} \\
      I_{L_2} &= 3.33mA
    \end{align*}
\end{enumerate}
