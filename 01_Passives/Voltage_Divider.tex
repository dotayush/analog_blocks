\subsubsection{Voltage Divider}

\paragraph{Overview:}
Constructed from two resistors in series, a voltage
divider is a simple circuit that turns a large voltage
into a smaller one.

\subsubsubsection{Transfer Function}
Applying KCL at $V_{out}$ of Full Schematic, we get:
\[
  (V_{out} - V_{in}){G_1} + (V_{out} - V_{ee}){G_2} = 0
\]
\paragraph{Superposition:}
\begin{enumerate}
  \item Due to \(V_{in}\) alone (\(V_{ee} = 0V\)):
    \begin{align*}
      (V_{out} - V_{in}){G_1} + (V_{out} - 0){G_2} &= 0 \\
      V_{out}G_1 - V_{in}G_1 + V_{out}G_2 &= 0 \\
      V_{out}(G_1 + G_2) - V_{in}G_1 &= 0 \\
      V_{out}(G_1 + G_2) &= V_{in}G_1 \\
      \boxed{H_1(s) = \frac{N_1(s)}{D_1(s)} = \frac{V_{out_1}}{V_{in}} = \frac{G_1}{G_1 + G_2}}
    \end{align*}
  \item Due to \(V_{ee}\) alone (\(V_{in} = 0V\)):
    \begin{align*}
      (V_{out} - 0){G_1} + (V_{out} - V_{ee}){G_2} &= 0 \\
      V_{out}G_1 + V_{out}G_2 - V_{ee}G_2 &= 0 \\
      V_{out}(G_1 + G_2) - V_{ee}G_2 &= 0 \\
      V_{out}(G_1 + G_2) &= V_{ee}G_2 \\
      \boxed{H_2(s) = \frac{N_2(s)}{D_2(s)} = \frac{V_{out_2}}{V_{ee}} = \frac{G_2}{G_1 + G_2}}
    \end{align*}
\end{enumerate}
\paragraph{Total Output:}
\begin{align*}
  V_{out} &= V_{out_1} + V_{out_2} \\
  V_{out} &= V_{in} \frac{G_1}{G_1 + G_2} + V_{ee} \frac{G_2}{G_1 + G_2}
\end{align*}

\subsubsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Resistor Ratio (\(\frac{R_2}{R_1}\)):}
    From KCL at \(V_{out}\) of Full Schematic:
    \begin{align*}
      G_1(V_{out} - V_{in}) - G_2(V_{ee} - V_{out}) &= 0 \\
      G_1(V_{out} - V_{in}) &= G_2(V_{ee} - V_{out}) \\
      \frac{G_1}{G_2} &= \frac{V_{ee} - V_{out}}{V_{out} - V_{in}} \\
      \hspace{1cm} \text{Substituting } G_1 = \frac{1}{R_1} \text{ and } G_2 = \frac{1}{R_2} \text{ gives:} \hspace{1cm} \\
      \boxed{\frac{R_2}{R_1} = \frac{V_{ee} - V_{out}}{V_{out} - V_{in}}}
    \end{align*}
  \item \textbf{Impedances:}
    Substituting \(G_1 = \frac{1}{R_1}\) and \(G_2 = \frac{1}{R_2}\):
    \paragraph{At Input:}
      \begin{align*}
        Z_{in} &= \frac{1}{G_1} + \frac{1}{G_2} \\
        \boxed{Z_{in} = R_1 + R_2}
      \end{align*}
    \paragraph{At Output:}
      \begin{align*}
        Z_{out} &= \frac{1}{\frac{1}{R_1} + \frac{1}{R_2}} \\
        Z_{out} &= \frac{1}{\frac{R_1 + R_2}{R_1 R_2}} \\
        \boxed{Z_{out} = \frac{R_1 R_2}{R_1 + R_2}}
      \end{align*}
\end{enumerate}

\subsubsubsection{Question}
Design a voltage divider to step down a \(10V\) supply
to \(3.3V\). Calculate the input and output impedances
of the voltage divider.
\paragraph{Solution:}
\begin{enumerate}
  \item Calculate the resistor ratio (\(\frac{R_2}{R_1}\)):
    \begin{align*}
      \frac{R_2}{R_1} &= \frac{V_{ee} - V_{out}}{V_{out} - V_{in}} \\
      &= \frac{0V - 3.3V}{3.3V - 10V} \\
      &= \frac{-3.3V}{-6.7V} \\
    \frac{R_2}{R_1} &\approx 0.4925
    \end{align*}
  \item Fix one of the resistors (\(R_2 = 10 k\Omega\)) to a standard
    value and calculate the other resistor (\(R_1\)):
    \begin{align*}
      R_1 &= \frac{R_2}{0.4925} \\
      &= \frac{10 k\Omega}{0.4925} \\
      R_1 &\approx 20.3 k\Omega
    \end{align*}
  \item Calculating Input and Output Impedances:
    \[
        \begin{aligned}
        Z_{in} &= R_1 + R_2 \\
        &= 20.3 k\Omega + 10 k\Omega \\
        Z_{in} &\approx 30.3 k\Omega
        \end{aligned}
      \]
      \[
        \begin{aligned}
          Z_{out} &= \frac{R_1 R_2}{R_1 + R_2} \\
          Z_{out} &= \frac{(20.3 k\Omega)(10 k\Omega)}{20.3 k\Omega + 10 k\Omega} \\
          Z_{out} &\approx 6.77 k\Omega
        \end{aligned}
      \]
\end{enumerate}
