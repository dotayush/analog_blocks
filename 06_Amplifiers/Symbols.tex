\section*{Symbols and Labels}
\addcontentsline{toc}{section}{Symbols and Labels}
Every symbol and label that appears in this chapter. A number after a device quantity names the transistor (\(g_{m1}\) belongs to \(M_1\)). Upper case node subscripts are large signal (Q5), lower case are small signal (Q4).
\begin{symtable}
\multicolumn{3}{|l|}{\textbf{General}} \\ \hline
\(V_{in},\ V_{out},\ V_{IN},\ V_{OUT}\) & small signal and large signal input and output voltage & V \\ \hline
\(V_{dd},\ V_{cc}\) & positive supply (MOS, BJT) & V \\ \hline
\(H,\ H(s),\ N(s),\ D(s)\) & transfer function, its numerator and denominator & - \\ \hline
\(A_V,\ A_{V0}\) & voltage gain; its value at DC & - \\ \hline
\(Z_{in},\ Z_{out},\ Z_{in,k}\) & input impedance, output impedance, input impedance of input \(k\) & \(\Omega\) \\ \hline
\(I_{test},\ V_{test},\ I_t,\ V_t\) & test current and voltage applied to find an impedance & A, V \\ \hline
\(s,\ j,\ \omega,\ f,\ \pi\) & Laplace variable, \(\sqrt{-1}\), angular frequency, frequency, pi & rad/s, -, rad/s, Hz, - \\ \hline
\(p_1, p_2,\ f_{p1}, f_{p2}\) & first (dominant) and second pole, as angular frequency and in Hz & rad/s, Hz \\ \hline
\(p_{in},\ p_{out}\) & pole at the input node and at the output node & rad/s \\ \hline
\(z,\ s_z,\ f_z\) & a zero, as a value of \(s\), in Hz & rad/s, Hz \\ \hline
\(a_0, a_1, a_2\) & coefficients of a second order denominator \(a_0 + a_1 s + a_2 s^2\) & - \\ \hline
\(\omega_n,\ Q,\ f_n\) & natural angular frequency, quality factor, natural frequency & rad/s, -, Hz \\ \hline
\(GB,\ f_{-3dB}\) & gain-bandwidth product, \(-3\,dB\) bandwidth & Hz \\ \hline
\(SR\) & slew rate & V/s \\ \hline
\(PM\) & phase margin \(180^\circ + \angle A\) at the unity loop gain frequency & \(^\circ\) \\ \hline
\(dV_{in},\ dV_{out},\ dV_{OUT},\ dV_{O1},\ dV_{O2},\ dV_{ID}\) & small changes (slopes) of the named voltages & V \\ \hline
\(e\) & Euler's number; in the difference amplifier the fractional resistor error & - \\ \hline
\multicolumn{3}{|l|}{\textbf{Common Stages}} \\ \hline
\(M_1,\ M_2,\ Q_1\) & the MOS or bipolar device of the stage, its cascode device & - \\ \hline
\(V_G, V_S, V_B, V_E,\ V_g, V_s, V_b, V_e\) & large and small signal gate, source, base, emitter node voltages & V \\ \hline
\(G_S,\ R_S\) & conductance and resistance of the driving source (CS, CG, CC, CB) & S, \(\Omega\) \\ \hline
\(G_D, R_D,\ G_C, R_C\) & drain and collector load & S, \(\Omega\) \\ \hline
\(G_E, R_E\) & emitter resistor (degeneration or follower load) & S, \(\Omega\) \\ \hline
\(G_L, R_L\) & load conductance and resistance (\(G_L = g_{o,sink}\) for a current sink load) & S, \(\Omega\) \\ \hline
\(G_o,\ R_o\) & total output conductance \(G_D + g_o\) and its inverse & S, \(\Omega\) \\ \hline
\(C_L,\ C_{gs},\ C_{gd},\ C_{sb},\ C_s\) & load, gate-source, gate-drain, source-body capacitance; \(C_s = C_{gs} + C_{sb}\) & F \\ \hline
\(C_{bc}\) & base-collector capacitance & F \\ \hline
\(I_D,\ I_d,\ I_G\) & large and small signal drain current, gate current & A \\ \hline
\(I_B, I_C, I_E,\ I_b, I_c, I_e\) & large and small signal base, collector, emitter currents & A \\ \hline
\(I_S,\ V_T,\ V_{BE},\ V_{CE}\) & BJT saturation current, thermal voltage, base-emitter and collector-emitter voltage & A, V \\ \hline
\(A_F,\ B_F\) & common base and common emitter current gain & - \\ \hline
\(V_{GS}, V_{DS}, V_{SB}, V_{SG}, V_{SD}\) & large signal MOS voltages & V \\ \hline
\(V_{gs}, V_{ds}, V_{bs}, V_{be}\) & small signal device voltages & V \\ \hline
\(V_{th},\ V_{ov},\ V_{ov,edge}\) & threshold, overdrive, overdrive at the edge of saturation & V \\ \hline
\(V_X\) & auxiliary \(V_{IN} - V_{th}\) in the CS triode solution; cascode node in the cascode stages & V \\ \hline
\(k_n,\ K_P,\ K_{CLM},\ W, L\) & MOS device factor, process transconductance, channel length modulation, width, length & A/V\(^2\), A/V\(^2\), V\(^{-1}\), m \\ \hline
\(g_m, g_{mb}, g_o, g_{be},\ r_o, r_{be}\) & transconductance, body transconductance, output conductance, base-emitter conductance; resistances & S, \(\Omega\) \\ \hline
\(V_B,\ V_{B2}\) & gate bias of a common gate or cascode device & V \\ \hline
\multicolumn{3}{|l|}{\textbf{Functional (Op-Amp) Blocks}} \\ \hline
\(V_+, V_-\) & non-inverting and inverting op-amp inputs & V \\ \hline
\(A, A(s), A_0\) & op-amp open loop gain, as a function of \(s\), at DC & - \\ \hline
\(F\) & feedback factor & - \\ \hline
\(R_{id},\ R_o\) & op-amp differential input and output resistance & \(\Omega\) \\ \hline
\(G_1, G_2, R_1, R_2\) & input and feedback conductance and resistance & S, \(\Omega\) \\ \hline
\(G_3, G_4, R_3, R_4\) & difference amplifier non-inverting divider & S, \(\Omega\) \\ \hline
\(G_F, R_F,\ G_k, R_k,\ V_k,\ N\) & summing feedback element, element and voltage of input \(k\), number of inputs & S, \(\Omega\), V, - \\ \hline
\(V_1,\ V_2\) & the two inputs of the difference amplifier & V \\ \hline
\(a,\ b\) & difference amplifier weights of \(V_2\) and \(V_1\) & - \\ \hline
\(k\) & nominal ratio \(\frac{R_2}{R_1} = \frac{R_4}{R_3}\) & - \\ \hline
\(V_{ic},\ V_{id},\ A_c,\ A_d,\ CMRR\) & common mode and differential input, their gains, rejection ratio & V, - \\ \hline
\(C,\ G_s,\ R_s\) & integrator/differentiator capacitor; differentiator series resistor & F, S, \(\Omega\) \\ \hline
\(V_{OS}\) & op-amp input offset voltage & V \\ \hline
\(p_1\) & op-amp dominant pole & rad/s \\ \hline
\(V_s,\ V_{s,\max}\) & buffer input step height and largest step that does not slew & V \\ \hline
\(t_{-10V}\) & time for the integrator output to reach \(-10\,V\) & s \\ \hline
\multicolumn{3}{|l|}{\textbf{Differential Blocks}} \\ \hline
\(V_{i1}, V_{i2},\ V_{I1}, V_{I2}\) & small and large signal gate voltages of the pair & V \\ \hline
\(V_{ic}, V_{id},\ V_{IC}, V_{ID}\) & small and large signal common mode and differential input & V \\ \hline
\(V_{IC,\min},\ V_{IC,\max}\) & input common mode range & V \\ \hline
\(V_{o1}, V_{o2},\ V_{O1}, V_{O2}\) & small and large signal outputs of the two halves & V \\ \hline
\(V_{O,\min},\ V_{O,\max}\) & output swing limits & V \\ \hline
\(V_{O1,2},\ I_{D1,2},\ V_{GS1,2}\) & the same quantity for side 1 and side 2 & V, A \\ \hline
\(I_T\) & tail current & A \\ \hline
\(I_{Dk},\ I_{dk}\) & large and small signal drain current of \(M_k\) & A \\ \hline
\(I_{dd}\) & supply current & A \\ \hline
\(g_{mk},\ g_{ok},\ r_{ok},\ g_{mbk}\) & transconductance, output conductance and resistance, body transconductance of \(M_k\) & S, \(\Omega\) \\ \hline
\(g_{m2}'\) & \(g_{m2} + g_{mb2}\), transconductance including body effect & S \\ \hline
\(g_{o,tail},\ g_{oB},\ g_{oL},\ r_{oB},\ r_{oL}\) & output conductance (resistance) of the tail, the folding sink, a load & S, \(\Omega\) \\ \hline
\(K_{P,n}, K_{P,p},\ K_{CLM,n}, K_{CLM,p}\) & NMOS and PMOS process transconductance and channel length modulation & A/V\(^2\), V\(^{-1}\) \\ \hline
\(V_{ovk},\ V_{ov,T},\ V_{ov,B}\) & overdrive of \(M_k\), of the tail, of the folding sink & V \\ \hline
\(V_{SGk},\ V_{GSk},\ V_{DSk},\ V_{DSk,\min}\) & device voltages of \(M_k\) and the smallest saturated \(V_{DS}\) & V \\ \hline
\(V_x,\ V_s,\ V_X,\ V_S\) & cascode node and common source node (small, large signal) & V \\ \hline
\(V_{B3}, V_{B5}, V_{B7}, V_{BB}\) & cascode and sink gate biases & V \\ \hline
\(C_x,\ C_f\) & capacitance at the mirror or cascode node, at the fold node & F \\ \hline
\(C_{gsk},\ C_{gd1},\ C_{db1},\ C_{db,B}\) & gate-source of \(M_k\), gate-drain and drain-body of \(M_1\), drain-body of the sink & F \\ \hline
\(Z_{dn},\ Z_{up}\) & impedance looking down into the NMOS stack and up into the PMOS stack & \(\Omega\) \\ \hline
\(Z_{dn,telescopic},\ Z_{out,plain}\) & the same for the telescopic and the plain (unboosted) cascode & \(\Omega\) \\ \hline
\(G_m\) & short circuit transconductance of a cascode & S \\ \hline
\(R_c\) & plain cascode output resistance \(g_{m2} r_{o2} r_{o1}\) & \(\Omega\) \\ \hline
\(A_b, A_{b0}, A_{bn}, A_{bp}\) & auxiliary (boost) amplifier gain; its DC value; NMOS and PMOS boosters & - \\ \hline
\(M_A,\ g_{mA},\ r_{oA}\) & device of a single stage booster and its parameters & -, S, \(\Omega\) \\ \hline
\(p_b,\ \omega_b,\ p_d,\ z_d\) & booster pole and unity gain frequency; doublet pole and zero & rad/s \\ \hline
\(V_{ref},\ V_{refn},\ V_{refp}\) & booster reference voltages & V \\ \hline
\(V_{g2}, V_{G2},\ V_{g3}, V_{g5}\) & boosted gate voltages & V \\ \hline
\(V_f, V_F,\ V_u, V_U\) & fold node; node between \(M_5\) and \(M_7\) (small, large signal) & V \\ \hline
\(R_f, G_f\) & \(r_{o1} \parallel r_{oB}\) at the fold node and its conductance & \(\Omega\), S \\ \hline
\(I_B,\ I_{DB}\) & folding sink current & A \\ \hline
\(I_{in},\ I_{sc}\) & signal current injected at the fold node; short circuit output current & A \\ \hline
\(B\) & mirror gain of the current mirror op-amp & - \\ \hline
\(V_a, V_b, V_c\) & mirror nodes of the current mirror op-amp & V \\ \hline
\(p_a,\ p_c\) & mirror poles at \(V_a\) and \(V_c\) & rad/s \\ \hline
\(G_1,\ G_2,\ C_1\) & two-stage: first and second stage output conductance, first stage node capacitance & S, F \\ \hline
\(C_c,\ R_z\) & Miller compensation capacitor and nulling resistor & F, \(\Omega\) \\ \hline
\(I_5,\ I_7\) & two-stage tail current and second stage sink current & A \\ \hline
\(V_1\) & two-stage first stage output node & V \\ \hline
\(x\) & normalised crossover frequency or a substitution variable as defined where used & - \\ \hline
