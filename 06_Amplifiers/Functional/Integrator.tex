\subsection{Integrator}\label{sec:amp_integrator}

\subsubsection{Overview}
An inverting amplifier (Section~\ref{sec:amp_inverting}) with a capacitor \(C\) as the
feedback element. Op-amp model: \(V_{out} = A(V_+ - V_-)\), \(A = A_0\) in Q5,
\(A = A(s) = \frac{A_0 p_1}{s + p_1}\) in Q4.

\subsubsection{Schematic}

  \draw (n) |- (-0.5,1.6) to[capacitor, l={$C$}] (1.6,1.6) -| (A.out);

  \draw (A.+) -- ++(-0.6,0) node[ground]{};
  \draw (A.out) to[short, -o] ++(0.8,0) node[right]{$V_{out}$};

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}: \(A = A_0\), \(G_F\)
    absent, constant \(V_{IN}\) applied at \(t = 0\), \(V_{OUT}(0) = 0\).
    \begin{align*}
      V_{OUT} &= A_0(0 - V_-) \hspace{1cm} \Rightarrow \hspace{1cm} V_- = -\frac{V_{OUT}}{A_0} \tag{1}
    \end{align*}
    Applying KCL at \(V_-\) (capacitor current \(C\frac{d(V_- - V_{OUT})}{dt}\)):
    \begin{align*}
      (V_- - V_{IN}) G_1 + C\frac{d(V_- - V_{OUT})}{dt} &= 0 \\
      \left(-\frac{V_{OUT}}{A_0} - V_{IN}\right) G_1 - C\left(1 + \frac{1}{A_0}\right)\frac{dV_{OUT}}{dt} &= 0 \hspace{1cm} [\text{from (1)}] \\
      C\left(1 + \frac{1}{A_0}\right)\frac{dV_{OUT}}{dt} &= -\left(\frac{V_{OUT}}{A_0} + V_{IN}\right) G_1 \\
      \frac{dV_{OUT}}{dt} &= -\frac{G_1}{C(1 + A_0)}\left(V_{OUT} + A_0 V_{IN}\right)
    \end{align*}
    With \(U = V_{OUT} + A_0 V_{IN}\) (\(\frac{dU}{dt} = \frac{dV_{OUT}}{dt}\), \(V_{IN}\) constant) and
    \(T_c = R_1 C(1 + A_0)\):
    \begin{align*}
      \frac{dU}{dt} &= -\frac{U}{T_c} \hspace{1cm} \Rightarrow \hspace{1cm} \frac{dU}{U} = -\frac{dt}{T_c} \\
      \int_{U(0)}^{U(t)} \frac{dU}{U} &= -\int_0^t \frac{dt'}{T_c} \hspace{1cm} [\text{Integrating both sides}] \\
      \ln\frac{U(t)}{U(0)} &= -\frac{t}{T_c} \hspace{1cm} \Rightarrow \hspace{1cm} U(t) = U(0) e^{-\frac{t}{T_c}} = A_0 V_{IN} e^{-\frac{t}{T_c}} \hspace{1cm} [U(0) = 0 + A_0 V_{IN}] \\
      V_{OUT}(t) &= -A_0 V_{IN}\left(1 - e^{-\frac{t}{T_c}}\right) \tag{2}
    \end{align*}
    For \(t \ll T_c\), \(e^{-\frac{t}{T_c}} \approx 1 - \frac{t}{T_c}\):
    \begin{align*}
      V_{OUT}(t) &\approx -A_0 V_{IN}\frac{t}{T_c} = -\frac{A_0}{1 + A_0}\cdot\frac{V_{IN}}{R_1 C} t \xrightarrow{A_0 \to \infty} -\frac{V_{IN}}{R_1 C} t
    \end{align*}
    a ramp, valid while \(V_{OL} \leq V_{OUT} \leq V_{OH}\).
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: KCL at \(V_-\)
    with \(V_- = -\frac{V_{out}}{A(s)}\):
    \begin{align*}
      (V_- - V_{in}) G_1 + (V_- - V_{out}) sC &= 0 \\
      -\frac{V_{out}}{A(s)}(G_1 + sC) - V_{out} sC &= V_{in} G_1 \\
      H(s) = \frac{V_{out}}{V_{in}} &= -\frac{G_1}{sC + \frac{(G_1 + sC)(s + p_1)}{A_0 p_1}} \hspace{1cm} \left[\tfrac{1}{A(s)} = \tfrac{s + p_1}{A_0 p_1}\right] \\
      &= -\frac{G_1 A_0 p_1}{sC A_0 p_1 + (G_1 + sC)(s + p_1)} \\
      H(s) &= -\frac{\frac{G_1}{C} A_0 p_1}{s^2 + s\left(p_1(1 + A_0) + \frac{G_1}{C}\right) + \frac{G_1}{C} p_1} \tag{3}
    \end{align*}
    \(A_0 \to \infty\):
    \begin{align*}
      H(s) &\to -\frac{G_1}{sC} = -\frac{1}{s R_1 C} \\
      V_{out}(t) &= -\frac{1}{R_1 C}\int_0^t V_{in}(t')\, dt' + V_{out}(0) \hspace{1cm} [\text{Inverse Laplace}, \tfrac{1}{s} \leftrightarrow \textstyle\int_0^t]
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: \(s^2 + b s + c = 0\) with
    \(b = p_1(1 + A_0) + \frac{G_1}{C}\), \(c = \frac{G_1}{C} p_1\), \(b^2 \gg c\) [widely
    separated roots]:
    \begin{align*}
      s_{p1} &\approx -\frac{c}{b} = -\frac{\frac{G_1}{C} p_1}{p_1(1 + A_0) + \frac{G_1}{C}} \approx -\frac{1}{R_1 C(1 + A_0)} \hspace{1cm} [p_1(1 + A_0) \gg \tfrac{G_1}{C}] \\
      s_{p2} &\approx -b \approx -p_1(1 + A_0) \approx -2\pi GB
    \end{align*}
    \(f_{p1} = \frac{1}{2\pi R_1 C(1 + A_0)}\) (the \(T_c\) of (2)), \(f_{p2} \approx GB\).
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(N(s) = -\frac{G_1}{C} A_0 p_1 \neq 0\): no finite zero.
  \item \textbf{DC Gain (\(H(0)\))}: \(s = 0\) in (3),
    \(H(0) = -\frac{\frac{G_1}{C} A_0 p_1}{\frac{G_1}{C} p_1} = -A_0\): a real integrator is a low pass.
  \item \textbf{Gain-Bandwidth Product (\(GBW\))}:
    \begin{align*}
      GBW &= |H(0)| \cdot f_{p1} = \frac{A_0}{2\pi R_1 C(1 + A_0)} \approx \frac{1}{2\pi R_1 C}
    \end{align*}
    the unity gain frequency: between \(f_{p1}\) and \(f_{p2}\),
    \(H(j\omega) \approx -\frac{1}{j\omega R_1 C}\), \(|H| = 1\) at \(f = \frac{1}{2\pi R_1 C}\).
  \item \textbf{Feedback Factor (\(F\))}: \(V_{in} = 0\), KCL at \(V_-\):
    \begin{align*}
      V_- G_1 + (V_- - V_{out}) sC &= 0 \hspace{1cm} \Rightarrow \hspace{1cm} F(s) = \frac{sC}{G_1 + sC} = \frac{s R_1 C}{1 + s R_1 C}
    \end{align*}
    \(F \to 0\) at DC (no DC feedback), \(F \to 1\) at high frequency.
  \item \textbf{DC Drift (Large Signal, Time Domain, DC Input Analysis (Q5))}: offset
    \(V_{OS}\) at \(V_+\), \(V_{IN} = 0\), so \(V_- = V_{OS}\). KCL at \(V_-\):
    \begin{align*}
      (V_{OS} - 0) G_1 + C\frac{d(V_{OS} - V_{OUT})}{dt} &= 0 \\
      \frac{dV_{OUT}}{dt} &= \frac{V_{OS}}{R_1 C} \hspace{1cm} [V_{OS} \text{ constant}] \\
      \int_{V_{OS}}^{V_{OUT}(t)} dV_{OUT} &= \int_0^t \frac{V_{OS}}{R_1 C} dt' \hspace{1cm} [\text{Integrating both sides}, V_{OUT}(0) = V_{OS}] \\
      V_{OUT}(t) &= V_{OS} + \frac{V_{OS}}{R_1 C} t
    \end{align*}
    a ramp to saturation. \(G_F\) in parallel with \(C\) bounds it; with \(V_- = 0\):
    \begin{align*}
      (0 - V_{in}) G_1 + (0 - V_{out})(G_F + sC) &= 0 \\
      H(s) &= -\frac{G_1}{G_F + sC} = -\frac{\frac{1}{R_1}}{\frac{1 + s R_F C}{R_F}} = -\frac{R_F}{R_1} \cdot \frac{1}{1 + s R_F C}
    \end{align*}
    the single pole active low pass (Section~\ref{sec:filt_active_sp}), integrating for
    \(f \gg \frac{1}{2\pi R_F C}\); \(V_{OS}\) now sees the DC gain \(1 + \frac{R_F}{R_1}\).
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(V_- \to 0\), \(Z_{in} = \frac{V_{in}}{(V_{in} - 0) G_1} = R_1\).
    \paragraph{At Output:} as Section~\ref{sec:amp_inverting}, \(Z_{out} = \frac{R_o}{1 + A F(s)}\).
  \item \textbf{Output Swing (Max)}: with constant \(V_{IN}\) the ramp reaches the rail
    \(V_{OL}\) (for \(V_{IN} > 0\)) at \(t_{sat} = \frac{R_1 C |V_{OL}|}{V_{IN}}\).
  \item \textbf{Slew Rate (Large Signal, Time Domain, AC Input Analysis (Q6))}: the output
    slope \(\frac{|V_{IN}|}{R_1 C} \leq SR\), i.e.\ \(|V_{IN}| \leq SR \cdot R_1 C\).
  \item \textbf{Phase Error}: the ideal integrator has \(-90^\circ\) (\(-\frac{1}{j\omega R_1 C}\)). (3) divided by the
    ideal \(-\frac{G_1}{sC}\), with \(\omega_{GB} = p_1(1 + A_0) + \frac{G_1}{C} \approx A_0 p_1 = 2\pi GB\):
    \begin{align*}
      \frac{H(s)}{H_{ideal}(s)} &= \frac{A_0 p_1\, s}{s^2 + s\,\omega_{GB} + \frac{G_1}{C} p_1}
       \approx \frac{1}{1 + \frac{s}{\omega_{GB}} + \frac{G_1 p_1}{C\omega_{GB} s}} \hspace{1cm} [A_0 p_1 \approx \omega_{GB}] \\
      \frac{H(j\omega)}{H_{ideal}(j\omega)} &\approx \frac{1}{1 + j\left(\frac{\omega}{\omega_{GB}} - \frac{\omega_{l}}{\omega}\right)} \hspace{1cm} \left[\omega_l = \frac{G_1 p_1}{C\omega_{GB}} \approx \frac{1}{A_0 R_1 C}\right] \\
      \phi_{err} &= -\tan^{-1}\left(\frac{\omega}{\omega_{GB}} - \frac{\omega_l}{\omega}\right)
    \end{align*}
    finite \(GB\) adds lag at high frequency, finite \(A_0\) adds lead at low frequency; the error is
    smallest at \(\omega = \sqrt{\omega_l \omega_{GB}}\).
  \item \textbf{Advantages \& Limitations}: exact integration (\(-\frac{1}{s R_1 C}\)) between
    \(f_{p1}\) and \(GB\); but no DC feedback, so offset and bias currents ramp the output
    to a rail (needs \(G_F\) or a reset switch), and the output is inverted.
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(R_1 C = \frac{1}{2\pi f_u}\) for the required unity gain frequency \(f_u\).
  \item \(R_F \gg R_1\) so \(\frac{1}{2\pi R_F C}\) is below the lowest signal frequency.
  \item Check \(|V_{IN}| \leq SR \cdot R_1 C\) and \(f_u \ll GB\).
\end{enumerate}

\subsubsection{Question}
\(R_1 = 10\,k\Omega\), \(C = 10\,nF\), \(V_{OUT}(0) = 0\), ideal op-amp. A \(1\,V\) step is
applied. Find the output slope, the time to reach \(-10\,V\) and the unity gain
frequency.
\paragraph{Solution:}
\begin{align*}
  R_1 C &= 10^4\,\Omega \cdot 10^{-8}\,F = 10^{-4}\,s \\
  V_{OUT}(t) &= -\frac{1}{R_1 C}\int_0^t 1\,V\, dt' = -\frac{1\,V}{10^{-4}\,s} t = -10^4\,\frac{V}{s}\cdot t \\
  t_{-10V} &= \frac{10\,V}{10^4\,V/s} = 1\,ms \\
  f_u &= \frac{1}{2\pi R_1 C} = \frac{1}{2\pi \cdot 10^{-4}\,s} = 1.59\,kHz
\end{align*}
