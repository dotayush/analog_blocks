\subsection{Difference Amplifier}\label{sec:amp_difference}

\subsubsection{Overview}
Inverting path \(V_1 \to G_1 \to V_-\) with feedback \(G_2\), non-inverting
path \(V_2 \to G_3 \to V_+\) with \(G_4\) to ground. Op-amp model as in
Section~\ref{sec:amp_inverting}: \(V_{out} = A(V_+ - V_-)\), \(A = A_0\) in Q5,
\(A = A(s) = \frac{A_0 p_1}{s + p_1}\) in Q4.

\subsubsection{Schematic}
\begin{circuitfig}
  \draw (0,0) node[op amp](A){};
  \draw (A.-) -- ++(-0.6,0) coordinate(n) node[circ]{} to[european resistor, l_={$G_1$}] ++(-2.4,0) node[circ]{} node[left]{$V_1$};
  \draw (n) |- (-0.5,1.6) to[european resistor, l={$G_2$}] (1.6,1.6) -| (A.out);
  \draw (A.+) -- ++(-0.6,0) coordinate(p) node[circ]{} to[european resistor, l={$G_3$}] ++(-2.4,0) node[circ]{} node[left]{$V_2$};
  \draw (p) to[european resistor, l_={$G_4$}] ++(0,-2) node[ground]{};
  \draw (A.out) to[short, -o] ++(0.8,0) node[right]{$V_{out}$};
\end{circuitfig}

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}: \(A = A_0\), no
    current into the op-amp inputs. Applying KCL at \(V_+\):
    \begin{align*}
      (V_+ - V_2) G_3 + (V_+ - 0) G_4 &= 0 \hspace{1cm} \Rightarrow \hspace{1cm}
      V_+ = \frac{G_3}{G_3 + G_4} V_2 = F_+ V_2 \hspace{1cm} \left[F_+ = \tfrac{G_3}{G_3 + G_4}\right] \tag{1}
    \end{align*}
    Applying KCL at \(V_-\):
    \begin{align*}
      (V_- - V_1) G_1 + (V_- - V_{OUT}) G_2 &= 0 \\
      V_- &= \frac{G_1}{G_1 + G_2} V_1 + \frac{G_2}{G_1 + G_2} V_{OUT} \\
      V_- &= (1 - F) V_1 + F V_{OUT} \hspace{1cm} \left[F = \tfrac{G_2}{G_1 + G_2}\right] \tag{2}
    \end{align*}
    \begin{align*}
      V_{OUT} &= A_0\left(F_+ V_2 - (1 - F) V_1 - F V_{OUT}\right) \hspace{1cm} [\text{from (1), (2)}] \\
      V_{OUT}(1 + A_0 F) &= A_0\left(F_+ V_2 - (1 - F) V_1\right) \\
      V_{OUT} &= \frac{A_0}{1 + A_0 F}\left(F_+ V_2 - (1 - F) V_1\right) \tag{3}
    \end{align*}
    \(A_0 \to \infty\) (\(\frac{A_0}{1 + A_0 F} \to \frac{1}{F}\)):
    \begin{align*}
      V_{OUT} &= \underbrace{\frac{F_+}{F}}_{a} V_2 - \underbrace{\frac{1 - F}{F}}_{b} V_1 \\
      a &= \frac{G_3}{G_3 + G_4}\cdot\frac{G_1 + G_2}{G_2} = \frac{\frac{1}{R_3}\cdot\frac{R_1 + R_2}{R_1 R_2}}{\frac{R_3 + R_4}{R_3 R_4}\cdot\frac{1}{R_2}} = \frac{R_4 (R_1 + R_2)}{R_1 (R_3 + R_4)} \\
      b &= \frac{G_1}{G_1 + G_2}\cdot\frac{G_1 + G_2}{G_2} = \frac{G_1}{G_2} = \frac{R_2}{R_1}
    \end{align*}
    Matched ratios \(\frac{R_4}{R_3} = \frac{R_2}{R_1} = k\), i.e.\ \(R_4 = k R_3\), \(R_2 = k R_1\):
    \begin{align*}
      a &= \frac{k R_3 (R_1 + k R_1)}{R_1 (R_3 + k R_3)} = \frac{k R_1 R_3 (1 + k)}{R_1 R_3 (1 + k)} = k = b \\
      V_{OUT} &= k (V_2 - V_1) = \frac{R_2}{R_1}(V_2 - V_1) \tag{4}
    \end{align*}
    valid while \(V_{OL} \leq V_{OUT} \leq V_{OH}\) and \(V_+ = F_+ V_2\) is inside the input common mode range.
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: (3) with
    \(A_0 \to A(s)\):
    \begin{align*}
      V_{out} &= \frac{\frac{A_0 p_1}{s + p_1}}{1 + \frac{A_0 p_1 F}{s + p_1}}\left(F_+ V_2 - (1 - F) V_1\right) \hspace{1cm} \left[A(s) = \tfrac{A_0 p_1}{s + p_1}\right] \\
      V_{out} &= \frac{A_0 p_1}{s + p_1(1 + A_0 F)}\left(F_+ V_2 - (1 - F) V_1\right) \tag{5}
    \end{align*}
    Matched (\(F_+ = \frac{R_4}{R_3 + R_4} = \frac{k R_3}{R_3(1 + k)} = \frac{k}{1 + k}\),
    \(1 - F = \frac{R_2}{R_1 + R_2} = \frac{k}{1 + k}\), so \(F_+ = 1 - F\)):
    \begin{align*}
      H_d(s) = \frac{V_{out}}{V_2 - V_1} &= (1 - F)\frac{A_0 p_1}{s + p_1(1 + A_0 F)} \tag{6}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: \(s_p = -p_1(1 + A_0 F)\), \(f_p \approx F \cdot GB = \frac{GB}{1 + k}\) [\(A_0 F \gg 1\)].
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(N(s) = (1 - F) A_0 p_1 \neq 0\): no finite zero.
  \item \textbf{DC Gain (\(H_d(0)\))}: \(s = 0\) in (6),
    \begin{align*}
      H_d(0) &= (1 - F)\frac{A_0}{1 + A_0 F} = \frac{1 - F}{F}\cdot\frac{A_0 F}{1 + A_0 F} \xrightarrow{A_0 \to \infty} \frac{1 - F}{F} = k
    \end{align*}
  \item \textbf{Gain-Bandwidth Product (\(GBW\))}:
    \begin{align*}
      GBW &= |H_d(0)| \cdot f_p = (1 - F)\frac{A_0}{1 + A_0 F}\cdot\frac{p_1(1 + A_0 F)}{2\pi} = (1 - F)\, GB = \frac{k}{1 + k} GB
    \end{align*}
  \item \textbf{Feedback Factor (\(F\))}: from (2) with \(V_1 = 0\),
    \(F = \frac{G_2}{G_1 + G_2} = \frac{R_1}{R_1 + R_2} = \frac{1}{1 + k}\).
  \item \textbf{Common Mode Rejection (\(CMRR\))}: \(V_1 = V_{ic} - \frac{V_{id}}{2}\),
    \(V_2 = V_{ic} + \frac{V_{id}}{2}\) in \(V_{OUT} = a V_2 - b V_1\):
    \begin{align*}
      V_{OUT} &= a\left(V_{ic} + \frac{V_{id}}{2}\right) - b\left(V_{ic} - \frac{V_{id}}{2}\right)
               = (a - b) V_{ic} + \frac{a + b}{2} V_{id} \\
      A_c &= a - b \hspace{1cm} A_d = \frac{a + b}{2} \hspace{1cm}
      CMRR = \frac{|A_d|}{|A_c|} = \frac{a + b}{2|a - b|}
    \end{align*}
    \(R_4\) off by a fraction \(e\): \(\frac{R_4}{R_3} \to k(1 + e)\), \(b = k\);
    \(a\) with top and bottom divided by \(R_1 R_3\):
    \begin{align*}
      a &= \frac{\frac{R_1 + R_2}{R_1}\cdot\frac{R_4}{R_3}}{\frac{R_3 + R_4}{R_3}} = \frac{(1 + k)\, k(1 + e)}{1 + k(1 + e)} \\
      1 + k(1 + e) &= (1 + k)\left(1 + \frac{ke}{1 + k}\right) \hspace{1cm} \Rightarrow \hspace{1cm} a = \frac{k(1 + e)}{1 + \frac{ke}{1 + k}} \\
      a &\approx k(1 + e)\left(1 - \frac{ke}{1 + k}\right) \hspace{1cm} [\tfrac{1}{1 + x} \approx 1 - x, \ |x| \ll 1] \\
      a &\approx k\left(1 + e - \frac{ke}{1 + k}\right) = k\left(1 + \frac{e}{1 + k}\right) \hspace{1cm} [e^2 \to 0] \\
      a - b &\approx \frac{k e}{1 + k} \hspace{1cm} \frac{a + b}{2} \approx k \hspace{1cm} \Rightarrow \hspace{1cm}
      CMRR \approx \frac{k}{\frac{k|e|}{1 + k}} = \frac{1 + k}{|e|}
    \end{align*}
  \item \textbf{Impedances}:
    \paragraph{At Input:} into \(V_1\) with \(V_2 = 0\): \(V_+ = 0\), so \(V_- \to 0\) and
    \(Z_{in,1} = R_1\). Into \(V_2\): no current enters \(V_+\), \(G_3\) and \(G_4\) in series,
    \(Z_{in,2} = R_3 + R_4\). Unequal, so a source impedance on either input unbalances
    the ratios (lowers \(CMRR\)).
    \paragraph{At Output:} as Section~\ref{sec:amp_inverting}, \(Z_{out} = \frac{R_o}{1 + AF}\).
  \item \textbf{Output Swing (Max)}: \(V_{OL} \leq k(V_2 - V_1) \leq V_{OH}\) and
    \(V_{ICM,\min} \leq \frac{k}{1 + k} V_2 \leq V_{ICM,\max}\).
  \item \textbf{Source Resistance}: a source resistance \(R_s\) driving \(V_1\) adds to \(R_1\),
    \(R_1 \to R_1 + R_s\), i.e.\ a ratio error \(\frac{R_2}{R_1 + R_s} = \frac{R_2}{R_1}\cdot\frac{1}{1 + R_s/R_1} \approx k\left(1 - \frac{R_s}{R_1}\right)\)
    [\(\frac{1}{1 + x} \approx 1 - x\)]: the same as a mismatch \(|e| = \frac{R_s}{R_1}\), so
    \(CMRR \approx \frac{(1 + k) R_1}{R_s}\); e.g.\ \(k = 1\), \(R_1 = 10\,k\Omega\), \(R_s = 100\,\Omega\): \(CMRR \approx \frac{2 \cdot 10^4}{100} = 200\).
  \item \textbf{Offset}: \(V_{OS}\) in series with \(V_+\) sees the noise gain \(\frac{1}{F} = 1 + k\):
    \(V_{OUT,error} = (1 + k) V_{OS}\).
  \item \textbf{Advantages \& Limitations}: single op-amp subtraction with common mode
    rejection; but \(CMRR\) is set by resistor matching (\(\frac{1 + k}{|e|}\)) and the two
    input impedances are low and unequal (the instrumentation amplifier buffers both
    inputs first).
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(k = \frac{R_2}{R_1} = \frac{R_4}{R_3}\) for the differential gain.
  \item Resistor mismatch \(|e| \leq \frac{1 + k}{CMRR}\).
  \item Source impedances \(\ll R_1\), \(R_3 + R_4\), or buffer the inputs.
\end{enumerate}

\subsubsection{Question}
\(k = 1\) (\(R_1 = R_2 = R_3 = 10\,k\Omega\)) with \(R_4 = 10.1\,k\Omega\) (\(e = 1\%\)).
Find the CMRR, approximate and exact.
\paragraph{Solution:}
\begin{align*}
  CMRR &\approx \frac{1 + k}{|e|} = \frac{1 + 1}{0.01} = 200 \hspace{1cm} (20\log_{10} 200 = 46\,dB) \\
  a &= \frac{R_4 (R_1 + R_2)}{R_1 (R_3 + R_4)} = \frac{10.1 \cdot 20}{10 \cdot 20.1} = \frac{202}{201} = 1.004975 \hspace{1cm} b = 1 \\
  CMRR &= \frac{a + b}{2|a - b|} = \frac{2.004975}{2 \cdot 0.004975} = \frac{2.004975}{0.009950} = 201.5
\end{align*}
