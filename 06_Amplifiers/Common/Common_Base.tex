\subsection{Common Base (CB)}\label{sec:amp_cb}

\subsubsection{Overview}
The BJT counterpart of the common gate stage: input at the emitter,
output at the collector, base at AC ground. Low input impedance
(\(\approx \frac{1}{g_m}\)), current gain \(A_F \approx 1\), non-inverting
voltage gain.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}: \(I_E = \frac{V_E - V_{IN}}{R_S}\),
    \(V_E = V_B - V_{BE} = V_B - V_T \ln\frac{I_C}{I_S}\), \(I_C = A_F I_E\),
    \(V_{OUT} = V_{cc} - I_C R_C\). Substituting \(V_E\) into \(I_E\):
    \begin{align*}
      \frac{I_C}{A_F} &= \frac{V_B - V_T \ln\frac{I_C}{I_S} - V_{IN}}{R_S} \\
      I_C^{(k+1)} &= \frac{A_F}{R_S}\left(V_B - V_{IN} - V_T \ln\frac{I_C^{(k)}}{I_S}\right)
    \end{align*}
    iterated from \(V_{BE} = 0.7\,\)V as for the CE stage.
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: \(V_b = 0\), \(g_o\) neglected,
    \(I_c = -g_m V_e\), \(I_b = -g_{be} V_e\).\\
    Applying KCL at \(V_e\) (\(I_e = I_b + I_c\) leaves the device into the node):
    \begin{align*}
      (V_e - V_{in}) G_S &= I_b + I_c = -(g_m + g_{be}) V_e \\
      V_e G_S + (g_m + g_{be}) V_e &= G_S V_{in} \\
      V_e &= \frac{G_S}{G_S + g_m + g_{be}} V_{in}
    \end{align*}
    Applying KCL at \(V_{out}\):
    \begin{align*}
      V_{out} (G_C + sC_L) + I_c &= 0 \hspace{0.5cm} \Rightarrow \hspace{0.5cm} V_{out} = \frac{g_m}{G_C + sC_L} V_e \hspace{0.5cm} [I_c = -g_m V_e] \\
      H(s) = \frac{V_{out}}{V_{in}} &= \frac{g_m}{G_C + sC_L}\cdot\frac{G_S}{G_S + g_m + g_{be}} \tag{1}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: \(G_C + s_p C_L = 0 \Rightarrow s_p = -\frac{G_C}{C_L}\),
    \(f_p = \frac{1}{2\pi R_C C_L}\). \(C_\mu\) goes to AC ground (base), so there is no Miller
    effect; the emitter pole \(\frac{G_S + g_m + g_{be}}{C_\pi}\) is far higher.
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(N(s) = g_m G_S \neq 0\): no finite zero.
  \item \textbf{DC Gain (\(H(0)\))}: \(s = 0\) in (1), \(G_S \to \infty\) (top and bottom divided by \(G_S\)):
    \begin{align*}
      A_{V0} &= \lim_{G_S \to \infty} \frac{g_m R_C}{1 + \frac{g_m + g_{be}}{G_S}} = g_m R_C
    \end{align*}
    Current gain (\(I_c = -g_m V_e\); current into the emitter from outside \(-I_e = (g_m + g_{be}) V_e\)):
    \begin{align*}
      \frac{I_c}{-I_e} &= \frac{-g_m V_e}{-(g_m + g_{be}) V_e} = \frac{g_m}{g_m + \frac{g_m}{B_F}} = \frac{B_F}{B_F + 1} = A_F \hspace{1cm} [g_{be} = \tfrac{g_m}{B_F}]
    \end{align*}
  \item \textbf{Gain-Bandwidth Product (\(GBW\))}: \(G_S \to \infty\),
    \(GBW = g_m R_C \cdot \frac{1}{2\pi R_C C_L} = \frac{g_m}{2\pi C_L}\).
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(G_S\) removed, \(I_{test}\) into the emitter: \(I_{test} + I_b + I_c = 0\),
    so \(I_{test} = (g_m + g_{be}) V_e\):
    \begin{align*}
      Z_{in} &= \frac{V_e}{I_{test}} = \frac{1}{g_m + g_{be}} = \frac{1}{g_m}\cdot\frac{g_m}{g_m + g_{be}} = \frac{A_F}{g_m}
    \end{align*}
    \paragraph{At Output:} the CE output test (Section~\ref{sec:amp_ce}) with \(R_E\) replaced
    by \(R_S\) (\(V_{in} = 0\), base at AC ground):
    \begin{align*}
      Z_{out} &= R_C \parallel r_o\left(1 + (g_m + g_o)(R_S \parallel r_{be})\right) \approx R_C \parallel r_o\left(1 + g_m (R_S \parallel r_{be})\right)
    \end{align*}
  \item \textbf{Output Swing (Max)}: forward active \(V_{CB} \gtrsim -0.4\,\)V:
    \(V_B - 0.4\,V \lesssim V_{OUT} \leq V_{cc}\).
  \item \textbf{Advantages \& Limitations}: low \(Z_{in} = \frac{A_F}{g_m}\) (current buffer,
    gain \(A_F\)), high \(Z_{out}\), no Miller effect; but a voltage source with \(R_S\) is
    attenuated by \(\frac{G_S}{G_S + g_m + g_{be}}\), and the emitter carries the full signal current.
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item Choose \(I_C\): \(Z_{in} = \frac{A_F V_T}{I_C}\).
  \item \(R_C = \frac{V_{cc} - V_{OUT}}{I_C}\), \(V_B = V_E + V_T\ln\frac{I_C}{I_S}\).
\end{enumerate}

\subsubsection{Question}
\(I_C = 1\,\)mA, \(B_F = 100\), \(R_C = 5\,k\Omega\), \(R_S = 50\,\Omega\),
\(V_T = 26\,\)mV. Find \(Z_{in}\) and \(A_V\).
\paragraph{Solution:}
\begin{align*}
  g_m &= \frac{1\,mA}{26\,mV} = 38.46\,mS \hspace{1cm} g_{be} = \frac{38.46\,mS}{100} = 0.385\,mS \hspace{1cm} G_S = \frac{1}{50\,\Omega} = 20\,mS \\
  g_m + g_{be} &= 38.46\,mS + 0.385\,mS = 38.85\,mS \hspace{1cm} Z_{in} = \frac{1}{38.85\,mS} = 25.7\,\Omega \\
  A_V &= \frac{(38.46\,mS)(5\,k\Omega)(20\,mS)}{20\,mS + 38.85\,mS} = 192.3 \cdot \frac{20}{58.85} = 65.4
\end{align*}
