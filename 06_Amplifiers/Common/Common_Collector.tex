\subsection{Common Collector (CC, Emitter Follower)}\label{sec:amp_cc}

\subsubsection{Overview}
The BJT counterpart of the source follower: input at the base, output at
the emitter, collector at AC ground. Gain just below one, input impedance
multiplied by \(B_F + 1\), output impedance divided by \(B_F + 1\).

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}:
    KVL from the source through \(R_S\) and the base-emitter junction (forward
    active, \(V_{BE} = V_T \ln\frac{I_C}{I_S}\)):
    \begin{align*}
      V_{OUT} &= V_{IN} - I_B R_S - V_{BE} = V_{IN} - I_B R_S - V_T \ln\frac{I_C}{I_S} \\
      I_E &= \frac{V_{OUT}}{R_E} \hspace{1cm} I_C = A_F I_E = A_F \frac{V_{OUT}}{R_E} \hspace{1cm} I_B = \frac{I_C}{B_F} = \frac{A_F V_{OUT}}{B_F R_E} \\
      V_{OUT} &= V_{IN} - \frac{A_F V_{OUT}}{B_F R_E} R_S - V_T \ln\frac{A_F V_{OUT}}{I_S R_E} \hspace{1cm} \text{(Substituting)} \\
      V_{OUT}\left(1 + \frac{A_F R_S}{B_F R_E}\right) &= V_{IN} - V_T \ln\frac{A_F V_{OUT}}{I_S R_E} \\
      V_{OUT} &= \frac{V_{IN} - V_T \ln\frac{A_F V_{OUT}}{I_S R_E}}{1 + \frac{R_S}{(B_F + 1) R_E}} \hspace{1cm} [\tfrac{A_F}{B_F} = \tfrac{1}{B_F + 1}]
    \end{align*}
    solved by iteration starting from \(V_{BE} = 0.7\,\)V (the logarithm moves
    only \(V_T \ln 10 = 60\,\)mV per decade of current): the output sits one
    \(V_{BE} \approx 0.7\,\)V below the base.
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}:
    \(g_o\) neglected, \(V_{cc}\) is AC ground, \(V_{be} = V_b - V_{out}\), \(Y_E = G_E + sC_L\).\\
    Applying KCL at \(V_b\) (\(I_b = g_{be} V_{be}\) leaves into the base):
    \begin{align*}
      (V_b - V_{in}) G_S + g_{be}(V_b - V_{out}) &= 0 \tag{1}
    \end{align*}
    Applying KCL at \(V_{out}\) (\(I_b + I_c\) arrives from the device):
    \begin{align*}
      V_{out} Y_E &= (g_{be} + g_m)(V_b - V_{out}) \tag{2} \\
      V_b &= V_{out}\frac{Y_E + g_m + g_{be}}{g_m + g_{be}} \hspace{1cm}
      V_b - V_{out} = \frac{Y_E}{g_m + g_{be}} V_{out} \tag{3}
    \end{align*}
    Substituting (3) into (1):
    \begin{align*}
      \left(V_{out}\frac{Y_E + g_m + g_{be}}{g_m + g_{be}} - V_{in}\right) G_S + \frac{g_{be} Y_E}{g_m + g_{be}} V_{out} &= 0 \\
      V_{out} G_S (Y_E + g_m + g_{be}) - V_{in} G_S (g_m + g_{be}) + g_{be} Y_E V_{out} &= 0 \\
      H(s) = \frac{V_{out}}{V_{in}} &= \frac{G_S (g_m + g_{be})}{G_S (Y_E + g_m + g_{be}) + g_{be} Y_E} \tag{4}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: \(Y_E = G_E + s_p C_L\) in (4),
    \begin{align*}
      G_S (G_E + g_m + g_{be}) + g_{be} G_E + s_p C_L (G_S + g_{be}) &= 0 \\
      s_p &= -\frac{G_S (G_E + g_m + g_{be}) + g_{be} G_E}{C_L (G_S + g_{be})} \\
      s_p &\xrightarrow{G_S \to \infty} -\frac{g_m + g_{be} + G_E}{C_L} \approx -\frac{g_m}{C_L}
    \end{align*}
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(N(s) = G_S (g_m + g_{be}) \neq 0\): no finite zero (\(C_\pi\) neglected).
  \item \textbf{DC Gain (\(H(0)\))}: \(s = 0\) (\(Y_E = G_E\)) in (4), top and bottom multiplied by
    \(R_S r_{be} R_E\) [\(G_S R_S = g_{be} r_{be} = G_E R_E = 1\), \((g_m + g_{be}) r_{be} = B_F + 1\)]:
    \begin{align*}
      H(0) &= \frac{G_S R_S (g_m + g_{be}) r_{be} R_E}{G_S R_S \left(G_E R_E r_{be} + (g_m + g_{be}) r_{be} R_E\right) + g_{be} r_{be} G_E R_E R_S} \\
      H(0) &= \frac{(B_F + 1) R_E}{R_S + r_{be} + (B_F + 1) R_E}
    \end{align*}
    Ideal source (\(R_S = 0\)):
    \(A_{V0} = \frac{(B_F + 1) R_E}{(B_F + 1) R_E + r_{be}} \to 1\) for \((B_F + 1) R_E \gg r_{be}\).
  \item \textbf{Gain-Bandwidth Product (\(GBW\))}: \(G_S \to \infty\),
    \begin{align*}
      GBW &= A_{V0} \frac{|s_p|}{2\pi} = \frac{g_m + g_{be}}{g_m + g_{be} + G_E}\cdot\frac{g_m + g_{be} + G_E}{2\pi C_L} = \frac{g_m + g_{be}}{2\pi C_L}
    \end{align*}
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(s = 0\), from (3) and (2),
    \(V_b - V_{out} = \frac{G_E}{g_m + g_{be}}\cdot\frac{g_m + g_{be}}{G_E + g_m + g_{be}} V_b = \frac{G_E}{G_E + g_m + g_{be}} V_b\):
    \begin{align*}
      I_b &= g_{be}(V_b - V_{out}) = \frac{g_{be} G_E}{G_E + g_m + g_{be}} V_b \\
      Z_{in} &= \frac{V_b}{I_b} = \frac{G_E + g_m + g_{be}}{g_{be} G_E} = \frac{1}{g_{be}} + \left(\frac{g_m}{g_{be}} + 1\right) R_E = r_{be} + (B_F + 1) R_E
    \end{align*}
    \paragraph{At Output:} \(V_{in} = 0\), \(G_E\) removed, \(I_{test}\) into the emitter. (1) gives
    \(V_b = \frac{g_{be}}{G_S + g_{be}} V_{out}\); the device current into the node is
    \((g_m + g_{be})(V_b - V_{out})\):
    \begin{align*}
      I_{test} + (g_m + g_{be})\left(\frac{g_{be}}{G_S + g_{be}} - 1\right) V_{out} &= 0 \\
      I_{test} &= \frac{(g_m + g_{be}) G_S}{G_S + g_{be}} V_{out} \\
      Z_{out} &= \frac{G_S + g_{be}}{(g_m + g_{be}) G_S} = \frac{(G_S + g_{be}) R_S r_{be}}{(g_m + g_{be}) G_S R_S r_{be}} = \frac{R_S + r_{be}}{B_F + 1}
    \end{align*}
  \item \textbf{Output Swing (Max)}: \(V_{OUT} = V_{IN} - V_{BE} - I_B R_S\); forward active
    \(V_{CE} = V_{cc} - V_{OUT} \gtrsim 0.3\,V\): \(0 \leq V_{OUT} \lesssim V_{cc} - 0.3\,V\).
  \item \textbf{Advantages \& Limitations}: \(Z_{in}\) multiplied and \(Z_{out}\) divided by
    \(B_F + 1\), gain \(\approx 1\), wide bandwidth; but costs \(V_{BE} \approx 0.7\,V\) of headroom,
    gain \(< 1\), and \(Z_{out}\) depends on the source resistance \(R_S\).
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(I_C\) from \(Z_{out} = \frac{R_S + r_{be}}{B_F + 1}\), \(r_{be} = \frac{B_F V_T}{I_C}\).
  \item \(R_E = \frac{V_{OUT}}{I_E}\) with \(V_{OUT} \approx V_{IN} - 0.7\,V\).
\end{enumerate}

\subsubsection{Question}
\(I_C = 1\,\)mA, \(B_F = 100\), \(R_E = 1\,k\Omega\), \(R_S = 10\,k\Omega\),
\(V_T = 26\,\)mV. Find \(A_{V0}\) (ideal source), \(H\) (with \(R_S\)), \(Z_{in}\) and \(Z_{out}\).
\paragraph{Solution:}
\begin{align*}
  r_{be} &= \frac{B_F}{g_m} = \frac{B_F V_T}{I_C} = \frac{100 \cdot 26\,mV}{1\,mA} = 2.6\,k\Omega \hspace{1cm} (B_F + 1) R_E = 101 \cdot 1\,k\Omega = 101\,k\Omega \\
  A_{V0} &= \frac{(B_F + 1) R_E}{(B_F + 1) R_E + r_{be}} = \frac{101\,k}{2.6\,k + 101\,k} = \frac{101}{103.6} = 0.975 \\
  H &= \frac{(B_F + 1) R_E}{R_S + r_{be} + (B_F + 1) R_E} = \frac{101\,k}{10\,k + 2.6\,k + 101\,k} = \frac{101}{113.6} = 0.889 \\
  Z_{in} &= r_{be} + (B_F + 1) R_E = 2.6\,k + 101\,k = 103.6\,k\Omega \\
  Z_{out} &= \frac{R_S + r_{be}}{B_F + 1} = \frac{10\,k + 2.6\,k}{101} = \frac{12.6\,k}{101} = 125\,\Omega
\end{align*}
