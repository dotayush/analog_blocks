\subsection{Folded Cascode Op-Amp}\label{sec:amp_folded}

\subsubsection{Overview}
The input pair current is \emph{folded} into common gate devices of the
opposite type: a PMOS pair \(M_1\), \(M_2\) drives the sources of NMOS
cascodes \(M_3\), \(M_4\), each sitting on an NMOS current sink \(M_{B}\)
(conductance \(g_{oB}\)), loaded by PMOS cascodes \(M_5\)--\(M_8\). The input
and output stacks are separate, so the input common mode can reach a rail
and the output swing is larger than the telescopic cascode's.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}:
    \(M_B\) sinks \(I_B\); the pair supplies \(\frac{I_T}{2}\) into the fold node,
    so \(M_3\) carries \(I_B - \frac{I_T}{2}\) (\(I_B > I_T\) to avoid starving
    \(M_3\) during slewing). Swing:
    \(V_{o,\min} = V_{ov,B} + V_{ov3}\), \(V_{o,\max} = V_{dd} - |V_{ov5}| - |V_{ov7}|\).
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}:
    the fold node sees \(r_{o1} \parallel r_{oB}\) to AC ground. The cascode \(M_3\)
    is a common gate stage with source resistance \(R_S = r_{o1} \parallel r_{oB}\)
    (Section~\ref{sec:amp_cg}):
    \begin{align*}
      Z_{dn} &= r_{o3} + (r_{o1} \parallel r_{oB}) + g_{m3}' r_{o3} (r_{o1} \parallel r_{oB}) \\
      Z_{up} &= r_{o5} + r_{o7} + g_{m5}' r_{o5} r_{o7}
    \end{align*}
    The signal current \(\frac{g_{m1} V_{id}}{2}\) injected at the fold node
    splits between \(r_{o1} \parallel r_{oB}\) and the source of \(M_3\)
    (\(\approx \frac{1}{g_{m3}'}\)); the fraction entering \(M_3\) is
    \(\frac{g_{m3}'}{g_{m3}' + g_{o1} + g_{oB}} \approx 1\). Hence
    \[
      A_d = g_{m1} \frac{g_{m3}'}{g_{m3}' + g_{o1} + g_{oB}} (Z_{dn} \parallel Z_{up}) \approx g_{m1}\left[g_{m3} r_{o3}(r_{o1} \parallel r_{oB}) \parallel g_{m5} r_{o5} r_{o7}\right]
    \]
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Gain}: lower than telescopic by the factor \(r_{o1} \parallel r_{oB}\)
    instead of \(r_{o1}\) (about \(\frac{1}{2}\)--\(\frac{2}{3}\) of the telescopic gain).
  \item \textbf{Poles}: dominant \(p_1 = \frac{1}{(Z_{dn} \parallel Z_{up}) C_L}\),
    \(GB = \frac{g_{m1}}{2\pi C_L}\); non-dominant at the fold node
    \(p_2 \approx \frac{g_{m3}}{C_f}\), with \(C_f\) larger than in the telescopic
    case (it includes \(C_{db1}\), \(C_{db,B}\), \(C_{gs3}\)).
  \item \textbf{Power}: two current branches (\(I_T\) and \(2I_B\)) versus
    one; roughly twice the current for the same \(g_{m1}\).
  \item \textbf{Slew}: \(SR = \frac{I_T}{C_L}\) provided \(I_B \geq I_T\).
\end{enumerate}
