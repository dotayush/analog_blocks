\subsection{Cascode Amplifier}\label{sec:amp_cascode}

\subsubsection{Overview}
A common source \(M_1\) stacked under a common gate \(M_2\) (gate at DC
bias \(V_{B2}\)). \(M_2\) shields the drain of \(M_1\) from the output, which
multiplies the output impedance by \(\approx g_{m2} r_{o2}\) and removes the
Miller effect on \(C_{gd1}\). It is the half circuit of every cascoded op-amp
below.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}:
    \(I_{D1} = I_{D2}\); \(V_x = V_{B2} - V_{th} - V_{ov2}\). Both saturated needs
    \(V_x \geq V_{ov1}\) and \(V_{OUT} \geq V_x + V_{ov2}\), so
    \(V_{OUT,\min} = V_{ov1} + V_{ov2}\) (with \(V_{B2}\) chosen as
    \(V_{th} + V_{ov1} + V_{ov2}\)).
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}:
    \(I_{d1} = g_{m1} V_{in} + g_{o1} V_x\),
    \(I_{d2} = -(g_{m2} + g_{mb2}) V_x + g_{o2}(V_{out} - V_x)\); write
    \(g_{m2}' = g_{m2} + g_{mb2}\).\\
    KCL at \(V_x\) and \(V_{out}\):
    \begin{align*}
      I_{d1} &= I_{d2} \tag{1} \\
      V_{out} G_L + I_{d2} &= 0 \hspace{0.5cm} \Rightarrow \hspace{0.5cm} I_{d2} = -V_{out} G_L \tag{2}
    \end{align*}
    From (1), (2): \(g_{m1} V_{in} + g_{o1} V_x = -V_{out} G_L\), so
    \(V_x = -\frac{V_{out} G_L + g_{m1} V_{in}}{g_{o1}}\). From (2):
    \begin{align*}
      -(g_{m2}' + g_{o2}) V_x + g_{o2} V_{out} &= -V_{out} G_L \\
      (g_{m2}' + g_{o2}) V_x &= V_{out}(g_{o2} + G_L) \\
      -(g_{m2}' + g_{o2})\frac{V_{out} G_L + g_{m1} V_{in}}{g_{o1}} &= V_{out}(g_{o2} + G_L) \\
      -V_{out}\left[(g_{m2}' + g_{o2}) G_L + g_{o1}(g_{o2} + G_L)\right] &= (g_{m2}' + g_{o2}) g_{m1} V_{in}
    \end{align*}
    \[
      A_V = \frac{V_{out}}{V_{in}} = -\frac{g_{m1}(g_{m2}' + g_{o2})}{(g_{m2}' + g_{o2} + g_{o1}) G_L + g_{o1} g_{o2}}
    \]
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Output Impedance}: dividing numerator and denominator of
    \(A_V\) by \((g_{m2}' + g_{o2} + g_{o1})\), \(A_V = -G_m (Z_{dn} \parallel R_L)\) with
    \begin{align*}
      Z_{dn} &= \frac{g_{m2}' + g_{o2} + g_{o1}}{g_{o1} g_{o2}} = r_{o1} + r_{o2} + g_{m2}' r_{o1} r_{o2} \approx g_{m2} r_{o2} r_{o1} \\
      G_m &= \frac{g_{m1}(g_{m2}' + g_{o2})}{g_{m2}' + g_{o2} + g_{o1}} \approx g_{m1}
    \end{align*}
    which is the common gate result of Section~\ref{sec:amp_cg} with
    \(R_S = r_{o1}\).
  \item \textbf{Gain}: ideal current source load (\(G_L \to 0\)):
    \(A_{V0} = -\frac{g_{m1}(g_{m2}' + g_{o2})}{g_{o1} g_{o2}} \approx -(g_m r_o)^2\),
    the square of the intrinsic gain. With a simple current source load
    (\(G_L = g_{oL}\)) the gain falls back to \(-g_{m1} r_{oL}\): the load must be
    cascoded too (telescopic cascode).
  \item \textbf{Miller Effect}: the gain from the gate to the drain of \(M_1\)
    is \(\frac{V_x}{V_{in}} \approx -\frac{g_{m1}}{g_{m2}'}\approx -1\), so
    \(C_{gd1}\) is multiplied by only \(\approx 2\).
\end{enumerate}
