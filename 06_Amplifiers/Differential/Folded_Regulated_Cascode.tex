\subsection{Folded Regulated Cascode Op-Amp}\label{sec:amp_folded_regulated}

\subsubsection{Overview}
The folded cascode of Section~\ref{sec:amp_folded} with both cascodes
(\(M_3\) and \(M_5\)) gain boosted as in Section~\ref{sec:amp_regulated}
by auxiliary amplifiers \(A_{bn}\), \(A_{bp}\). Folding keeps the input
range, regulation recovers (and exceeds) the telescopic gain.

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}: the
    currents are those of the folded cascode. The loops set \(V_F = V_{refn}\)
    and \(V_U = V_{refp}\); with \(V_{refn} = V_{ov,B}\) and
    \(V_{dd} - V_{refp} = |V_{ov7}|\) the swing is the folded cascode's,
    \(V_{O,\min} = V_{ov,B} + V_{ov3}\), \(V_{O,\max} = V_{dd} - |V_{ov7}| - |V_{ov5}|\).
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}: the
    references are fixed, so \(V_{g3} = A_{bn}(0 - V_f) = -A_{bn} V_f\) (the fold node
    sits at the inverting input of \(A_{bn}\)) and \(V_{g5} = -A_{bp} V_u\). Body
    effect is neglected (\(g_{mb} = 0\)), as in Section~\ref{sec:amp_regulated}.\\
    \textbf{Output Impedance Down} (\(I_{in} = 0\), test current \(I_t\) down through
    \(M_3\) and \(R_f = r_{o1} \parallel r_{oB}\), so \(V_f = I_t R_f\)):
    \begin{align*}
      I_t &= g_{m3}(V_{g3} - V_f) + g_{o3}(V_t - V_f) = g_{m3}(-A_{bn} V_f - V_f) + g_{o3} V_t - g_{o3} V_f \\
      I_t &= -\left(g_{m3}(1 + A_{bn}) + g_{o3}\right) I_t R_f + g_{o3} V_t \\
      I_t\left(1 + g_{m3}(1 + A_{bn}) R_f + g_{o3} R_f\right) &= g_{o3} V_t \\
      Z_{dn} &= \frac{V_t}{I_t} = r_{o3} + g_{m3}(1 + A_{bn}) r_{o3} R_f + R_f
    \end{align*}
    the folded cascode \(Z_{dn}\) with \(g_{m3} \to g_{m3}(1 + A_{bn})\).
    \textbf{Output Impedance Up}: the same steps for \(M_5\) over \(r_{o7}\):
    \(Z_{up} = r_{o5} + g_{m5}(1 + A_{bp}) r_{o5} r_{o7} + r_{o7}\).\\
    \textbf{Short Circuit Current} (\(V_{o1} = 0\)): \(I_{d3} = g_{m3}(-A_{bn} V_f - V_f) + g_{o3}(0 - V_f) = -\left(g_{m3}(1 + A_{bn}) + g_{o3}\right) V_f\);
    KCL at \(V_f\) as in Section~\ref{sec:amp_folded}:
    \begin{align*}
      |I_{sc}| &= \frac{g_{m3}(1 + A_{bn}) + g_{o3}}{g_{m3}(1 + A_{bn}) + g_{o3} + g_{o1} + g_{oB}} I_{in} \approx I_{in}
    \end{align*}
    \textbf{Gain}: with \(I_{in} = \frac{g_{m1} V_{id}}{2}\) and \(V_{o2} = -V_{o1}\) as before,
    \begin{align*}
      |A_d| &= \frac{2 |I_{sc}| (Z_{dn} \parallel Z_{up})}{V_{id}} \approx g_{m1}(Z_{dn} \parallel Z_{up})
             \approx g_{m1}\left[g_{m3} A_{bn} r_{o3} R_f \parallel g_{m5} A_{bp} r_{o5} r_{o7}\right]
    \end{align*}
    keeping the dominant term of each impedance (\(1 + A_b \approx A_b\)). With \(C_L\) at each output:
    \begin{align*}
      A_d(s) &= \frac{A_d}{1 + s (Z_{dn} \parallel Z_{up}) C_L} \tag{1}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: from (1),
    \(s_{p1} = -\frac{1}{(Z_{dn} \parallel Z_{up}) C_L}\) (dominant); non-dominant at the fold node
    \(\approx -\frac{g_{m3}}{C_f}\), plus the auxiliary doublets (Section~\ref{sec:amp_regulated}).
  \item \textbf{Zeroes (\(N(s) = 0\))}: none from (1); each auxiliary loop adds a zero at
    \(\approx \omega_b\) cancelling its doublet pole (Section~\ref{sec:amp_regulated}).
  \item \textbf{DC Gain (\(A_d(0)\))}: every \(g_m r_o\) equal, \(R_f \approx r_o\), \(A_{bn} = A_{bp} = A_b\):
    \begin{align*}
      Z_{dn} \approx Z_{up} &\approx g_m r_o \cdot A_b \cdot r_o \hspace{1cm}
      Z_{dn} \parallel Z_{up} \approx \frac{g_m r_o A_b r_o}{2} \\
      |A_d| &\approx g_m \cdot \frac{g_m r_o A_b r_o}{2} = \frac{(g_m r_o)^2 A_b}{2}
    \end{align*}
    single stage auxiliary amplifiers \(A_b \approx g_m r_o\): \(|A_d| \approx \frac{(g_m r_o)^3}{2}\); e.g. \(g_m r_o = 50\):
    \(\frac{50^3}{2} = 62500\) (\(20\log_{10} 62500 = 96\,dB\)) in a single stage.
  \item \textbf{Gain-Bandwidth Product (\(GB\))}:
    \(GB = |A_d| \frac{|s_{p1}|}{2\pi} = g_{m1}(Z_{dn} \parallel Z_{up})\cdot\frac{1}{2\pi(Z_{dn} \parallel Z_{up}) C_L} = \frac{g_{m1}}{2\pi C_L}\),
    unchanged. The auxiliary unity gain frequencies must lie between \(2\pi GB\) and the main
    non-dominant pole (Section~\ref{sec:amp_regulated}).
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(I_G = 0\): \(Z_{in} \to \infty\) at DC.
    \paragraph{At Output:} \(Z_{out} = Z_{dn} \parallel Z_{up} \approx \frac{g_m r_o A_b r_o}{2}\) per side.
  \item \textbf{Output Swing (Max)}: from Q5, \(V_{ov,B} + V_{ov3} \leq V_O \leq V_{dd} - |V_{ov7}| - |V_{ov5}|\).
  \item \textbf{Advantages \& Limitations}: telescopic level gain (or more) with the folded input
    range and swing; but two auxiliary amplifiers (power, area), doublets that must be placed
    correctly for settling, and \(I_{dd} \geq 2 I_T\) as the folded cascode.
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item The folded cascode design (Section~\ref{sec:amp_folded}) for \(SR\), \(GB\), swing.
  \item \(A_b\) from the required gain: \(A_b \approx \frac{2 |A_d|}{(g_m r_o)^2}\).
  \item \(V_{refn} = V_{ov,B}\), \(V_{dd} - V_{refp} = |V_{ov7}|\) plus margin.
\end{enumerate}

\subsubsection{Question}
\(g_m = 1\,mS\), \(r_o = 100\,k\Omega\), \(R_f \approx r_o\), \(A_b = 50\), \(C_L = 1\,pF\). Find \(|A_d|\),
\(f_{p1}\) and \(GB\).
\paragraph{Solution:}
\begin{align*}
  Z_{dn} \approx Z_{up} &\approx g_m r_o A_b r_o = 10^{-3} \cdot 10^{5} \cdot 50 \cdot 10^{5} = 500\,M\Omega \hspace{1cm}
  Z_{dn} \parallel Z_{up} = 250\,M\Omega \\
  |A_d| &= g_m (Z_{dn} \parallel Z_{up}) = 10^{-3} \cdot 2.5 \times 10^{8} = 250000 \hspace{1cm} (108\,dB) \\
  f_{p1} &= \frac{1}{2\pi \cdot 2.5 \times 10^{8}\,\Omega \cdot 10^{-12}\,F} = \frac{1}{1.571 \times 10^{-3}\,s} = 637\,Hz \\
  GB &= \frac{g_m}{2\pi C_L} = \frac{10^{-3}}{2\pi \cdot 10^{-12}}\,Hz = 159\,MHz \hspace{1cm} [= 250000 \cdot 637\,Hz]
\end{align*}
