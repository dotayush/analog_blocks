\subsection{Regulated Cascode (Gain Boosting)}\label{sec:amp_regulated}

\subsubsection{Overview}
The gate of the cascode device \(M_2\) is driven by an auxiliary amplifier of
gain \(A_b\) that senses \(V_x\) and holds it constant. Any change of \(V_{out}\)
that would move \(V_x\) is fought by \(A_b\), so the output impedance is
multiplied by \(1 + A_b\) without stacking more devices (no swing cost).

\subsubsection{Transfer Function}
\begin{enumerate}
  \item \textbf{Large Signal, Time Domain, DC Input Analysis (Q5)}:
    the loop sets \(V_X = V_{ref}\) (\(\geq V_{ov1}\) for \(M_1\) saturated);
    \(V_{G2} = V_X + V_{GS2}\); \(M_2\) saturated needs \(V_{OUT} - V_X \geq V_{ov2}\), so
    \(V_{OUT,\min} = V_{ref} + V_{ov2}\), the same headroom as a plain cascode
    when \(V_{ref} = V_{ov1}\).
  \item \textbf{Small Signal, Frequency Domain, AC Input Analysis (Q4)}:
    \(V_{ref}\) is fixed, so \(V_{g2} = A_b(0 - V_x) = -A_b V_x\); body effect
    neglected (\(g_{mb2} = 0\)). Output impedance (\(V_{in} = 0\), \(I_{test}\) into the
    drain of \(M_2\), which flows on through \(r_{o1}\)): \(V_x = \frac{I_{test}}{g_{o1}}\).
    \begin{align*}
      I_{test} &= g_{m2}(V_{g2} - V_x) + g_{o2}(V_{out} - V_x) \\
      I_{test} &= g_{m2}(-A_b V_x - V_x) + g_{o2} V_{out} - g_{o2} V_x = -g_{m2}(1 + A_b) V_x - g_{o2} V_x + g_{o2} V_{out} \\
      g_{o2} V_{out} &= I_{test} + \left(g_{m2}(1 + A_b) + g_{o2}\right)\frac{I_{test}}{g_{o1}} \\
      \frac{V_{out}}{I_{test}} &= \frac{1}{g_{o2}} + \frac{g_{m2}(1 + A_b)}{g_{o1} g_{o2}} + \frac{g_{o2}}{g_{o1} g_{o2}} \\
      Z_{out} &= \frac{V_{out}}{I_{test}} = r_{o2} + r_{o1} + g_{m2}(1 + A_b) r_{o1} r_{o2} \tag{1}
    \end{align*}
    the plain cascode \(Z_{dn}\) (Section~\ref{sec:amp_cascode}) with \(g_{m2} \to g_{m2}(1 + A_b)\).\\
    \textbf{Gain}: ideal load, \(C_L\) at the output, \(A_b(s) = \frac{A_{b0}}{1 + s/p_b}\) and
    \(R_c = g_{m2} r_{o2} r_{o1}\) (plain cascode value):
    \begin{align*}
      Z_{out}(s) &\approx R_c (1 + A_b(s)) = R_c \frac{1 + A_{b0} + s/p_b}{1 + s/p_b} \hspace{1cm} [\text{(1)}, g_{m2}(1 + A_b) r_{o1} r_{o2} \gg r_{o1} + r_{o2}] \\
      A(s) &= -\frac{g_{m1}}{\frac{1}{Z_{out}(s)} + sC_L} = -\frac{g_{m1} R_c (1 + A_{b0} + s/p_b)}{1 + s/p_b + sC_L R_c(1 + A_{b0} + s/p_b)} \\
      A(s) &= -\frac{g_{m1}}{C_L} \cdot \frac{s + (1 + A_{b0}) p_b}{s^2 + s\left((1 + A_{b0}) p_b + \frac{1}{R_c C_L}\right) + \frac{p_b}{R_c C_L}} \tag{2}
    \end{align*}
\end{enumerate}

\subsubsection{Analysis}
\begin{enumerate}
  \item \textbf{Poles (\(D(s) = 0\))}: sum \((1 + A_{b0}) p_b + \frac{1}{R_c C_L}\), product
    \(\frac{p_b}{R_c C_L}\); widely spaced, slow = product over sum, fast = sum:
    \begin{align*}
      p_1 &\approx \frac{p_b / (R_c C_L)}{(1 + A_{b0}) p_b} = \frac{1}{(1 + A_{b0}) R_c C_L} \hspace{1cm} \text{(dominant, boosted)} \\
      p_d &\approx (1 + A_{b0}) p_b + \frac{1}{R_c C_L}
    \end{align*}
  \item \textbf{Zeroes (\(N(s) = 0\))}: \(s_z = -(1 + A_{b0}) p_b\), \(z_d = (1 + A_{b0}) p_b \approx \omega_b\)
    (\(\omega_b = A_{b0} p_b\), the auxiliary unity gain frequency). \(p_d\) and \(z_d\) form a doublet
    at \(\approx \omega_b\), split by the plain cascode pole \(\frac{1}{R_c C_L}\): a slow settling tail
    \(\approx \frac{1}{\omega_b}\) unless \(\omega_b\) lies above the main closed loop bandwidth
    (\(2\pi GB\) in unity feedback); \(\omega_b\) must stay below the main non-dominant pole to keep
    the auxiliary loop stable.
  \item \textbf{DC Gain (\(A(0)\))}: \(s = 0\) in (2),
    \begin{align*}
      A(0) &= -\frac{g_{m1}}{C_L}\cdot\frac{(1 + A_{b0}) p_b}{\frac{p_b}{R_c C_L}} = -g_{m1} (1 + A_{b0}) R_c
    \end{align*}
    \(A_b = g_{mA} r_{oA} \approx g_m r_o\) (one CS stage) and an equally boosted load
    (\(Z_{up} \approx Z_{out}\), output \(\frac{Z_{out}}{2}\)):
    \(Z_{out} \approx (g_m r_o)^2 r_o\), \(A_{V0} \approx -g_m \frac{(g_m r_o)^2 r_o}{2} = -\frac{(g_m r_o)^3}{2}\).
  \item \textbf{Gain-Bandwidth Product (\(GB\))}: \(|A(0)| \frac{p_1}{2\pi} = g_{m1} (1 + A_{b0}) R_c \cdot \frac{1}{2\pi(1 + A_{b0}) R_c C_L} = \frac{g_{m1}}{2\pi C_L}\),
    unchanged by the boost.
  \item \textbf{Impedances}:
    \paragraph{At Input:} \(I_G = 0\): \(Z_{in} \to \infty\) at DC.
    \paragraph{At Output:} (1): \(Z_{out} \approx (1 + A_b) g_{m2} r_{o2} r_{o1}\), falling to \(R_c\) above \(\omega_b\).
  \item \textbf{Output Swing (Max)}: from Q5, \(V_{OUT} \geq V_{ref} + V_{ov2}\), as a plain cascode for
    \(V_{ref} = V_{ov1}\).
  \item \textbf{Auxiliary Amplifier Range}: \(A_b\) senses \(V_x = V_{ref}\) and drives \(V_{G2} = V_{ref} + V_{GS2}\), so its
    input common mode is \(V_{ref}\) (near ground: a PMOS input pair) and its output must reach \(V_{ref} + V_{th} + V_{ov2}\).
  \item \textbf{Advantages \& Limitations}: \(Z_{out}\) and gain multiplied by \(1 + A_b\) without
    stacking devices (no swing cost), \(GB\) unchanged; but the auxiliary amplifier costs power
    and area, and the doublet slows settling unless \(\omega_b\) is placed correctly.
\end{enumerate}

\subsubsection{Design}
\begin{enumerate}
  \item \(A_{b0}\) from the required gain: \(1 + A_{b0} = \frac{|A_{V0}|}{g_{m1} R_c}\).
  \item \(2\pi GB < \omega_b = A_{b0} p_b <\) main non-dominant pole.
  \item \(V_{ref} = V_{ov1}\) plus margin.
\end{enumerate}

\subsubsection{Question}
\(g_{m1} = g_{m2} = 1\,mS\), \(r_{o1} = r_{o2} = 100\,k\Omega\), \(A_b = 50\). Find
\(Z_{out}\) against the plain cascode, and the gain with an equally boosted load.
\paragraph{Solution:}
\begin{align*}
  Z_{out} &= r_{o2} + r_{o1} + g_{m2}(1 + A_b) r_{o1} r_{o2} = 100\,k + 100\,k + 10^{-3} \cdot 51 \cdot 10^{10} = 0.2\,M + 510\,M = 510.2\,M\Omega \\
  Z_{out,plain} &= r_{o2} + r_{o1} + g_{m2} r_{o1} r_{o2} = 0.2\,M + 10\,M = 10.2\,M\Omega \hspace{1cm}
  \frac{Z_{out}}{Z_{out,plain}} = \frac{510.2}{10.2} = 50.0 \approx 1 + A_b \\
  A_{V0} &\approx -g_{m1}\frac{Z_{out}}{2} = -10^{-3} \cdot \frac{510.2 \times 10^{6}}{2} = -255100 \hspace{1cm} (20\log_{10} 255100 = 108\,dB)
\end{align*}

\subsubsection{Code}
\begin{lstlisting}[language=Octave]
  clear; clc;
  pkg load symbolic;

  syms s vo It gm1 gm2 go1 go2 Ab Rc pb Ab0 CL
  % output impedance: Vx = It/go1, Vg2 = -Ab Vx
  vx = It/go1;
  vout = solve(It == gm2*(-Ab*vx - vx) + go2*(vo - vx), vo);
  printf('=== Zout, equation (1) ===\n'); pretty(expand(simplify(vout/It)))
  % gain with Ab(s) = Ab0/(1 + s/pb)
  Z = Rc*(1 + Ab0 + s/pb)/(1 + s/pb);
  A = simplify(-gm1/(1/Z + s*CL));
  printf('=== A(s), equation (2) ===\n'); pretty(A)
  printf('=== A(0) ===\n'); pretty(simplify(subs(A, s, 0)))

  % Question: gm = 1 mS, ro = 100k, Ab = 50
  Zo = 1e5 + 1e5 + 1e-3*51*1e10; Zc = 0.2e6 + 1e7;
  printf('Zout = %.1f M, plain = %.1f M, ratio = %.1f, Av0 = %.0f (%.0f dB)\n', ...
         Zo/1e6, Zc/1e6, Zo/Zc, 1e-3*Zo/2, 20*log10(1e-3*Zo/2));
\end{lstlisting}
